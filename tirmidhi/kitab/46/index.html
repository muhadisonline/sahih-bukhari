٤٨٤ حديثًا

باب ما جاء في الذي يفسر القرآن برأيه

٣١٨١من قال في القرْآنِ بِغيرِ عِلمٍ، فليتبوَّأ…٣١٨٢اتّقُوا الحديثَ عَنِّي إلا ما عَلِمتُم، فَمن…٣١٨٣من قال في القُرآنِ بِرأيِه فأَصابَ فقد أخطأَ٣١٨٤من صَلّى صلاةً لم يقرأْ فيها بِأُمِّ القُرآنِ،…٣١٨٥من صَلّى صلاةً لم يَقرأْ فيها بِأُمِّ القُرآنِ،…٣١٨٦إنِّي لأَرجُو أن يَجعلَ اللهُ يَدَه في يَدِي٣١٨٧اليهودُ مَغْضُوبٌ عليهم، والنَّصارَي ضُلَّالٌ٣١٨٨إنَّ اللهَ خَلَقَ آدَمَ من قَبضَةٍ قَبَضَها من…٣١٨٩دَخَلُوا مُتَزَحِّفِينَ على أوراكِهِم٣١٩٠قالوا حَبَّةٌ في شَعْرَةٍ٣١٩١حدَّثنا محمودُ بنُ غيلانَ، قال: حدَّثنا وكيعٌ،…٣١٩٢حدَّثنا عبدُ بنُ حُمَيدٍ، قال: أخبرنا يزِيدُ بنُ…٣١٩٣حدَّثنا بذلكَ محمدُ بنُ عبدِ الملكِ بنِ أبي…٣١٩٤حدَّثنا بذلكَ أبو كُرَيبٍ محمد بن العلاء، قال:…٣١٩٥حدَّثنا عبدُ بن حُمَيدٍ، قال: حدَّثنا الحجَّاجُ…٣١٩٦حدَّثنا أحمدُ بن مَنِيعٍ، قال: حدَّثنا هُشَيمٌ،…٣١٩٧عَدْلًا٣١٩٨يُدْعَى نُوحٌ، فَيقالُ: هل بَلَّغْتَ؟ فيقولُ: نعم،…٣١٩٩حدَّثنا محمدُ بنُ بشارٍ، قال: حدَّثنا جعفرُ بنُ…٣٢٠٠حدَّثنا هنَّادٌ، قال: حدَّثنا وكيعٌ، عن إسرائِيل،…٣٢٠١حدَّثنا هنَّادٌ، قال: حدَّثنا وكيعٌ، عن سفيانَ، عن…٣٢٠٢حدَّثنا هَنَّادٌ وأبو عمَّارٍ، قالا: حدَّثنا…٣٢٠٣حدَّثنا ابنُ أبي عمرَ، قال: حدَّثنا سفيانُ، قال:…٣٢٠٤حدَّثنا عبدُ بن حُمَيدٍ، قال: حدَّثنا يزيدُ بن أبي…٣٢٠٥حدَّثنا ابنُ أبي عمرَ، قال: حدَّثنا سفيانُ، عن…٣٢٠٦حدَّثنا عبدُ بنُ حُمَيدٍ، قال: حدَّثنا عُبَيدُ…٣٢٠٧الدُّعاءُ هو العِبادَةُ٣٢٠٨إنَّما ذلكَ بَياضُ النّهارِ من سَوادِ اللّيلِ٣٢٠٩حدَّثنا أحمدُ بنُ مَنِيعٍ، قال: حدَّثنا هُشَيمٌ،…٣٢١٠إنّما هو اللّيلُ والنّهارُ٣٢١١حدَّثنا عبدُ بنُ حُمَيدٍ، قال: حدَّثنا الضّحَّاكُ…٣٢١٢كأنَّ هَوَامَّ رَأسِكَ تُؤْذِيكَ؟٣٢١٣حدَّثنا عليُّ بنُ حُجْرٍ، قال: أخبرنا هُشَيمٌ، عن…٣٢١٤حدَّثنا عليُّ بنُ حُجْرٍ، قال: حدَّثنا هُشَيمٌ، عن…٣٢١٥أيُؤْذِيكَ هوامُّكَ؟٣٢١٦الحجُّ عرفاتٌ، الحجُّ عرفاتٌ، الحجُّ عرفاتٌ،…٣٢١٧أبغَضُ الرِّجالِ إلى اللهِ الألَدُّ الخَصِمُ٣٢١٨حدَّثنا عبدُ بنُ حمَيدٍ، حدَّثني سليمانُ بنُ…٣٢١٩حدَّثنا محمَّدُ بن عبد الأعلى، قال: حدَّثنا عبدُ…٣٢٢٠حدَّثنا ابنُ أبي عمرَ، قال: حدَّثنا سفيانُ، عن…٣٢٢١حدَّثنا محَمّدُ بن بَشَّارٍ، قال: حدَّثنا عبد…٣٢٢٢وما أهْلَكَكَ٣٢٢٣حدَّثنا عبدُ بن حُمَيدٍ، قال: حدَّثنا هاشِمُ بن…٣٢٢٤وصلاةِ العصرِ٣٢٢٥صلاةُ الوُسطى صلاةُ العصرِ٣٢٢٦اللَّهُمَّ امْلأ قبورَهُم وبيوتَهُم نارًا كما…٣٢٢٧صلاةُ الوُسْطى صلاةُ العصرِ٣٢٢٨حدَّثنا أحمدُ بنُ مَنِيعٍ، قال: حدَّثنا مروانُ بنُ…٣٢٢٩حدَّثنا أحمدُ بنُ مَنِيعٍ، قال: حدَّثنا هُشَيمٌ،…٣٢٣٠حدَّثنا عبدُ اللهِ بنُ عبدِ الرحمنِ، قال: أخبرنا…٣٢٣١حدَّثنا هنَّادٌ، قال: حدَّثنا أبو الأحوَصِ، عن…٣٢٣٢يا أيُّها النَّاسُ إنّ اللهَ طَيِّبٌ ولا يَقْبَلُ…٣٢٣٣حدَّثنا عبدُ بنُ حُمَيدٍ، قال: حدَّثنا عُبَيدُ…٣٢٣٤هذهِ مُعاتَبَةُ اللهِ العبدَ بما يُصِيبُه من…٣٢٣٥قولوا سَمِعْنا وأطَعْنا٣٢٣٦إذا رأَيتم الذين يتَّبعون ما تَشابَه منه فأُولئك…٣٢٣٧فإذا رَأَيتيهِم فاعرِفِيهِم٣٢٣٨حدَّثنا محمودُ بن غَيلانَ، قال: حدَّثنا أبو أحمدَ،…٣٢٣٩حدَّثنا محمودٌ قال: حدَّثنا أبو نُعَيمٍ، قال:…٣٢٤٠حدَّثنا أبو كُرَيبٍ، قال: حدَّثنا وكيعٌ، عن…٣٢٤١مَن حَلَفَ على يمينٍ وهو فيها فاجِرٌ لِيَقْتطِعَ…٣٢٤٢اجعَلْهُ في قرابَتِكَ أو أقْرَبيكَ٣٢٤٣الشَّعِثُ التَّفِلُ٣٢٤٤اللَّهُمَّ هؤُلاءِ أهْلِي٣٢٤٥حدَّثنا أبو كُرَيبٍ، قال: حدَّثنا وكيعٌ، عن ربيع…٣٢٤٦أنْتُم تُتِمُّونَ سَبْعينَ أمَّةً أنتمْ خَيْرها…٣٢٤٧كيف يُفْلِحُ قومٌ فَعَلوا هذا بِنَبيِّهم وهو…٣٢٤٨كيف تُفْلِحُ أُمَّةٌ، فَعَلوا هذا بِنَبيِّهم وهو…٣٢٤٩اللهُمَّ العَنْ أبا سُفيانَ، اللهُمَّ العَنِ…٣٢٥٠حدَّثنا يحيى بن حَبيبِ بن عَرَبيًّ البَصرِيُّ،…٣٢٥١حدَّثنا قُتيبةُ، قال حدَّثنا أبو عَوَانةَ، عن…٣٢٥٢حدَّثنا عبدُ بن حُميدٍ، قال: حدَّثنا رَوْحُ بن…٣٢٥٣حدَّثنا عبدٌ، قال: حدَّثنا رَوحُ بن عُبادةَ، عن…٣٢٥٤حدَّثنا يوسفُ بن حَمَّادٍ، قال: حدَّثنا عبدُ…٣٢٥٥حدَّثنا قُتَيبةُ، قال: حدَّثنا عبدُ الواحِدِ بنُ…٣٢٥٦يا جابِرُ مَا لِي أراكَ مُنكَسِرًا٣٢٥٧حدَّثنا ابنُ أبي عمرَ، قال: حدَّثنا سفيانُ، عن…٣٢٥٨حدَّثنا ابنُ أبي عمرَ، قال: حدَّثنا سفيانُ، عن…٣٢٥٩ما مِن رجلٍ لا يُؤدَّيَ زكاةَ مالِه إلَّا جَعَلَ…٣٢٦٠حدَّثنا عبدُ بنُ حُمَيدٍ، قال: حدَّثنا يزيدُ بنُ…٣٢٦١حدَّثنا الحسنُ بنُ محمدٍ الزَّعْفَرانِيُّ، قال:…٣٢٦٢حدَّثنا عبدُ بنُ حُمَيدٍ، قال: حدَّثنا يحيى بنُ…٣٢٦٣حدَّثنا الفَضْلُ بنُ صبَّاح البَغدادِيُّ، قال:…٣٢٦٤حدَّثنا عبدُ بنُ حُمَيدٍ، قال: حدثنا حَبَّانُ بنُ…٣٢٦٥حدَّثنا أحمدُ بنُ مَنِيعٍ، قال: أخبرنا هُشَيمٌ،…٣٢٦٦الشَّركُ باللهِ، وعقوقُ الوالِدينِ، وقَتْلُ…٣٢٦٧ألا أُحَدَّثُكُم بأكْبَرِ الكبائِرِ؟٣٢٦٨إنَّ من أكْبَرِ الكبائِرِ الشَّرْكَ باللهِ،…٣٢٦٩الكبائِرُ: الإشراكُ باللهِ، وعُقوقُ الوالِدَينِ،…٣٢٧٠حدَّثنا ابنُ أبي عمرَ، قال: حدَّثنا سفيانُ، عن ابن…٣٢٧١حدَّثنا ابنُ أبي عمرَ، قال: حدَّثنا سفيانُ، عن…٣٢٧٢حدَّثنا هَنَّادٌ، قال: حدَّثنا أبو الأحوَصِ، عن…٣٢٧٣اقرأْ عليَّ٣٢٧٤حدَّثنا سُوَيْدُ بن نصر، قال: أخبرنا ابنُ…٣٢٧٥حدَّثنا عبدُ بنُ حُمَيدٍ، قال: حدَّثنا عبدُ…٣٢٧٦اسْقِ يا زُبَيرُ وأرسِلِ الماءَ إلى جارِكَ٣٢٧٧إنَّها طَيْبةُ٣٢٧٨يَجِيءُ المَقْتولُ بالقاتِلِ يومَ القيامةِ…٣٢٧٩حدَّثنا عبدُ بنُ حُمَيدٍ، قال: حدَّثنا عبدُ…٣٢٨٠ائْتُونِي بالكتِفِ والدَّواةِ، أو اللّوحِ والدَّواةِ٣٢٨١حدَّثنا الحسنُ بنُ محمدٍ الزَّعفَرَانِيُّ، قال:…٣٢٨٢حدَّثنا عبدُ بنُ حُمَيدٍ، قال: حدثني يعقوبُ بنُ…٣٢٨٣صَدقةٌ تَصَدَّقَ اللهُ بِها عليكُم، فاقْبلوا…٣٢٨٤حدَّثنا محمودُ بنُ غَيلانَ، قال: حدَّثنا عبدُ…٣٢٨٥سآمُرُ في ذلكَ٣٢٨٦حدَّثنا خَلّادُ بنُ أسْلَمَ البغدادي، قال: حدَّثنا…٣٢٨٧قارِبُوا وسَدَّدُوا، وفي كُلَّ ما يُصيبُ المُؤمِنَ…٣٢٨٨يا أبا بكرٍ ألا أُقْرِئكَ آيةً أُنْزِلَتْ عليَّ٣٢٨٩حدَّثنا محمدُ بن المُثَنّى، قال: حدَّثنا أبو داودَ…٣٢٩٠حَدَّثَنا عبدُ بن حُمَيدٍ، قال: حَدَّثَنا أبو…٣٢٩١تُجْزِئُك آيةُ الصَّيفِ٣٢٩٢حدَّثنا ابنُ أبي عمرَ، قال: حدَّثنا سفيانُ، عن…٣٢٩٣حدَّثنا عبدُ بن حُمَيدٍ، قال: حدثنا يزيدُ بن…٣٢٩٤يَمينُ الرَّحمنِ مَلأْى سَحَّاءُ لا يَغِيضُها…٣٢٩٥يا أيُّها النَّاسُ انْصَرِفوا، فقد عَصَمَني اللهُ٣٢٩٦حدَّثنا نَصْرُ بن عليًّ، قال: حدَّثنا مسلمُ بن…٣٢٩٧لمَّا وَقَعَتْ بنو إسرائيلَ في المَعاصي،…٣٢٩٨إنَّ بني إسرائيلَ لمَّا وَقَعَ فيهِمُ النَّقصُ كان…٣٢٩٩حدَّثنا محمد بن بشَّار، قال: حدَّثنا أبو داودَ…٣٣٠٠حدَّثنا أبو حفص عمرو بن عليًّ، قال: حدثنا أبو…٣٣٠١حدَّثنا عبدُ الله بن عبدِ الرَّحمنِ، قال: أخبرنا…٣٣٠٢حدَّثنا محمدُ بنُ العلاءِ، قال: حدَّثنا وكيعٌ، عن…٣٣٠٣حدَّثنا عبدُ بنُ حُمَيدٍ، قال: أخبرنا عُبَيدُ…٣٣٠٤حدَّثنا بذلكَ محمدُ بنُ بشارٍ، قال: حدَّثنا محمدُ…٣٣٠٥حدَّثنا عبدُ بنُ حُمَيدٍ، قال: أخبرنا عبدُ…٣٣٠٦أنتَ مِنهُم٣٣٠٧لا، ولو قلتُ نَعَمْ لوَجَبَتْ٣٣٠٨أبوكَ فُلانٌ٣٣٠٩إنَّ النّاسَ إذا رأوا ظالمًا، فلم يأخُذوا على…٣٣١٠بل ائْتَمِرُوا بالمَعْروفِ، وتَناهَوا عن المنكَرِ،…٣٣١١حدَّثنا الحسنُ بنُ أحمدَ بنِ أبي شُعَيْبٍ…٣٣١٢حدَّثنا سفيانُ بنُ وكيعٍ، قال: حدَّثنا يحيى بنُ…٣٣١٣أُنْزِلَتِ المائدَةُ من السَّماءِ خُبزًا ولحمًا،…٣٣١٤حدَّثنا حُمَيدُ بنُ مَسْعَدَةَ، قال: حدَّثنا…٣٣١٥حدَّثنا ابنُ أبي عمرَ قال: حدَّثنا سفيانُ بنُ…٣٣١٦حدَّثنا قُتَيبةُ، قال: حدَّثنا عبدُ اللهِ بن وهبٍ،…٣٣١٧حَدَّثنا أبو كُريبٍ، قال: حدّثنا معاويةُ بن هشام،…٣٣١٨حدَّثنا إسحاقُ بنُ منصُورٍ، قال: حدثنا عبدُ…٣٣١٩هاتانِ أهْوَنُ، أو هاتانِ أيْسَرُ٣٣٢٠أما إنَّها كائِنةٌ ولم يأتِ تأويلُها بعدُ٣٣٢١حدَّثنا عليُّ بن خَشْرَمٍ، قال: أخبرنا عِيسى بن…٣٣٢٢إنَّما ذاك جبريلُ، ما رأيتُه فىِ الصُّورَةِ التي…٣٣٢٣حدَّثنا محمَّد بن موسى البَصريُّ الحَرَشيُّ، قال:…٣٣٢٤حَدَّثَنا الفَضلُ بن الصَّبَّاحِ البَغْداديُّ،…٣٣٢٥طُلُوعُ الشَّمسِ مِن مَغرِبِها٣٣٢٦ثلاثٌ إذا خَرَجْنَ ﴿لَا يَنْفَعُ نَفْسًا…٣٣٢٧حدَّثنا ابنُ أبي عمرَ، قال: حدَّثنا سفيانُ، عن أبي…٣٣٢٨حدَّثنا عَبدُ الله بن عبدِ الرَّحمنِ، قال: أخبرنا…٣٣٢٩حدَّثنا عبد الوَهَّابِ الوَرَّاقُ البغدادي، قال:…٣٣٣٠إنَّ الله خَلَقَ آدَمَ، ثمَّ مَسَحَ ظَهرَه…٣٣٣١لَمَّا خَلَقَ اللهُ آدمَ مَسَحَ ظَهرَه، فسَقَطَ من…٣٣٣٢لمَّا حَمَلَتْ حَوَّاءُ طافَ بها إبليسُ وكانَ لا…٣٣٣٣هذا ليسَ لي ولا لكَ٣٣٣٤اللَّهُمَّ أنْجِزْ لِي ما وَعَدْتَني، اللَّهم…٣٣٣٥صَدَقْتَ٣٣٣٦أنْزَلَ اللهُ عليَّ أمانَينِ لأُمَّتي ٣٦٥٦حدَّثنا عبدُ بنُ حُمَيدٍ، قال: حَدَّثَنا سُليمانُ…٣٦٥٧حدَّثنا محمدُ بنُ بشَّارٍ، قال: حدَّثنا محمدُ بنُ…

ومن سورة تبت

٣٦٥٨يا صَباحَاهُ٣٦٥٩حدَّثنا أحمدُ بنُ مَنِيعٍ، قال: حدَّثنا أَبو…٣٦٦٠حدَّثنا عَبد بنُ حُمَيْدٍ، قال: أخبرنا عُبيدُ…

ومن سورة المعوذتين

٣٦٦١يا عائشةُ استَعِيذي باللهِ من شَرِّ هذا، فإنَّ هذا…٣٦٦٢حَدَّثَنا محمدُ بنُ بشَّارٍ، قال: حدَّثنا يحيى بن…
