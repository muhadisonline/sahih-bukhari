٤٦ حديثًا

١٤٦٥الدعاء هو العبادة١٤٦٦وعن عائشة، ، قالت: كان رسول الله ، يستحب الجوامع…١٤٦٧اللهم آتنا في الدنيا حسنة، وفي الآخرة حسنة، وقنا…١٤٦٨اللهم إني أسألك الهدى، والتقى، والعفاف، والغنى١٤٦٩اللهم اغفر لي، وارحمني، واهدني، وعافني، وارزقني١٤٧٠اللهم مصرف القلوب صرف قلوبنا على طاعتك١٤٧١تعوذوا بالله من جهد البلاء، ودرك الشقاء، وسوء…١٤٧٢اللهم أصلح لي ديني الذي هو عصمة أمري، وأصلح لي…١٤٧٣قل: اللهم اهدني، وسددني١٤٧٤اللهم إني أعوذ بك من العجز والكسل والجبن والهرم،…١٤٧٥قل: اللهم إني ظلمت نفسي ظلمًا كثيرًا، ولا يغفر…١٤٧٦اللهم اغفر لي خطيئتي وجهلي، وإسرافي في أمري، وما…١٤٧٧اللهم إني أعوذ بك من شر ما عملت ومن شر ما لم أعمل١٤٧٨اللهم إني أعوذ بك من زوال نعمتك، وتحول عافيتك،…١٤٧٩اللهم إني أعوذ بك من العجز والكسل، والبخل والهرم،…١٤٨٠اللهم لك أسلمت، وبك آمنت، وعليك توكلت، وإليك أنبت…١٤٨١اللهم إني أعوذ بك من فتنة النار، وعذاب النار، ومن…١٤٨٢اللهم إني أعوذ بك من منكرات الأخلاق، والأعمال،…١٤٨٣قل: اللهم إني أعوذ بك من شر سمعي، ومن شر بصري، ومن…١٤٨٤اللهم إني أعوذ بك من البرص والجنون، والجذام، وسيئ…١٤٨٥اللهم إني أعوذ بك من الجوع، فإنه بئس الضجيع، وأعوذ…١٤٨٦اللهم اكفني بحلالك عن حرامك، وأغنني بفضلك عمن سواك١٤٨٧اللهم ألهمني رشدي، وأعذني من شر نفسي١٤٨٨سلوا الله العافية١٤٨٩يا مقلب القلوب ثبت قلبي على دينك١٤٩٠كان من دعاء داود :١٤٩١ألظوا بيا ذا الجلال والإكرام١٤٩٢يا رسول الله دعوت بدعاء كثير لم نحفظ منه شيئا،١٤٩٣اللهم إني أسألك موجبات رحمتك، وعزائم مغفرتك،…١٤٩٤ما من عبد مسلم يدعو لأخيه بظهر الغيب إلا قال الملك…١٤٩٥دعوة المرء المسلم لأخيه بظهر الغيب مستجابة، عند…١٤٩٦من صنع إليه معروف، فقال لفاعله: جزاك الله خيرًا،…١٤٩٧لا تدعوا على أنفسكم، ولا تدعوا على أولادكم، ولا…١٤٩٨أقرب ما يكون العبد من ربه وهو ساجد، فأكثروا الدعاء١٤٩٩يستجاب لأحدكم ما لم يعجل: يقول: قد دعوت ربي، فلم…١٥٠٠جوف الليل الآخر ودبر الصلوات المكتوبات١٥٠١ما على الأرض مسلم يدعو الله تعالى بدعوة إلا آتاه…١٥٠٢لا إله إلا الله العظيم الحليم، لا إله إلا الله رب…١٥٠٣من كان عنده طعام اثنين، فليذهب بثالث، ومن كان عنده…١٥٠٤لقد كان فيما قبلكم من الأمم ناس محدثون، فإن يك في…١٥٠٥وعن جابر بن سَمُرة، ، قال: شكا أهل الكوفة سعدًا،…١٥٠٦من أخذ شبرًا من الأرض ظلمًا، طوقه إلى سبعين أرضين١٥٠٧وعن جابر بن عبد الله الله عنهما قال: لما حضرت أحد…١٥٠٨وعن أنس أن رجلين من أصحاب النبي خرجا من عند النبي…١٥٠٩اقتلهم بَِددًا١٥١٠وعن ابن عمر قال: ما سمعت عمر يقول لشيء قط: إني…
