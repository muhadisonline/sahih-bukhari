كتاب العلم

١٧ حديثًا

١٣٧٦من يرد الله به خيرًا يفقه في الدين١٣٧٧لا حسد إلا في اثنتين: رجل آتاه الله مالا فسلطه على…١٣٧٨مثل ما بعثني الله به من الهدى والعلم كمثل غيث أصاب…١٣٧٩فوالله لأن يهدي الله بك رجلا واحدا خير من حمر النعم١٣٨٠بلغوا عني ولو آية وحدثوا عن بني إسرائيل ولا حرج،…١٣٨١ومن سلك طريقًا يلتمس فيه علما سهل الله له به…١٣٨٢وعنه أيضًا أن رسول الله قال: "من دعا إلى هدى كان…١٣٨٣إذا مات ابن آدم انقطع عمله إلا من ثلاث: صدقة…١٣٨٤الدنيا ملعونة ملعون ما فيها إلا ذكر الله تعالى،…١٣٨٥من خرج في طلب العلم، فهو في سبيل الله حتى يرجع١٣٨٦لن يشبع مؤمن من خير حتى يكون منتهاه الجنة١٣٨٧فضل العالم على العابد كفضلي على أدناكم١٣٨٨من سلك طريقًا يبتغي فيه علمًا سهل الله له طريقًا…١٣٨٩نضر الله امرءًا سمع منا شيئًا فبلغه كما سمعه فرب…١٣٩٠من سئل عن علم فكتمه ألجم يوم القيامة بلجام من نار١٣٩١من تعلم علمًا مما يبتغى به وجه الله عز وجل لا…١٣٩٢وعن عبد الله بن عمرو بن العاص قال: سمعت رسول الله…
