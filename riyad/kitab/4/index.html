كتاب اللباس

٣٥ حديثًا

٧٧٨وعن ابن عباس رضى الله عنهما أن رسول الله قال:…٧٧٩إلبسوا البياض؛ فإنها أطهر وأطيب، وكفنوا فيها موتاكم٧٨٠وعن البراء رضى الله عنه قال: كان رسول الله…٧٨١وعن أبى جحيفة وهب بن عبد الله رضى الله عنه قال:…٧٨٢وعن أبى رمثة رفاعة التيمى رضى الله عنه قال: رأيت…٧٨٣وعن جابر رضى الله عنه، أن رسول الله دخل يوم فتح…٧٨٤وعن أبى سعيد عمرو بن حريث رضى الله عنه قال: كأني…٧٨٥وعن عائشة رضى الله عنها قالت: كفن رسول الله في…٧٨٦وعنها قالت: خرج رسول الله ذات غداة، وعليه مرط مرحل…٧٨٧أمعك ماء٧٨٨عن أم سلمة رضى الله عنها قالت: كان أحب الثياب إلى…٧٨٩عن أسماء بنت يزيد الأنصارية رضى الله عنها قالت:…٧٩٠من جر ثوبه خيلاء لم ينظر الله إليه يوم القيامة٧٩١لا ينظر الله يوم القيامة إلى من جر إزاره بطراً٧٩٢ما أسفل من الكعبين من الإزار ففي النار٧٩٣ثلاثة لا يكلمهم الله يوم القيامة، ولا ينظر إليهم،…٧٩٤الإسبال من الإزار، والقميص، والعمامة؛ من جر شيئاً…٧٩٥لا تقل عليك السلام، عليك السلام تحية الموتى -قلت:…٧٩٦اذهب فتوضأ٧٩٧سبحان الله؟لا بأس أن يؤجر ويحمد٧٩٨إزارة المسلم إلى نصف الساق، ولا حرج -أو لا جناح-…٧٩٩يا عبد الله، ارفع إزارك٨٠٠فقالت أم سلمة: فكيف تصنع النساء بذيولهن، قال:٨٠١من ترك اللباس تواضعاً لله، وهو يقدر عليه، دعاه…٨٠٢إن الله يحب أن يرى أثر نعمته على عبده٨٠٣لا تلبسوا الحرير؛ فإن من لبسه في الدنيا لم يلبسه…٨٠٤إنما يلبس الحرير من لا خلاق له٨٠٥من لبس الحرير في الدنيا لم يلبسه في الآخرة٨٠٦إن هذين حرام على ذكور أمتي٨٠٧حرم لباس الحرير والذهب على ذكور أمتي،وأحل لإناثهم٨٠٨وعن حذيفة رضى الله عنه قال: نهانا النبي أن نشرب في…٨٠٩عن أنس رضى الله عنه قال: رخص رسول الله، ، للزبير…٨١٠لا تركبوا الخز ولا النمار٨١١وعن أبى المليح عن أبيه، رضى الله عنه، أن رسول الله…٨١٢اللهم لك الحمد أنت كسوتنيه، أسألك خيره وخير ما صنع…
البخاري مسلم أبو داود الترمذي النسائي ابن ماجه الموطأ الدارمي الأدب المفرد رياض الصالحين
