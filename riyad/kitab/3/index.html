كتاب أدب الطعام

٥١ حديثًا

٧٢٧سم الله وكل بيمينك، وكل مما يليك٧٢٨إذا أكل أحدكم فليذكر اسم الله تعالى، فإذا نسى أن…٧٢٩إذا دخل الرجل بيته، فذكر الله تعالى عند دخوله وعند…٧٣٠إن الشيطان يستحل الطعام أن لا يذكر اسم الله تعالى…٧٣١ما زال الشيطان يأكل معه، فلما ذكر اسم الله استقاء…٧٣٢أما إنه لو سمى لكفاكم٧٣٣الحمد لله حمداً كثيراً طَيِّباً مباركاً فيه، غير…٧٣٤من أكل طعاماً فقال: الحمد الله الذي أطعمني هذا،…٧٣٥ما عاب رسول الله طعاماً قط، إن اشتهاه أكله، وإن…٧٣٦نعم الأدم الخل، نعم الأدم الخل٧٣٧إذا دعي أحدكم، فليجب، فإن كان صائما فليصل، وإن كان…٧٣٨إن هذا تبعنا؛ فإن شئت أن تأذن له، وإن شئت رجع٧٣٩يا غلام سم الله تعالى، وكل بيمينك،وكل مما يليك٧٤٠كُلْ بِيَمِينِكَ٧٤١إلا أن يستأذن الرجل أخاه٧٤٢فلعلكم تفترقون٧٤٣البركة تنزل وسط الطعام، فكلوا من حافتيه ولا تأكلوا…٧٤٤كلوا من حواليها، ودعوا ذروتها يبارك فيها٧٤٥لا آكل متكئاً٧٤٦وعن أنس رضى الله عنه قال: رأيت رسول الله جالساً…٧٤٧إذا أكل أحدكم طعاماً، فلا يمسح أصابعه حتى يلعقها…٧٤٨وعن كعب بن مالك رضى الله عنه قال: رأيت رسول الله…٧٤٩إنكم لا تدرون في أي طعامكم البركة٧٥٠إذا وقعت لقمة أحدكم، فليأخذها فليمط ما كان بها من…٧٥١إن الشيطان يحضر أحدكم عند كل شئ من شأنه، حتى يحضره…٧٥٢إذا سقطت لقمة أحدكم فليأخذها، وليمط عنها الأذى،…٧٥٣وعن سعيد بن الحارث أنه سأل جابراً رضى الله عنه عن…٧٥٤عن أبى هريرة رضى الله عنه قال: قال رسول الله :طعام…٧٥٥طعام الواحد يكفى الاثنين، وطعام الاثنين يكفى…٧٥٦عن أنس رضى الله عنه أن رسول الله كان يتنفس في…٧٥٧لا تشربوا واحداً كشرب البعير، ولكن اشربوا مثنى…٧٥٨وعن أبى قتادة رضى الله عنه أن النبي نهى أن يتنفس…٧٥٩الأيمن فالأيمن٧٦٠أتأذن لي أن أعطى هؤلاء.؟٧٦١عن أبى سعيد الخدرى رضى الله عنه قال: نهى رسول الله…٧٦٢وعن أبى هريرة رضى الله عنه قال: نهى رسول الله أن…٧٦٣وعن أم ثابت كبشة بنت ثابت أخت حسان بن ثابت رضى…٧٦٤أهرقها٧٦٥وعن ابن عباس رضى الله عنهما أن النبي نهى أن يتنفس…٧٦٦وعن ابن عباس رضى الله عنهما قال: سقيت النبي من…٧٦٧وعن النزال بن سبرة رضى الله عنه قال: أتى علي رضى…٧٦٨وعن ابن عمر رضى الله عنهما قال:كنا نأكل على عهد…٧٦٩وعن عمرو بن شعيب عن أبيه عن جده رضى الله عنه قال:…٧٧٠وعن أنس رضى الله عنه عن النبي أنه نهى أن يشرب…٧٧١وعن أبى هريرة رضى الله عنه قال: قال رسول الله : لا…٧٧٢ساقي القوم آخرهم شرباً٧٧٣عن أنس رضى الله عنه قال: حضرت الصلاة، فقام من كان…٧٧٤وعن عبد الله بن زيد رضى الله عنه قال: أتانا النبي…٧٧٥إن كان عندك ماء بات هذه الليلة في سنة وإلا كرعنا٧٧٦هي لهم في الدنيا، وهى لكم في الآخرة٧٧٧الذي يشرب في آنية الفضة إنما يجرجر في بطنه نار جهنم
