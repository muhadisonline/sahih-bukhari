٢٧٧ حديثًا

٩٩١اقرءوا القرآن فإنه يأتي يوم القيامة شفيعًا لأصحابه٩٩٢يؤتى يوم القيامة بالقرآن وأهله الذين كانو يعملون…٩٩٣خيركم من تعلم القرآن وعلمه٩٩٤الذي يقرأ القرآن وهو ماهر به مع السفرة الكرام…٩٩٥مثل المؤمن الذي يقرأ القرآن مثل الأترجة: ريحها…٩٩٦إن الله يرفع بهذا الكتاب أقوامًا ويضع به آخرين٩٩٧لا حسد إلا في اثنتين: رجل آتاه الله القرآن، فهو…٩٩٨تلك السكينة تنزلت للقرآن٩٩٩من قرأ حرفًا من كتاب الله فله حسنة، والحسنة بعشر…١٠٠٠إن الذي ليس في جوفه شيء من القرآن كالبيت الخرب١٠٠١يقال لصاحب القرآن: اقرأ وارتقِ ورتل كما كنت ترتل…١٠٠٢تعاهدوا هذا القرآن فوالذي نفس محمد بيده لهو أشد…١٠٠٣إنما مثل صاحب القرآن كمثل الإبل المعقلة، إن عاهد…١٠٠٤ما أذن الله لشيء ما أذن لنبي حسن الصوت يتغنى…١٠٠٥لقد أتيت مزمارا من مزامير آل داود١٠٠٦وعن البراء بن عازب قال:سمعت النبي قرأ في العشاء…١٠٠٧وعن أبي لبابة بشير بن عبد المنذر قال، أن النبي…١٠٠٨اقرأ علي القرآن١٠٠٩ألا أعلمك أعظم سورة في القرآن قبل أن تخرج من…١٠١٠والذي نفسي بيده، إنها لتعدل ثلث القرآن١٠١١والذي نفسي بيده، إنها لتعدل ثلث القرآن١٠١٢إنها تعدل ثلث القرآن١٠١٣قل هو الله أحد١٠١٤ألم ترَ آيات أنزلت هذه الليلة لم يرَ مثلهن قط؟ قل…١٠١٥وعن أبي سعيد الخدري قال: كان رسول الله يتعوذ من…١٠١٦من القرآن سورة ثلاثون آية شفعت لرجل حتى غفرت…١٠١٧من قرأ بالآيتين من آخر سورة البقرة في ليلة كفتاه١٠١٨لا تجعلوا بيوتكم مقابر، إن الشيطان ينفر من البيت…١٠١٩يا أبا المنذر أتدري أي آية من كتاب الله معك أعظم؟…١٠٢٠يا أبا هريرة، ما فعل أسيرك البارحة؟١٠٢١من حفظ عشر آيات من أول سورة الكهف، عصم من الدجال١٠٢٢هذا باب من السماء فتح اليوم ولم يفتح قط إلا اليوم،…١٠٢٣ما اجتمع قوم في بيت من بيوت الله يتلون كتاب الله،…١٠٢٤إن أمتي يدعون يوم القيامة غرًا محجلين من آثار…١٠٢٥تبلغ الحلية من المؤمن حيث يبلغ الوضوء١٠٢٦من توضأ فأحسن الوضوء، خرجت خطاياه من جسده حتى تخرج…١٠٢٧من توضأ هكذا، غفر له ما تقدم من ذنبه، وكانت صلاته…١٠٢٨إذا توضأ العبد المسلم - أو المؤمن- فغسل وجهه، خرج…١٠٢٩السلام عليكم دار قوم مؤمنين وإنا إن شاء الله بكم…١٠٣٠إسباغ الوضوء على المكاره، وكثرة الخطا إلى المساجد،…١٠٣١الطهور شطر الإيمان١٠٣٢ما منكم من أحد يتوضا فيبلغ-أو فيسبغ الوضوء- ثم…١٠٣٣لو يعلم الناس ما في النداء والصف الأول ثم لم يجدوا…١٠٣٤المؤذنون أطول الناس أعناقًا يوم القيامة١٠٣٥إني أراك تحب الغنم والبادية فإذا كنت في غنمك- أو…١٠٣٦إذا نودي بالصلاة، أدبر الشيطان، له ضراط حتى لا…١٠٣٧إذا سمعتم النداء فقولوا مثل ما يقول، ثم صلوا علي،…١٠٣٨إذا سمعتم النداء، فقولوا كما يقول المؤذن١٠٣٩من قال حين يسمع النداء: اللهم رب هذه الدعوة…١٠٤٠من قال حين يسمع المؤذن: أشهد أن لا إله إلا الله…١٠٤١الدُّعَاءُ لا يُرَدُّ بَيْنَ الأَذَانِ وَالإقَامَةِ١٠٤٢أرأيتم لو أن نهرًا بباب أحدكم يغتسل منه كل يوم خمس…١٠٤٣مثل الصلوات الخمس كمثل نهر غمر جار على باب أحدكم…١٠٤٤لجميع أمتي كلهم١٠٤٥الصلوات الخمس، والجمعة إلى الجمعة، كفارة لما…١٠٤٦ما من امرئ مسلم تحضره صلاة مكتوبة فيحسن وضوءها…١٠٤٧من صلى البردين دخل الجنة١٠٤٨لن يلج النار أحد صلى قبل طلوع الشمس وقبل غروبها١٠٤٩من صلى الصبح فهو في ذمة الله فانظر يابن آدم، لا…١٠٥٠يتعاقبون فيكم ملائكة بالليل، وملائكة بالنهار،…١٠٥١إنكم سترون ربكم كما ترون هذا القمر، لاتضامون في…١٠٥٢من ترك صلاة العصر فقد حبط عمله١٠٥٣من غدا إلى المسجد أو راح، أعد الله له في الجنة…١٠٥٤من تطهر في بيته، ثم مضى إلى بيت من بيوت الله،…١٠٥٥قد جمع الله لك ذلك كله١٠٥٦بلغني أنكم تريدون أن تنتقلوا قرب المسجد؟ قالوا:…١٠٥٧إن أعظم الناس أجرًا في الصلاة أبعدهم إليها ممشى،…١٠٥٨بشروا المشائين في الظلم إلى المساجد بالنور التام…١٠٥٩ألا أدلكم على ما يمحو الله به الخطايا، ويرفع به…١٠٦٠وعن أبي سعيد الخدري عن النبي قال: "إذا رأيتم الرجل…١٠٦١لا يزال أحدكم في صلاة مادامت الصلاة تحبسه لا يمنعه…١٠٦٢وعنه أن رسول الله قال: الملائكة تصلي على أحدكم ما…١٠٦٣صلى الناس ورقدوا ولم تزالوا في صلاة منذ انتظرتموها١٠٦٤وعن ابن عمر أن رسول الله قال: "صلاة الجماعة أفضل…١٠٦٥صلاة الرجل في جماعة تُضعَّف على صلاته في بيته وفي…١٠٦٦هل تسمع النداء بالصلاة؟١٠٦٧تسمع حي على الصلاة، حي على الفلاح، فحيهلا١٠٦٨والذي نفسي بيده، لقد هممت أن آمر بحطب فيحتطب، ثم…١٠٦٩وعن ابن مسعود قال: من سره أن يلقى الله تعالى غدًا…١٠٧٠ما من ثلاثة في قرية ولا بدو لا تقام فيهم الصلاة…١٠٧١من صلى العشاء في جماعة، فكأنما قام نصف الليل ومن…١٠٧٢ولو يعلمون ما في العتمة والصبح لأتوهما ولو حبوًا١٠٧٣ليس صلاة أثقل على المنافقين من صلاة الفجر والعشاء…١٠٧٤الصلاة على أوقتها١٠٧٥بني الإسلام على خمس: شهادة أن لا إله إلا الله وأن…١٠٧٦أمرت أن أقاتل الناس حتى يشهدوا أن لا إله إلا الله…١٠٧٧إنك تأتي قومًا من أهل الكتاب، فادعهم إلى شهادة أن…١٠٧٨إن بين الرجل وبين الشرك والكفر ترك الصلاة١٠٧٩العهد الذي بيننا وبينهم الصلاة، فمن تركها فقد كفر١٠٨٠وعن شقيق بن عبد الله التابعي المتفق على جلالته…١٠٨١إن أول ما يحاسب به العبد يوم القيامة من عمله…١٠٨٢ألا تصفون كما تصف الملائكة عند ربها؟١٠٨٣لو يعلم الناس ما في النداء والصف الأول، ثم لم…١٠٨٤خير صفوف الرجال أولها، وشرها آخرها، وخير صفوف…١٠٨٥تقدموا فَأتموا بي وليأتم بكم مَن بعدكم، لا يزال…١٠٨٦استووا ولا تختلفوا فتختلف قلوبكم، ليلني منكم أولو…١٠٨٧سووا صفوفكم، فإن تسوية الصف من تمام الصلاة١٠٨٨أقيموا صفوفكم وتراصوا، فإني أراكم من وراء ظهري١٠٨٩لتسون صفوفكم، أو ليخالفن الله بين وجوهكم١٠٩٠لا تختلفوا فتختلف قلوبكم١٠٩١أقيموا الصفوف، وحاذوا المناكب، وسدوا الخلل، ولينوا…١٠٩٢رصوا صفوفكم، وقاربوا بينها وحاذوا الأعناق فوالذي…١٠٩٣أتموا الصف المقدم، ثم الذي يليه، فما كان من نقص…١٠٩٤إن الله وملائكته يصلون على ميامن الصفوف١٠٩٥كنا إذا صلينا خلف رسول الله ، أحببنا أن نكون عن…١٠٩٦وسطوا الإمام، وسدوا الخلل١٠٩٧عن أم المؤمنين أم حبيبة رملة بنت أبي سفيان، ، قالت…١٠٩٨وعن ابن عمر ، قال: صليت مع رسول الله ، ركعتين قبل…١٠٩٩بين كل أذانين صلاة، بين كل أذانين صلاة، بين كل…١١٠٠عن عائشة ، أن النبي كان لا يدع أربعًا قبل الظهر،…١١٠١وعنها قالت: لم يكن النبي، على شيء من النوافل أشد…١١٠٢ركعتا الفجر خير من الدنيا وما فيها١١٠٣إني كنت ركعت ركعتي الفجر١١٠٤عن عائشة أن رسول الله ، كان يصلي ركعتين خفيفتين…١١٠٥وعن حفصة أن رسول الله كان إذا أذن المؤذن للصبح،…١١٠٦وعن ابن عمر قال: كان رسول الله يصلي من الليل مثنى…١١٠٧وعن ابن عباس أن رسول الله ، كان يقرأ في ركعتي…١١٠٨وعن أبي هريرة أن رسول الله ، قرأ في ركعتي الفجر:…١١٠٩وعن ابن عمر ، قال رمقت النبي شهرًا يقرأ في…١١١٠عن عائشة قالت: كان النبي إذا صلى ركعتي الفجر،…١١١١وعنها قالت: كان النبي يصلي فيما بين أن يفرغ من…١١١٢إذا صلى أحدكم ركعتي الفجر، فليضطجع على يمينه١١١٣عن ابن عمر قال: صليت مع رسول الله ركعتين قبل…١١١٤وعن عائشة أن النبي كان لا يدع أربعًا قبل الظهر.…١١١٥وعنها قالت: كان النبي يصلي في بيتي قبل الظهر…١١١٦من حافظ على أربع ركعات قبل الظهر وأربع بعدها، حرمه…١١١٧إنها ساعة تفتح فيها أبواب السماء، فأحب أن يصعد لي…١١١٨وعن عائشة أن النبي كان إذا لم يصلِ أربعًا قبل…١١١٩عن علي بن أبي طالب، قال كان النبي يصلي قلب العصر…١١٢٠رحم الله امرءًا صلى قبل العصر أربعًا١١٢١وعن علي بن أبي طالب أن النبي كان يصلي قبل العصر…١١٢٢صلوا قبل المغرب١١٢٣- وعن أنس، قال: لقد رأيت كبار أصحاب رسول الله…١١٢٤وعنه قال: كنا نصلي على عهد رسول الله ركعتين بعد…١١٢٥وعنه قال: كنا بالمدينة فإذا أذن المؤذن لصلاة…١١٢٦إذا صلى أحدكم الجمعة، فليصلِ بعدها أربعًا١١٢٧وعن ابن عمر ، أن النبي كان لا يصلي بعد الجمعة حتى…١١٢٨صلوا أيها الناس في بيوتكم، فإن الصلاة صلاة المرء…١١٢٩وعن ابن عمر عن النبي قال: اجلعوا من صلاتكم في…١١٣٠إذا قضى أحدكم صلاته في مسجده، فليجعل لبيته نصيبًا…١١٣١وعن عمر بن عطاء أن نافع بن جبير أرسله إلى السائب…١١٣٢إن الله وتر يحب الوتر، فأوتروا يا أهل القرآن١١٣٣وعن عائشة قالت: من كل الليل قد أوتر رسول الله : من…١١٣٤اجعلوا آخر صلاتكم بالليل وترًا١١٣٥أوتروا قبل أن تصبحوا١١٣٦قومي فأوتري يا عائشة١١٣٧بادروا الصبح بالوتر١١٣٨من خاف أن لا يقوم الليل فليوتر أوله، ومن طمع أن…١١٣٩عن أبي هريرة قال: أوصاني خليلي بصيام ثلاثة أيام من…١١٤٠يصبح على كل سلامى من أحدكم صدقة: فكل تسبيحة صدقة،…١١٤١وعن عائشة قالت: كان رسول الله يصلي الضحى أربعًا…١١٤٢وعن أم هانئ فاختة بنت أبي طالب قالت: ذهبت إلى رسول…١١٤٣صلاة الأوابين حين ترمض الفصال١١٤٤إذا دخل أحدكم المسجد فلا يجلس حتى يصلي ركعتين١١٤٥صلِ ركعتين١١٤٦وعن أبي هريرة أن رسول الله قال لبلال: يا بلال…١١٤٧خير يوم طلعت عليه الشمس يوم الجمعة: فيه خلق آدم،…١١٤٨من توضأ فأحسن الوضوء ثم أتي الجمعة، فاستمع وأنصت،…١١٤٩الصلوات الخمس والجمعة إلى الجمعة، ورمضان إلى…١١٥٠لينتهين أقوام عن وَدعهم الجمعات، أو ليختمن الله…١١٥١إذا جاء أحدكم الجمعة، فليغتسل١١٥٢غسل الجمعة واجب على كل محتلم١١٥٣من توضأ يوم الجمعة، فبها ونعمت، ومن اغتسل فالغسل…١١٥٤لا يغتسل رجل يوم الجمعة، ويتطهر ما استطاع من طهر،…١١٥٥من اغتسل يوم الجمعة غسل الجنابة، ثم راح في الساعة…١١٥٦فيها ساعة لا يوافقها عبد مسلم، وهو قائم يصلي يسأل…١١٥٧هي ما بين أن يجلس الإمام إلى أن تقضى الصلاة١١٥٨إن من أفضل أيامكم يوم الجمعة، فأكثروا علي من…١١٥٩إني سألت ربي، وشفعت لأمتي، فأعطاني ثلث أمتي، فخررت…١١٦٠أفلا أكون عبدًا شكورًا١١٦١ألا تصليان؟١١٦٢نعم الرجل عبد الله لو كان يصلي من الليل١١٦٣يا عبد الله لا تكن مثل فلان: كان يقوم الليل فترك…١١٦٤ذاك رجل بال الشيطان في أذنيه- أو قال: في أذنه-١١٦٥يعقد الشيطان على قافية، رأس أحدكم، إذا هو نام،…١١٦٦أيها الناس أفشوا السلام، وأطعموا الطعام، وصلوا…١١٦٧أفضل الصيام بعد رمضان شهر الله المحرم، وأفضل…١١٦٨صلاة الليل مثنى مثنى، فإذا خفت الصبح فأوتر بواحدة١١٦٩وعنه قال: كان النبي ، يصلي من الليل مثنى مثنى،…١١٧٠وعن أنس ، قال: كان رسول الله ، يفطر من الشهر حتى…١١٧١وعن عائشة ، أن رسول الله ، كان يصلي إحدى عشرة…١١٧٢يا عائشة إن عيني تنامان ولا ينام قلبي١١٧٣وعنها أن النبي ، كان ينام أول الليل، ويقوم آخره…١١٧٤وعن ابن مسعود ، قال: صليت مع النبي ، ليلة، فلم يزل…١١٧٥وعن حذيفة ، قال: صليت مع النبي ، ذات ليلة فافتتح…١١٧٦طول القنوت١١٧٧أحب الصلاة إلى الله صلاة داود، وأحب الصيام إلى…١١٧٨إن في الليل لساعة، لا يوافقها رجل مسلم يسأل الله…١١٧٩إذا قام أحدكم من الليل فليفتتح الصلاة بركعتين…١١٨٠وعن عائشة قالت كان رسول الله إذا قام من الليل…١١٨١وعنها ، قالت: كان رسول الله ، إذا فاتته الصلاة من…١١٨٢من نام عن حزبه، أو عن شيء منه،
