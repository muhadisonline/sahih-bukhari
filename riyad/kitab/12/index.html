٩١ حديثًا

١٢٨٥إيمان بالله ورسوله١٢٨٦الصلاة على وقتها١٢٨٧الإيمان بالله والجهاد في سبيله١٢٨٨لغدوة في سبيل الله أو روحة خير من الدنيا وما فيها١٢٨٩مؤمن يجاهد بنفسه وماله في سبيل الله١٢٩٠وعن سهل بن سعد أن رسول الله قال: "رباط يوم في سبيل…١٢٩١رباط يوم وليلة خير من صيام شهر وقيامه، وإن مات فيه…١٢٩٢وعن فضالة بن عبيد أن رسول الله قال: "كل ميت يختم…١٢٩٣رباط يوم في سبيل الله خير من ألف يوم فيما سواه من…١٢٩٤تضمن الله لمن خرج في سبيله لا يخرجه إلا جهاد في…١٢٩٥وعنه قال: قال: رسول الله "ما من مكلوم يكلم في سبيل…١٢٩٦من قاتل في سبييل الله من رجل مسلم فواق ناقة وجبت…١٢٩٧لا تفعل فإن مقام أحدكم في سبيل الله أفضل من صلاته…١٢٩٨لاتستطيعونه١٢٩٩من خير معاش الناس لهم رجل ممسك بعنان فرسه في سبيل…١٣٠٠إن في الجنة مائة درجة أعدها للمجاهدين في سبيل الله…١٣٠١من رضي بالله ربًا وبالإسلام دينًا، وبمحمد رسولا…١٣٠٢إن أبواب الجنة تحت ظلال السيوف١٣٠٣ما اغبرت قدما عبد في سبيل الله فتمسه النار١٣٠٤لا يلج النار رجل بكي من خشية الله حتى يعود اللبن…١٣٠٥وعن ابن عباس، قال: سمعت رسول الله يقول: “عينان لا…١٣٠٦من جهز غازيًا في سبيل الله فقد غزا ومن خلف غازيا…١٣٠٧أفضل الصدقات ظل فسطاط في سبيل الله، ومنيحة خادم في…١٣٠٨وعن أنس أن فتى من أسلم قال: يا رسول الله إني أريد…١٣٠٩لا ينبعث من كل رجلين أحدهما، والأجر بينهما١٣١٠أسلم، ثم قاتل١٣١١وعن أنس أن النبي قال: "ما أحد يدخل الجنة يحب أن…١٣١٢يغفر الله للشهيد كل شيء إلا الدين١٣١٣نعم إن قتلت في سبيل الله وأنت صابر محتسب مقبل غير…١٣١٤فألقى تمرات كن في يده ثم قاتل حتى قتل١٣١٥لا يقدمن أحد منكم إلى شيء حتى أكون أنا دونه١٣١٦إن إخوانكم قد قتلوا وإنهم قالوا: اللهم بلغ عنا…١٣١٧وعنه قال: غاب عمي أنس بن النضر عن قتال بدر، فقال:…١٣١٨وعن سمرة قال: قال رسول الله : “رأيت الليلة رجلين…١٣١٩يا أم حارثة إنها جنان في الجنة، وإن ابنك أصاب…١٣٢٠وعن جابر بن عبد الله قال: جيء بأبي إلى النبي قد…١٣٢١من سأل الله تعالى الشهادة بصدق بلغه الله منازل…١٣٢٢من طلب الشهادة صادقًا أعطيها ولو لم تصبه١٣٢٣ما يجد الشهيد من مس القتل إلا كما يجد أحدكم من مس…١٣٢٤أيها الناس، لا تتمنوا لقاء العدو، وسلو الله…١٣٢٥ثنتان لا تردان، أو قلما تردان: الدعاء عند النداء…١٣٢٦اللهم أنت عضدي ونصيري، بك أحول وبك أجول وبك أصول،…١٣٢٧وعن أبي موسى، أن النبي كان إذا خاف قومًا قال:…١٣٢٨الخيل معقود في نواصيها الخير إلى يوم القيامة١٣٢٩وعن عروة البارقي أن النبي قال: الخيل معقود في…١٣٣٠وعن أبي هريرة قال: قال رسول الله : "من احتبس فرسًا…١٣٣١لك بها يوم القيامة سبعمائة ناقة كلها مخطومة١٣٣٢. ((رواه مسلم)). وعنه قال: سمعت رسول الله يقول:١٣٣٣ستفتح عليكم أرضون ويكفيكم الله فلا يعجز أحدكم أن…١٣٣٤من عُلم الرمي ثم تركه فليس منا أو فقد عصى١٣٣٥إن الله يدخل بالسهم الواحد ثلاثة نفر الجنة: صانعه…١٣٣٦ارموا بني إسماعيل فإن أباكم كان راميًا١٣٣٧وعن عمرو بن عبسة قال: سمعت رسول الله يقول: من رمى…١٣٣٨من أنفق نفقة في سبيل الله كتب له سبعمائة ضعف١٣٣٩ما من عبد يصوم يومًا في سبيل الله إلا باعد الله…١٣٤٠من صام يومًا في سبيل الله جعل الله بينه وبين النار…١٣٤١من مات ولم يغزُ، ولم يحدث نفسه بالغزو، مات على…١٣٤٢إن بالمدينة لرجالا ما سرتم مسيرًا، ولا قطعتم…١٣٤٣من قاتل لتكون كلمة الله هي العليا، فهو في سبيل الله١٣٤٤ما من غازية أو سرية تغزو، فتغنم وتسلم، إلا كانوا…١٣٤٥إن سياحة أمتي الجهاد في سبيل الله، عز وجل١٣٤٦قفلة كغزوة١٣٤٧وعن السائب بن يزيد، قال: لما قدم النبي من غزوة…١٣٤٨وعن أبي أمامة عن النبي قال: من لم يغزُ أو يجهز…١٣٤٩جاهدوا المشركين بأموالكم وأنفسكم وألسنتكم١٣٥٠وعن أبي عمرو ويقال: أبو حكيم النعمان بن مقرن قال:…١٣٥١لا تتمنوا لقاء العدو، فإذا لقيتموه فاصبروا١٣٥٢الحرب خُدعة١٣٥٣الشهداء خمسة: المطعون والمبطون، والغريق، وصاحب…١٣٥٤ما تعدون الشهداء فيكم؟ قالوا: يا رسول الله من قتل…١٣٥٥وعن عبد الله بن عمرو بن العاص، قال: قال رسول الله…١٣٥٦من قتل دون ماله فهو شهيد، ومن قتل دون دمه فهو…١٣٥٧فلا تعطه مالك١٣٥٨من أعتق رقبة مسلمة أعتق الله بكل عضو منه عضوًا منه…١٣٥٩الإيمان بالله، والجهاد في سبيل الله١٣٦٠إنك امرؤ فيك جاهلية١٣٦١إذا أتى أحدكم خادمه بطعامه، فإن لم يجلسه معه…١٣٦٢إن العبد إذا نصح لسيده، وأحسن عبادة الله، فله أجره…١٣٦٣للعبد المملوك المصلح أجران، والذي نفس أبي هريرة…١٣٦٤المملوك الذي يحسن عبادة ربه ويؤدي إلى سيده الذي…١٣٦٥ثلاثة لهم أجران: رجل من أهل الكتاب آمن بنبيه، وآمن…١٣٦٦العبادة في الهرج كهجرة إلي١٣٦٧دعوه فإن لصاحب الحق مقالا١٣٦٨رحم الله رجلا سمحًا إذا باع وإذا اشترى وإذا اقتضى١٣٦٩من سره أن ينجيه الله من كرب يوم القيامة، فلينفس عن…١٣٧٠كان رجل يداين الناس، وكان يقول لفتاه: إذا أتيت…١٣٧١حوسب رجل ممن كان قبلكم فلم يوجد له من الخير شيء…١٣٧٢أنا أحق بذا منك، تجاوزوا عن عبدي١٣٧٣من أنظر معسرًا أو وضع له، أظله الله يوم القيامة…١٣٧٤وعن جابر، أن النبي اشترى منه بعيرًا فوزن له فأرجح.…١٣٧٥زن وأرجح
