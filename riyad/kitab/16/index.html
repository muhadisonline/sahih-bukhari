٥٧ حديثًا

١٤٠٨كلمتان خفيفتان على اللسان، ثقيلتان في الميزان،…١٤٠٩وعنه قال: قال رسول الله : لأن أقول: سبحان الله،…١٤١٠من قال لا إله إلا الله وحده لا شريك له، له الملك،…١٤١١من قال لا إله إلا الله وحده لا شريك له، له الملك…١٤١٢ألا أخبرك بأحب الكلام إلى الله؟ إن أحب الكلام إلى…١٤١٣الطهور شطر الإيمان، والحمد لله تملأ الميزان،…١٤١٤قل لا إله إلا الله وحده لا شريك له، الله أكبر…١٤١٥اللهم أنت السلام ومنك السلام تباركت يا ذا الجلال…١٤١٦لا إله إلا الله وحده لا شريك له، له الملك وله…١٤١٧وعن عبد الله ابن الزبير أنه كان يقول دبر كل صلاة،…١٤١٨ذهب أهل الدثور بالدرجات العلى، والنعيم المقيم:…١٤١٩وعنه عن رسول الله قال: "من سبح الله في دبر كل صلاة…١٤٢٠معقبات لا يخيب قائلهن -أو فاعلهن- دبر كل صلاة…١٤٢١وعن سعد بن أبي وقاص أن رسول الله كان يتعوذ دبر…١٤٢٢يا معاذ والله إني لأحبك١٤٢٣إذا تشهد أحدكم فليستعذ بالله من أربع، يقول: اللهم…١٤٢٤اللهم اغفر لي ما قدمت وما أخرت، وما أسررت وما…١٤٢٥وعن عائشة قالت: كان النبي يكثر أن يقول في ركوعه…١٤٢٦سبوح قدوس رب الملائكة والروح١٤٢٧فأما الركوع فعظموا فيه الرب، وأما السجود فاجتهدوا…١٤٢٨أقرب ما يكون العبد من ربه وهو ساجد، فأكثروا الدعاء١٤٢٩وعنه أن رسول الله كان يقول في سجوده: “اللهم اغفر…١٤٣٠سبحانك وبحمدك لا إله إلا أنت١٤٣١أيعجز أحدكم أن يكسب في كل يوم ألف حسنة!١٤٣٢وعن أبي ذر أن رسول الله قال: “يصبح على كل سلامى من…١٤٣٣مازلت على الحالة التي فارقت عليها؟١٤٣٤مثل الذي يذكر ربه والذي لا يذكره، مثل الحي والميت١٤٣٥يقول الله تعالى: أنا عند ظن عبدي بي، وأنا معه إذا…١٤٣٦سبق المفردون١٤٣٧أفضل الذكر: لا إله إلا الله١٤٣٨لا يزال لسانك رطبًا من ذكر الله١٤٣٩من قال: سبحان الله وبحمده، غرست له نخلة في الجنة١٤٤٠لقيت إبراهيم ليلة أسري بي، فقال: يا محمد أقرئ أمتك…١٤٤١ألا أنبئكم بخير أعمالكم، وأزكاها عند مليككم،…١٤٤٢أخبرك بما هو أيسر عليك من هذا - أو أفضل١٤٤٣ألا أدلك على كنز من كنوز الجنة؟١٤٤٤وعن عائشة قالت: كان رسول الله يذكر الله تعالى على…١٤٤٥لو أن أحدكم إذا أتى أهله قال: بسم الله، اللهم…١٤٤٦باسمك اللهم أموت وأحيا١٤٤٧إن لله تعالى ملائكة يطوفون في الطرق يلتمسون أهل…١٤٤٨لا يقعد قوم يذكرون الله عز وجل إلا حفتهم الملائكة،…١٤٤٩وعن أبي واقد الحارث بن عوف أن رسول الله ، بينما هو…١٤٥٠ما أجلسكم؟١٤٥١من قال حين يصبح وحين يمسي: سبحان الله وبحمده مائة…١٤٥٢أما لو قلت حين أمسيت: أعوذ بكلمات الله التامات من…١٤٥٣اللهم بك أصبحنا، وبك أمسينا، وبك نحيا، وبك نموت،…١٤٥٤اللهم فاطر السماوات والأرض عالم الغيب والشهادة، رب…١٤٥٥قال الراوي: أراه قال فيهن:١٤٥٦اقرأ: قل هو الله أحد، والمعوذتين حين تمسي وحين…١٤٥٧ما من عبد يقول في صباح كل يوم ومساء كل ليلة: بسم…١٤٥٨باسمك اللهم أحيا وأموت١٤٥٩إذا أويتما إلى فراشكما، أو: إذا أخذتما مضجعكما -…١٤٦٠إذا أوى أحدكم إلى فراشه، فلينفض فراشه بداخلة إزاره…١٤٦١أن النبي ، كان إذا أوى إلى فراشه كل ليلة جمع كفيه،…١٤٦٢إذا أتيت مضجعك فتوضأ وضوءك للصلاة، ثم اضطجع على…١٤٦٣الحمد لله الذي أطعمنا وسقانا، وكفانا وآوانا، فكم…١٤٦٤اللهم قني عذابك يوم تبعث عبادك
