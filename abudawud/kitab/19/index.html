كتاب البيوع

٢٤٥ حديثًا

باب في التجارة يخالطها الحلف واللغو

٣٣٢٦يا مَعشَرَ التجّار، إن البيعَ يحضُره اللغوُ…٣٣٢٧يحضُره الحلف والكذبُ

باب في استخراج المعادن

٣٣٢٨مِن أين أصَبْتَ هذا الذهبَ؟

باب اجتناب الشبهات

٣٣٢٩إن الحلال بَيِّنٌ، وإن الحرامَ بَيِّنٌ، وبينهما…٣٣٣٠حدَّثنا إبراهيمُ بن موسى الرازيُّ، أخبرنا عيسى،…٣٣٣١لَيأتيننَّ على النَّاس زمانٌ لا يبقَى أحدٌ إلا أكل…٣٣٣٢أوْسِعْ من قِبَلِ رجلَيه، أوسِعْ من قِبَل رأسِه

باب في آكل الربا وموكله

٣٣٣٣حدَّثنا أحمدُ بن يونَس، حدَّثنا زهيرٌ، حدَّثنا…

باب في وضع الربا

٣٣٣٤ألا إنَّ كلَّ رباً من ربا الجاهليةِ مَوضوعٌ، لكم…

باب كراهية اليمين في البيع

٣٣٣٥الحَلِفُ مَنْفَقةٌ للسِّلعةِ مَمْحَقةٌ للبركة -قال…

باب في الرجحان في الوزن، والوزن بالأجر

٣٣٣٦زِنء وأَرْجِحْ٣٣٣٧حدَّثنا حفصُ بن عُمر ومسلمُ بن إبراهيمَ -المعنى…٣٣٣٨حدَّثنا ابنُ أبي رِزْمَةَ، سمعتُ أبي يقول: قال…٣٣٣٩حدَّثنا أحمدُ بن حنبل، حدَّثنا وكيع، قال: قال…

باب في قول النبي -صلى الله عليه وسلم-: "المكيال مكيال أهل المدينة"

٣٣٤٠الوزْنُ وزْنُ أهلِ مكةَ، والمكيالُ مكيالُ أهلِ…

باب التشديد في الدين

٣٣٤١ها هُنا أحدٌ من بني فلانِ؟٣٣٤٢إنَّ من أعظمِ الذنوبِ عندَ الله أن يَلقاهُ بها…٣٣٤٣أعليه دين؟٣٣٤٤لا أشتري بعدَها شيئاً إلا وعندي ثمنُه

باب في المطل

٣٣٤٥مَطْلُ الغنى ظُلمٌ، وإذا أُتبِعَ أحدُكم على مَليءٍ…

باب في حسن القضاء

٣٣٤٦أعْطِه إياه، فإن خِيارَ الناسِ أحسنُهم قضاء٣٣٤٧حدَّثنا أحمدُ بن حنبلٍ، حدَّثنا يحيى، عن مِسعَرٍ،…

باب في الصرف

٣٣٤٨الذهبُ بالورِقِ رِباً إلا هاءَ وهاءَ، والبُرُّ…٣٣٤٩الذهب بالذهبِ تِبْرُها وعَينُها، والفضةُ بالفضةِ…٣٣٥٠فإذا اختلفتْ هذه الأصنافُ فبيعوا كيف شئتُم، إذا…

باب في حلية السيف تباع بالدراهم

٣٣٥١لا، حتى تُمَيِّزَ بينه وبينه٣٣٥٢لا تُبَاع حتى تُفَصَّل٣٣٥٣لا تَبِيعُوا الذهبَ بالذهب إلا وَزناً بوَزن

باب اقتضاء الذهب من الورق

٣٣٥٤لا بَأسَ أن تأخُذَها بسعْرِ يومِها، ما لم…٣٣٥٥بِسعْرِ يومِها

باب في الحيوان بالحيوان نسيئة

٣٣٥٦حدَّثنا موسى بن إسماعيلَ، حدَّثنا حمادٌ، عن…

باب في الرخصة فيه

٣٣٥٧حدَّثنا حفصُ بن عُمر، حدَّثنا حمادُ بن سلمةَ، عن…

باب في ذلك إذا كان يدا بيد

٣٣٥٨حدَّثنا يزيدُ بن خالدٍ الهَمْدانيُّ وقتيبةُ بن…

باب في الثمر بالتمر

٣٣٥٩. وقال: سمعتُ رسولَ الله -صلَّى الله عليه وسلم-…٣٣٦٠حدَّثنا الربيعُ بن نافع أبو توبةَ، حدَّثنا معاويةُ…

باب في المزابنة

٣٣٦١حدَّثنا أبو بكر بن أبي شيبةَ، حدَّثنا ابن أبي…

باب في بيع العرايا

٣٣٦٢حدَّثنا أحمدُ بن صالحٍ، حدَّثنا ابنُ وهبٍ، أخبرني…٣٣٦٣حدَّثنا عثمانُ بن أبي شيبةَ، حدَّثنا ابنُ عُيينةَ،…

باب في مقدار العرية

٣٣٦٤حدَّثنا عبد الله بن مَسْلَمةَ، حدَّثنا مالك، عن…

باب تفسير العرايا

٣٣٦٥حدَّثنا أحمدُ بن سعيد الهَمْدَانيُّ، حدَّثنا ابنُ…٣٣٦٦حدَّثنا هنّادُ بن السَّريِّ، عن عبدةَ عن ابن…

باب في بيع الثمار قبل أن يبدو صلاحها

٣٣٦٧حدَّثنا عبدُ الله بن مَسلمةَ القَعْنبيُّ، عن…٣٣٦٨حدَّثنا عبدُ الله بن محمد النُّفَيليُّ،،، حدَّثنا…٣٣٦٩حدَّثنا حفصُ بن عُمر النَّمَريُّ، حدَّثنا شعبةُ،…٣٣٧٠تَحْمارُّ وتَصْفارُّ، ويُؤكَل منها٣٣٧١حدَّثنا الحسنُ بن عليٍّ، حدَّثنا أبو الوليد، عن…٣٣٧٢فإمَّا لا، فلا تَتبايَعوا الثمَرةَ حتى يَبدُوَ…٣٣٧٣حدَّثنا إسحاقُ بن إسماعيلَ الطالْقانيُّ، حدَّثنا…

باب في بيع السنين

٣٣٧٤حدَّثنا أحمدُ بن حنبل ويحيى بنُ معَينٍ، قالا:…٣٣٧٥حدَّثنا مُسدَّدٌ، حدَّثنا حماد، عن أيوبَ، عن أبي…

باب في بيع الغرر

٣٣٧٦حدَّثنا أبو بكر وعثمانُ ابنا أبي شيبةَ، قالا:…٣٣٧٧حدَّثنا قتيبةُ بن سَعيدٍ وأحمدُ بن عَمرو بن…٣٣٧٨حدَّثنا الحسنُ بن عليٍّ، حدَّثنا عبدُ الرزَّاق،…٣٣٧٩حدَّثنا أحمدُ بن صالحِ، حدَّثنا عنبسةُ، حدَّثنا…٣٣٨٠حدَّثنا عبدُ الله بن مَسلمةَ، عن مالكِ، عن نافع عن…٣٣٨١حدَّثنا أحمدُ بن حنبل، حدَّثنا يحيى، عن عُبيدِ…٣٣٨٢حدَّثنا محمدُ بن عيسى، حدَّثنا هُشَيمٌ، أخبرنا…

باب في الشركة

٣٣٨٣إن الله يقول: أنا ثالثُ الشريكَين، ما لم يَخُنْ…

باب في المضارب يخالف

٣٣٨٤حدَّثنا مُسدَّدٌ، حدَّثنا سفيانُ، عن شَبيبِ بن…٣٣٨٥حدَّثنا الحسنُ بن الصبَّاح، حدَّثنا أبو المُنذرِ،…٣٣٨٦حدَّثنا محمدُ بن كثير العبْديُّ، أخبرنا سفيانُ،…

باب في الرجل يتجر في مال الرجل بغير إذنه

٣٣٨٧من استطاع منكم أن يكون مثل صاحب فَرَقِ الأرزِّ…

باب في الشركة على غير رأس مال

٣٣٨٨حدَّثنا عُبيد الله بن معاذ، حدَّثنا يحيى، حدَّثنا…

باب في المزارعة

٣٣٨٩لأن يمنحَ أحدُكم أرضَه خيرٌ من أن يأخذ خَراجاً…٣٣٩٠إنْ كان هذا شأنكم فلا تُكرُوا المزارعَ٣٣٩١حدَّثنا عثمانُ بن أبي شيبةَ، حدَّثنا يزيدُ بن…٣٣٩٢حدَّثنا إبراهيمُ بن موسى الرازيُّ، أخبرنا عيسى،…٣٣٩٣حدَّثنا قتيبةُ بن سعيدٍ، عن مالكٍ، عن ربيعةَ بن…

باب التشديد في ذلك

٣٣٩٤حدَّثنا عبدُ الملك بن شُعيبِ بن الليثِ، حدَّثني…٣٣٩٥مَن كانت له أرضٌ فليَزرَعْها، أو فليُزْرِعْها…٣٣٩٦حدَّثنا محمدُ بن عُبيدٍ، حدَّثنا حمادُ بن زيدٍ، عن…٣٣٩٧حدَّثنا أبو بكر بن أبي شيبةَ، حدَّثنا وكيعٌ،…٣٣٩٨مَنِ اسْتَغنى عن أرضه فليَمنحْها أخاه أو لِيَدَعْ٣٣٩٩ما أحسنَ زرعَ ظُهَيْرٍ!٣٤٠٠إنما يزرعُ ثلاثة: رجل له أرضٌ فهو يزرعُها، ورجل…٣٤٠١قرأتُ على سعيدِ بن يعقوبَ الطالْقاني: حدَّثكم ابنُ…٣٤٠٢فقال: زرعي ببَذْرِي وعملي، ليَ الشَّطْرُ ولبني…

باب إذا زرع الرجل في الأرض بغير إذن صاحبها

٣٤٠٣مَن زَرَعَ في أرضِ قومٍ بغيرِ إذنِهم، فليس لهُ مِن…

باب في المخابرة

٣٤٠٤حدَّثنا أحمدُ بن حنبل، حدَّثنا إسماعيلُ. وحدَّثنا…٣٤٠٥حدَّثنا أبو حفصٍ عُمر بن يزيدَ السَّيَّاري،…٣٤٠٦من لم يَذَرِ المُخابرةَ، فليؤذن بحرب من الله ورسوله٣٤٠٧حدَّثنا أبو بكر بن أبي شيبةَ، حدَّثنا عُمر بن…

باب في المساقاة

٣٤٠٨حدَّثنا أحمدُ بن حنبلٍ، حدَّثنا يحيى، عن عبيدِ…٣٤٠٩حدَّثنا قتيبةُ بن سعيدٍ، عن الليثِ، عن محمد بن عبد…٣٤١٠حدَّثنا أيوبُ بن محمد الرَّقِّيُّ، حدَّثنا عُمر بن…٣٤١١وكل صفراء وبيضاء٣٤١٢حدَّثنا محمد بن سليمانَ الأنباريُّ، حدَّثنا كثير…

باب في الخرص

٣٤١٣حدَّثنا يحيى بن مَعين، حدَّثنا حجاجٌ، عن ابن جُريج…٣٤١٤حدَّثنا ابن أبي خَلَفٍ، حدَّثنا محمدُ بن سابقٍ، عن…٣٤١٥حدَّثنا أحمدُ بن حنبل، حدَّثنا عبدُ الرزاق ومحمد…

باب في كسب المعلم

٣٤١٦إن كنتَ تُحبُّ أن تُطَوَّقَ طَوقاً من نارٍ فاقبَلْها٣٤١٧جَمرَةٌ بين كتفَيك تَقَلّدْتَها -أو تعلقْتَها-

باب في كسب الأطباء

٣٤١٨مِن أين علمتم أنها رُقْيَةٌ؟ أحسنتم، واضْرِبُوا لي…٣٤١٩حدَّثنا الحسن بن علي، حدَّثنا يزيدُ بنُ هارون،…٣٤٢٠كُلْ، فَلَعَمْري لَمَنْ أكَلَ برقْيَةِ باطل، لقد…

باب في كسب الحجام

٣٤٢١كَسْبُ الحَجَّامِ خبيث، وثمنُ الكلب خبيث، ومهر…٣٤٢٢حدَّثنا عبدُ الله بنُ مَسْلَمَةَ القعنبيُّ، عن…٣٤٢٣حدَّثنا مُسَدَّدٌ، حدَّثنا يزيد بنُ زُرَيع،…٣٤٢٤حدَّثنا القعنبيُّ، عن مالك، عن حُمَيدٍ الطويلِ عن…

باب في كسب الإماء

٣٤٢٥حدَّثنا عُبيد الله بنُ معاذ، حدَّثنا أبي، حدَّثنا…٣٤٢٦حدَّثنا هارونُ بنُ عبد الله، حدَّثنا هاشم بن…٣٤٢٧حدَّثنا أحمدُ بن صالح، حدَّثنا ابن أبي فُدَيكِ، عن…

باب في حلوان الكاهن

٣٤٢٨حدَّثنا قتيبة بن سعيد، عن سفيان، عن الزهري، عن أبي…

باب في عسب الفحل

٣٤٢٩حدَّثنا مُسَدَّدُ بن مُسَرْهَدِ، حدَّثنا إسماعيلُ،…

باب في الصائغ

٣٤٣٠إني وهبتُ لخالتي غُلاماً، وأنا أرجو أن يبارَكَ لها…٣٤٣١حدَّثنا يوسف بن موسى، حدَّثنا سَلَمَةُ بنُ الفضل،…٣٤٣٢حدَّثنا الفضل بن يُعقوب، حدَّثنا عبد الأعلى، عن…

باب في العبد يباع وله مال

٣٤٣٣مَنْ بَاعَ عَبْداً وله مال، فمالُه للبائع إلا أن…٣٤٣٤حدَّثنا القعنبيُّ، عن مالك، عن نافع، عن ابنِ عمر…٣٤٣٥من باع عبدا وله مال، فمالُه للبائع، إلا أن يشترط…

باب في التلقي

٣٤٣٦لا يَبعْ بَغضُكُم على بَيعِ بعض، ولا تَلَقَّوا…٣٤٣٧حدَّثنا الربيعُ بنُ نافع أبو تَوبةَ، حدَّثنا عُبيد…

باب في النهي، عن النجش

٣٤٣٨لا تَنَاجَشُوا

باب النهي أن يبيع حاضر لباد

٣٤٣٩حدَّثنا محمد بن عُبَيدٍ، حدَّثنا محمد بن ثور، عن…٣٤٤٠لا يَبِيعُ حاضرٌ لبادِ، وإن كان أخاه أو أباه٣٤٤١حدَّثنا موسى بنُ إسماعيل، حدَّثنا حمَّاد، عن محمد…٣٤٤٢لا يَبِيعُ حَاضِرٌ لِبَادِ، وَذَرُوا النَّاسَ…

باب من اشترى مصراة فكرهها

٣٤٤٣لا تَلَقَّوُا الرُّكْبَانَ لِلبَيعِ، ولا يَبِعْ…٣٤٤٤من اشترى شَاةً مُصَرَّاةَ، فهو بالخِيار ثلاثة…٣٤٤٥من اشترى غَنَماً مُصَرَّاةً احْتَلَبَها: فإن رضيها…٣٤٤٦من ابتاع مُحَفَّلَةً، فهو بالخيار ثلاثة أيام، فإن…

باب النهي عن الحكرة

٣٤٤٧لا يَحْتكِرُ إلا خَاطِىءٌ٣٤٤٨حدَّثنا محمد بن يحيى بن فياض، حدَّثنا أبي.…

باب كسر الدراهم

٣٤٤٩حدَّثنا أحمدُ بنُ محمد بن حنبل، حدَّثنا معتمِرٌ،…

باب في التسعير

٣٤٥٠بل أدعو٣٤٥١حدَّثنا عثمانُ بنُ أبي شيبةَ، حدَّثنا عفانُ،…

باب في النهي عن الغش

٣٤٥٢كيف تبيعُ؟٣٤٥٣حدَّثنا الحسنُ بنُ الصبَّاح، عن علي، عن يحيى، قال:…

باب خيار المتبايعين

٣٤٥٤المتبايعانِ كُلُّ واحدٍ منهما بالخِيارِ على صاحبه…٣٤٥٥أو يقول أحدهما لصاحبه: اخْتَرْ٣٤٥٦المُتَبايعانِ بالخِيار ما لَمْ يَفترِقَا، إلا أن…٣٤٥٧البيِّعَانِ بالخِيار ما لم يتفرّقا٣٤٥٨لا يفترِقَنَّ اثنان إلا عَنْ تراضٍ٣٤٥٩البَيِّعان بالخِيار ما لم يفْتَرِقا، فإن صَدَقا…

باب في فضل الإقالة

٣٤٦٠حدَّثنا يحيى بُن مَعينِ، حدَّثنا حفْصٌ، عن الأعمش،…

باب فيمن باع بيعتين في بيعة

٣٤٦١حدَّثنا أبو بكر بنُ أبي شيبةَ، عن يحيى بن زكريا،…

باب في النهي عن العينة

٣٤٦٢إذا تبايعتُم بالعِينَةِ، وأخذتم أذنابَ البقرِ،…

باب في السلف

٣٤٦٣من أسلفَ في تَمرٍ فليُسْلفِ في كَيْلٍ معلوم،…٣٤٦٤حدَّثنا حفصُ بنُ عمر، حدَّثنا شعبة. وحدَّثنا محمد…٣٤٦٥حدَّثنا محمد بن بشار، حدَّثنا يحيى وابنُ مهديّ،…٣٤٦٦حدَّثنا محمد بنُ المصفَّى، حدَّثنا أبو المغيرة،…

باب في السلم في ثمرة بعينها

٣٤٦٧ثم قال:

باب من أسلف في شيء ثم حوله إلى غيره

٣٤٦٨من أسلفَ في شيءِ فلا يصرِفْه إلى غيره

باب في وضع الجائحة

٣٤٦٩تصدَّقُوا عليه٣٤٧٠إن بعتَ مِن أخيك تمراً، فأصابتها جائحةٌ، فلا…

باب تفسير الجائحة

٣٤٧١حدَّثنا سليمانُ بن داود المَهْرِيّ، أخبرنا ابنُ…٣٤٧٢حدَّثنا سليمانُ بن داودَ، أخبرنا ابنُ وهب، أخبرني…

باب في منع الماء

٣٤٧٣لا يُمْنَعُ فضلُ الماءِ ليُمنعَ به الكلأُ٣٤٧٤ثَلاثَة لا يكلِّمهمُ الله يومَ القيامة: رجلٌ…٣٤٧٥ولا يزكِّيهم ولهم عذابٌ أليم٣٤٧٦الماءُ٣٤٧٧حدَّثنا علي بنُ الجَعْدِ اللؤلؤيُّ، أخبرنا حَرِيزُ…

باب في بيع فضل الماء

٣٤٧٨حدَّثنا عبدُ الله بنُ محمد النُّفيلي، حدَّثنا…

باب في ثمن السنور

٣٤٧٩حدَّثنا إبراهيمُ بنُ موسى الرازيُّ، أخبرنا.…٣٤٨٠حدَّثنا أحمدُ بنُ حنبل، حدَّثنا عبدُ الرزاق،…

باب في أثمان الكلاب

٣٤٨١حدَّثنا قُتيبة بنُ سعيدٍ، حدَّثنا سفيانُ، عن…٣٤٨٢حدَّثنا الربيعُ بن نافعِ أبو توبةَ، حدَّثنا عُبيد…٣٤٨٣حدَّثنا أبو الوليد الطيالسيُّ، حدَّثنا شعبة أخبرني…٣٤٨٤لا يَحِل ثمنُ الكلب، ولا حُلوانُ الكاهِنِ، ولا…

باب في ثمن الخمر والميتة

٣٤٨٥إن الله حَرَّمَ الخمرَ وثمنَها، وحَرمَ الميتة…٣٤٨٦فقيل: يا رسول الله - صلَّى الله عليه وسلم -،…٣٤٨٧هو حَرامٌ٣٤٨٨لَعَنَ الله اليهودَ! -ثلاثاً- إن الله حرم عليهم…٣٤٨٩مَنْ بَاعَ الخَمرَ فَلْيُشَقِّصِ الخَنازيرَ٣٤٩٠حُرِّمَتِ التجارةُ في الخمرِ٣٤٩١حدَّثنا عثمانُ بنُ أبي شيبةَ، حدَّثنا أبو معاويةَ،…

باب في بيع الطعام قبل أن يستوفى

٣٤٩٢حدَّثنا عبدُ الله بنُ مسلمةَ، عن مالك، عن نافع عن…٣٤٩٣حدَّثنا عبدُ الله بنُ مَسلمةَ، عن مالك، عن نافع عن…٣٤٩٤حدَّثنا أحمدُ بنُ حنبل، حدَّثنا يحيى، عن عُبيد…٣٤٩٥حدَّثنا أحمدُ بنُ صالحْ، حدَّثنا ابنُ وهب، حدَّثنا…٣٤٩٦من ابتاعَ طَعاماً فلا يَبِعْه حتى يكتاله٣٤٩٧إذا اشترى أحدُكُم طعاماً فلا يَبِعْه حتى يَقبِضَه٣٤٩٨حدَّثنا الحسنُ بن علي، حدَّثنا عبدُ الرزاق،…٣٤٩٩حدَّثنا محمدُ بنُ عَوفٍ الطائيُّ، حدَّثنا أحمدُ…

باب في الرجل يقول عند البيع: "لا خلابة"

٣٥٠٠وإذا بايعتَ فقل: لا خِلابَة٣٥٠١إن كنتَ غَيرَ تاركٍ البيع فقل: هاء وهاء ولا خِلابَةَ

باب في العربان

٣٥٠٢حدَّثنا عبدُ الله بنُ مَسلمةَ، قال: قرأتُ على مالك…

باب في الرجل يبيع ما ليس عنده

٣٥٠٣لا تَبِعْ ما لَيسَ عندَك٣٥٠٤إلا يَحِلُّ سَلفٌ وبَيْعٌ، ولا شرطانِ في بَيعٍ،…

باب في شرط في بيع

٣٥٠٥تُراني إنما ماكَسْتُكَ لأذْهَبَ بجملِكَ؟! خُذْ…

باب في عهدة الرقيق

٣٥٠٦عُهدةُ الرقِيقِ ثلاثةُ أيام٣٥٠٧حدَّثنا هارونُ بن عبد الله، حدَّثني عبدُ الصمدِ،…

باب فيمن اشترى عبدا فاستعمله، ثم وجد به عيبا

٣٥٠٨الخَرَاجُ بالضَّمَانِ٣٥٠٩الخَرَاجُ بالضمَانِ٣٥١٠الخَرَاجُ بالضمَانِ

باب إذا اختلف البيعان والبيع قائم

٣٥١١إذا اختلف البيِّعَان ولَيْسَ بينَهما بينة، فهو ما…٣٥١٢حدَّثنا عبدُ الله بن محمد النُّفيليُّ، حدَّثنا…

باب في الشفعة

٣٥١٣الشُفعةُ في كل شِرْكٍ رَبْعَةٍ أو حائِطِ، لا…٣٥١٤حدَّثنا أحمد بن حنبلٍ، حدَّثنا عبدُ الرزاق، أخبرنا…٣٥١٥حدَّثنا محمد بنُ يحيى بنِ فارس، حدَّثنا الحسنُ بن…٣٥١٦الجارُ أحقُّ بِسَقَبِهِ٣٥١٧جارُ الدَّارِ أحقُّ بدارِ الجار أو الأرضِ٣٥١٨الجارُ أحَقُّ بِشُفعة جَارِه: يُنْتَظَر بها وإن…

باب في الرجل يفلس فيجد الرجل ماله بعينه عنده

٣٥١٩حدَّثنا عبدُ الله بن مَسلمةَ، عن مالك (ح) وحدَّثنا…٣٥٢٠أيُّما رجلِ باعَ متاعاً فأفلسَ الذي ابتاعَه ولم…٣٥٢١وإن كان قد قضى من ثمنها شيئاً، فهو أُسوةُ الغرماء٣٥٢٢فإن كان قضاه من ثمنها شيئاً فما بقي هو أسوةُ…٣٥٢٣مَنْ أفْلَسَ أو ماتَ، فوجد رَجُلٌ متَاعَهُ بعينه…

باب فيمن أحيا حسيرا

٣٥٢٤مَنْ وجد دَابةً قد عَجَزَ عنها أهْلُها أن…٣٥٢٥حدَّثنا محمدُ بن عُبيدِ، عن حماد -يعني ابنَ زيدٍ-…

باب في الرهن

٣٥٢٦لَبَن الدَّرِّ يُحْلَبُ بِنَفَقَتِه، إذا كانَ…٣٥٢٧إن مِنْ عِبادِ الله لأناساً ما هُمْ بأنبياء، ولا…

باب في الرجل يأكل من مال ولده

٣٥٢٨إن مِن أطيبِ ما أكلَ الرجُلُ من كَسْبه، وَوَلدُهُ…٣٥٢٩وَلَدُ الرّجُلِ مِن كسْبِهِ، مِن أطيبِ كسْبِه،…٣٥٣٠أنت ومالُكَ لِوالدك، إنَّ أولادَكم مِن اْطيبِ…

باب الرجل يجد عين ماله عند رجل

٣٥٣١مَن وجَدَ عَينَ مالِه عندَ رَجُلٍ، فهو أحَقُّ به…

باب في الرجل يأخذ حقه من تحت يده

٣٥٣٢خُذِي ما يكْفِيكِ وبَنِيكِ بالمعروفِ٣٥٣٣لا حَرَجَ عليكِ أن تُنفقي عليهم بالمعروف٣٥٣٤أدِّ الأمانَةَ إلى مَنِ ائتَمَنَكَ، ولا تَخُنْ…٣٥٣٥أدِّ الأمانةَ إلى مَنِ ائْتمَنكَ، ولا تَخُن مَنْ…

باب في قبول الهدايا

٣٥٣٦حدَّثنا عليُّ بنُ بحر وعبدُ الرحيم بن مُطَرِّف…٣٥٣٧وَايْمُ اللهِ، لا أقبلُ بَعْدَ يَوْمِي هذا مِن…

باب الرجوع في الهبة

٣٥٣٨العائدُ في هبتِهِ كالعائِدِ في قَيئه٣٥٣٩لا يَحل لِرَجُلٍ أن يُعطِيَ عَطِيَّةً أو يَهَبَ…٣٥٤٠مَثَلُ الذي يَسْترِدُّ ما وهَب كمثل الكلبِ يَقيءُ…

باب الهدية لقضاء الحاجة

٣٥٤١مَنْ شَفَعَ لأخيه شَفَاعةً، فأهدى له هديةً عليها…

باب في الرجل يفضل بعض ولده على بعض في النحل

٣٥٤٢ألكَ وَلَدٌ سِوَاهُ؟٣٥٤٣ما هذا الغُلامُ؟ قال: غلامي أعطانيه أبي، قَالَ:…٣٥٤٤اعدِلُوا بينَ أبنائِكم، اعدِلُوا بينَ أبنائِكم٣٥٤٥فقال: نعم، قال: فكُلَّهم أعطَيتَ ما أعطيتَه؟

باب عطية المرأة بغير إذن زوجها

٣٥٤٦لا يجوز لامرأةٍ أمْرٌ في مالها إذا مَلَكَ زوجُها…٣٥٤٧لا يَجُوُز لامْرأةٍ عَطيَّةٌ إلا بإذنِ زوجها

باب في العمرى

٣٥٤٨العُمرى جائزةٌ٣٥٤٩حدَّثنا أبو الوليد، حدَّثنا هَمَّامٌ، عن قتادةَ،…٣٥٥٠العُمْرَى لِمَنْ وُهِبَتْ له٣٥٥١من أُعْمِرَ عُمْرَى فهىَ لهُ ولِعَقِبهِ يَرِثُها…٣٥٥٢حدَّثنا أحمدُ بنُ أبي الحَواريّ، حدَّثنا الوليدُ،…

باب من قال فيه: ولعقبه

٣٥٥٣أيّما رجلٍ أُعْمِرَ عُمرى له ولعقبهِ، فإنها للذي…٣٥٥٤حدَّثنا حجاجُ بنُ أبي يعقوب، حدَّثنا يعقوبُ بن…٣٥٥٥حدَّثنا أحمدُ بن حنبل، حدَّثنا عبدُ الرزاق، أخبرنا…٣٥٥٦لا تُرْقِبُوا، ولا تُعْمِرُوا، فمن أُرْقِبَ شيئاً…٣٥٥٧هي لها حياتَها وموتَها

باب في الرقبى

٣٥٥٨العُمرَى جائِزَةٌ لأهلِها، والرُّقبَى جائزةٌ لأهلِها٣٥٥٩مَن أَعْمَرَ شيئاً، فهو لِمُعْمَرِهِ محياه…٣٥٦٠حدَّثنا عبدُ الله بن الجرّاحِ، عن عُبيدِ الله بن…

باب في تضمين العارية

٣٥٦١على اليَدِ ما أخذت حتى تُؤَدِّيَ٣٥٦٢لا، بل عاريَّةٌ مضمونَةٌ٣٥٦٣يا صَفوانُ، هل عندَكِ من سلاحٍ؟٣٥٦٤حدَّثنا مُسدَّدٌ، حدَّثنا أبو الأحوص، حدَّثنا عبد…٣٥٦٥إنَّ الله عز وجل قد أعطى كلَّ ذي حق حقهُ، فلا…٣٥٦٦إذا أتتك رُسُلي، فأعطِهِم ثلاثين درعاً، وثلاثين…

باب فيمن أفسد شيئا يضمن مثله

٣٥٦٧غَارَتْ أمُّكُم٣٥٦٨إناءٌ مثلُ إناءٍ، وطعامٌ مِثلُ طَعامٍ
