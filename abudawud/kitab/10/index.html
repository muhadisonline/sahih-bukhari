كتاب الصوم

١٦٤ حديثًا

٢٣١٣حدَّثنا أحمدُ بنُ محمد ابن شَبُّوْيَهْ، حدثني…٢٣١٤حدَّثنا نصرُ بنُ على بنِ نصرٍ الجَهضَمِيُّ، أخبرنا…

باب نسخ قوله تعالى:

٢٣١٥حدَّثنا قتيبةُ بنُ سعيدٍ، حدَّثنا بَكرٌ -يعني ابنَ…٢٣١٦حدَّثنا أحمدُ بنُ محمد، حدثني عليُّ بنُ حسين، عن…

باب من قال: هي مثبتة للشيخ والحبلى

٢٣١٧حدَّثنا موسى بنُ إسماعيل، حدَّثنا أبانُ، حدَّثنا…٢٣١٨حدَّثنا ابن المثنى، حدَّثنا ابنُ أبي عَدي، عن…

باب الشهر يكون تسعا وعشرين

٢٣١٩إنَا أُمَّة أميَّة، لا نكتُبُ، ولا نحسُبُ، الشهرُ…٢٣٢٠الشَهْرُ تِسْعٌ وعِشرونَ، فلا تَصُوموا حتى…٢٣٢١وإن أحْسنَ ما يُقْدَرُ له إذا رأينا هِلالَ شعبانَ…٢٣٢٢حدَّثنا أحمدُ بنُ منيع، عن ابنِ أبي زائدة، عن عيسى…٢٣٢٣شَهرَا عيدٍ لا يَنقُصَانِ: رمضانُ، وذو الحِجة

باب إذا أخطأ القوم الهلال

٢٣٢٤وفِطْرُكم يَوْمَ تُفْطِرُونَ، وأضْحَاكُم يَوْمَ…

باب إذا أغمي الشهر

٢٣٢٥حدَّثنا أحمدُ بن حنبل، حدثني عبدُ الرحمن بنُ مهدي،…٢٣٢٦لا تَقَدِّمُوا الشهرَ حتى تروا الهِلالَ أو تكمِلوا…

باب من قال: فإن غم عليكم فصوموا ثلاثين

٢٣٢٧لا تَقَدَمُوا الشهرَ بصيام يومٍ ولا يومينِ، إلا أن…

باب في التقدم

٢٣٢٨هَلْ صُمْتَ مِنْ سِرَرِ شعبانَ شيئاً؟٢٣٢٩صُومُوا الشَّهْرَ وسِرَّه٢٣٣٠حدَّثنا سليمانُ بنُ عبدِ الرحمن الدمشقي في هذا…٢٣٣١حدَّثنا أحمدُ بنُ عبدِ الواحدِ، حدَّثنا أبو…

باب إذا رئي الهلال في بلد قبل الآخرين بليلة

٢٣٣٢حدَّثنا موسى بنُ إسماعيلَ، حدَّثنا إسماعيلُ -يعني…٢٣٣٣حدَّثنا عُبيدُ الله بن مُعاذٍ، حدثني أبي، حدَّثنا…

باب كراهية صوم يوم الشك

٢٣٣٤حدَّثنا محمدُ بنُ عبدِ الله بن نُمَير، حدَّثنا أبو…

باب فيمن يصل شعبان برمضان

٢٣٣٥لا تَقَدمُوا صَوْمَ رمضَانَ بيوْمٍ ولا يومين، إلا…٢٣٣٦حدَّثنا أحمدُ بنُ حنبل، حدَّثنا محمدُ بنُ جعفر،…

باب في كراهية ذلك

٢٣٣٧إذا انتَصَفَ شعبانُ، فلا تَصُوموا

باب شهادة رجلين على رؤية هلال شوال

٢٣٣٨حدَّثنا محمدُ بنُ عبدِ الرحيم أبو يحيى البزاز،…٢٣٣٩حدَّثنا مُسَدَدٌ وخلفُ بنُ هشام المُقرئ، قالا:…

باب في شهادة الواحد على رؤية هلال رمضان

٢٣٤٠أتَشْهَدُ أن لا إله إلا الله؟٢٣٤١أتَشْهَدُ أنْ لا إله إلا الله، وأني رسولُ الله؟٢٣٤٢حدَّثنا محمودُ بنُ خالدِ وعبدُ الله بنُ عبدِ…

باب في توكيد السحور

٢٣٤٣إنَّ فَصْلَ ما بَيْنَ صِيامِنَا وصِيامِ أهْلِ…

باب من سمى السحور الغداء

٢٣٤٤هلُمَّ إلى الغَدَاء المُبَاركِ٢٣٤٥نِعْمَ سَحُورُ المُؤمِنِ التَّمْرُ

باب وقت السحور

٢٣٤٦لا يمنعنَ من سَحورِكُم أذانُ بِلالٍ، ولا بياض…٢٣٤٧لا يَمْنَعَن أحَدَكُم أذانُ بلالٍ من سَحورهِ، فإنه…٢٣٤٨كُلُوا واشْرَبُوا، ولا يَهيدَنكُمُ السَّاطِعُ…٢٣٤٩إن وِسَادَكَ إذاً لعَرِيضٌ طويلٌ، إنما هو الليل…

باب [في] الرجل يسمع النداء والإناء على يده

٢٣٥٠إذا سَمِعَ أحَدُكُم النداء والإناءُ على يَدِهِ،…

باب وقت فطر الصائم

٢٣٥١إذا جاء الليلُ مِن ها هنا، وذهبَ النهارُ مِن ها هنا٢٣٥٢يا بلالُ، انزِل فاجْدَحْ لنا

باب ما يستحب من تعجيل الفطر

٢٣٥٣لا يَزَالُ الذينُ ظاهِراً ما عجَّل الناسُ الفِطْر،…٢٣٥٤حدَّثنا مُسَدَدٌ، حدَّثنا أبو معاويةَ، عن الأعمشِ،…

باب ما يفطر عليه

٢٣٥٥إذا كان أحدُكُم صائماً فليفطِرْ على التمر، فإن لم…٢٣٥٦حدَّثنا أحمدُ بنُ حنبلٍ، حدَّثنا عبدُ الرزاق،…

باب القول عند الإفطار

٢٣٥٧ذهب الظمأ وابتلَّت العُرُوقُ وثبتَ الأجر إن شاء…٢٣٥٨اللَهُمَّ لك صُمْتُ، وعلى رِزْقِكَ أفْطَرْتُ

باب الفطر قبل غروب الشمس

٢٣٥٩حدَّثنا هارونُ بنُ عبدِ الله ومحمدُ بنُ العلاء…

باب في الوصال

٢٣٦٠إني لَسْتُ كَهَيْئَتِكُم، إني أُطْعَمُ وأُسْقَى٢٣٦١لا تُواصِلُوا، فأيُّكم أراد أن يُوَاصِلَ فليواصِلْ…

باب الغيبة للصائم

٢٣٦٢مَنْ لم يَدَعْ قولَ الزورِ والعملَ به، فليسَ لله…٢٣٦٣الصِّيامُ جُنّة، إذا كان أحدُكُم صائماً، فلا يرفُث…

باب السواك للصائم

٢٣٦٤حدَّثنا محمدُ بنُ الصبَّاح، حدَّثنا شريكٌ (ح)…

باب الصائم يصب عليه الماء من العطش ويبالغ في الاستنشاق

٢٣٦٥تَقَوَّوْا لِعدُوكُم٢٣٦٦بالِغْ في الاسْتِنْشَاق، إلا أن تكون صَائِماً

باب الصائم يحتجم

٢٣٦٧أفْطَرَ الحَاجِمُ والمَحْجُومُ٢٣٦٨حدَّثنا أحمد بن حنبل، حدَّثنا حسنُ بنُ موسى،…٢٣٦٩أفْطَرَ الحاجِمُ والمحجومُ٢٣٧٠أفْطَرَ الحَاجِم والمَحْجُومُ٢٣٧١أفْطَرَ الحَاجِمُ والمَحْجُوُم

باب في الرخصة في ذلك

٢٣٧٢حدَّثنا أبو مَعْمَرٍ عبدُ الله بنُ عَمرو، حدَّثنا…٢٣٧٣حدَّثنا حفصُ بنُ عُمَرَ، حدُّثنا شعبةُ، عن يزيدَ…٢٣٧٤إني أُواصِلُ إلى السَّحَرَ، وربي يُطْعِمُنِي…٢٣٧٥حدَّثنا عبدُ الله بنُ مسلمةَ، حدَّثنا سليمان -يعني…

باب في الصائم يحتلم نهارا في رمضان

٢٣٧٦لا يُفْطِرُ مَن قاء، ولا مَن احْتَلَمَ، ولا مَن…

باب في الكحل عند النوم

٢٣٧٧لِيتَّقِهِ الصَّائِمُ٢٣٧٨حدَّثنا وهبُ بنُ بقية، أخبرنا أبو مُعاويَةَ، عن…٢٣٧٩حدَّثنا محمدُ بنُ عبدِ الله المخرِّمي ويحيى بنُ…

باب الصائم يستقيء عامدا

٢٣٨٠مَنْ ذَرَعَهُ قيءٌ وهو صَائِمٌ، فليسَ عليهِ…٢٣٨١حدَّثنا أبو معمر عبدُ الله بنُ عمرو، حدَّثنا عبدُ…

باب القبلة للصائم

٢٣٨٢حدَّثنا مُسَدَّد، حدَّثنا أبو معاويةَ، عن الأعمشِ،…٢٣٨٣حدَّثنا أبو توبةَ الربيعُ بنُ نافعٍ، حدَّثنا أبو…٢٣٨٤حدَّثنا محمدُ بنُ كثير، أخبرنا سفيانُ، عن سعْدِ…٢٣٨٥أرأيتَ لو مَضْمَضْتَ مِنَ الماءِ وأنتَ صائِمٌ؟

باب الصائم يبلع الريق

٢٣٨٦حدَّثنا محمدُ بنُ عيسى، حدَّثنا محمدُ بنُ دينار،…

باب كراهيته للشاب

٢٣٨٧حدَّثنا نصرُ بنُ علي، حدَّثنا أبو أحمد -يعني…

باب فيمن أصبح جنبا في شهر رمضان

٢٣٨٨يُصبح جنباً في رمضان٢٣٨٩وأنا أُصْبِحُ جنباً وأنا أُريدُ الصِّيامَ،…

باب كفارة من أتى أهله في رمضان

٢٣٩٠ما شأنُكَ؟٢٣٩١واستغفِرِ الله٢٣٩٢اجْلِسْ٢٣٩٣كُلْهُ أنتَ وأهلُ بَيْتِكَ، وصُمْ يوماً واستغفرِ…٢٣٩٤ما شأنُه؟٢٣٩٥حدَّثنا محمدُ بنُ عوفِ، حدَّثنا سَعِيدُ بن أبي…

باب التغليظ في من أفطر عمدا

٢٣٩٦مَنْ أفطرَ يوماً مِن رمضانَ في غيرِ رُخصةٍ رخَّصها…٢٣٩٧حدَّثنا أحمدُ بنُ حنبل، حدَّثنا يحيى بنُ سعيدٍ، عن…

باب من أكل ناسيا

٢٣٩٨اللهُ أطعَمَكَ وسَقَاكَ

باب تأخير قضاء رمضان

٢٣٩٩حدَّثنا عبدُ الله بنُ مسلمةَ القعنبيُّ، عن مالك،…

باب فيمن مات وعليه صيام

٢٤٠٠من مَاتَ وعليه صيام صامَ عنهُ ولِيُّه٢٤٠١حدَّثنا محمدُ بنُ كثيرٍ، أخبرنا سفيانُ، عن أبي…

باب الصوم في السفر

٢٤٠٢صمْ إن شئتَ، وأفْطِرْ إن شِئتَ٢٤٠٣أيَّ ذلك شئتَ يا حمزةُ٢٤٠٤حدَّثنا مُسدَّدٌ، حدَّثنا أبو عوانة، عن منصور، عن…٢٤٠٥حدَّثنا أحمدُ بنُ يونس، حدَّثنا زائدةُ، عن حُميدِ…٢٤٠٦إنكم قد دنوتُم مِنْ عدوِّكُم، والفطرُ أقوى لكم

باب اختيار الفطر

٢٤٠٧ليس من البرِّ الصيامُ في السَّفَرِ٢٤٠٨اجْلِسْ فأصِبْ مِن طعامِنا هذا

باب فيمن اختار الصيام

٢٤٠٩حدَّثنا مُؤمَّلُ بنُ الفضل، حدَّثنا الوليدُ،…٢٤١٠من كانت لهُ حمولةٌ تأوي إلى شِبَعٍ، فليَصُمْ…٢٤١١مَنْ أدركَهُ رمضانُ في السفر

باب متى يفطر المسافر إذا خرج؟

٢٤١٢حدَّثنا عُبيدُ الله بنُ عُمر، حدثني عبدُ الله بنُ…

باب مسيرة ما يفطر فيه

٢٤١٣حدَّثنا عيسى بنُ حمّاد، أخبرنا الليثُ -يعني ابنَ…٢٤١٤حدَّثنا مُسَدَّدٌ، حدَّثنا المعتَمِرُ، عن عُبيدِ…

باب من يقول: صمت رمضان كله

٢٤١٥لا يقولنَّ أحدُكُم: إني صُمْتُ رمضانَ كُلَّه…

باب في صوم العيدين

٢٤١٦حدَّثنا قتيبةُ بنُ سعيد وزهيرُ بنُ حرب -وهذا…٢٤١٧حدَّثنا موسى بنُ إسماعيلَ، حدَّثنا وُهَيبٌ،…

باب صيام أيام التشريق

٢٤١٨حدَّثنا عبدُ الله بنُ مسلمةَ القعنبيُّ، عن مالكٍ،…٢٤١٩يومُ عرفة ويومُ النحرِ وأيامُ التشريقِ عيدُنا أهلَ…

باب النهي أن يخص يوم الجمعة بصوم

٢٤٢٠لا يَصُمْ أحدُكُم يومَ الجمعةِ، إلا أن يَصُومَ…

باب النهي أن يخص يوم السبت بصوم

٢٤٢١حدَّثنا حُميدُ بنُ مَسعدةَ، حدَّثنا سفيانُ بنُ…

باب الرخصة في ذلك

٢٤٢٢صُمْتِ أمْسِ؟٢٤٢٣حدَّثنا عبدُ الملك بنُ شعيب، حدَّثنا ابنُ وهبٍ،…٢٤٢٤حدَّثنا محمدُ بنُ الصَّباح بنِ سُفيانَ، حدَّثنا…

باب في صوم الدهر

٢٤٢٥لا صامَ ولا أفْطَرَ٢٤٢٦فيهِ وُلِدْتُ، وفيه أُنزِلَ عليَّ القرآنُ٢٤٢٧ألم أُحَدَّثْ أنك تقولُ: لأقومَنَّ الليلَ،…

باب في صوم أشهر الحرم

٢٤٢٨ومن أنت؟

باب في صوم المحرم

٢٤٢٩أفضلُ الصيامِ بَعْدَ شهرِ رمضانَ شهرُ اللهِ…٢٤٣٠حدَّثنا إبراهيمُ بنُ موسى، حدَّثنا عيسى، حدَّثنا…

باب في صوم شعبان

٢٤٣١حدَّثنا أحمدُ بنُ حنبلٍ، حدَّثنا عبدُ الرحمن بنُ…

باب في صوم شوال

٢٤٣٢إن لأهلِكَ عليكَ حقّاً، صُمْ رمضانَ والذي يليه،…

باب في صوم ستة أيام من شوال

٢٤٣٣مَنْ صامَ رمضانَ، ثم أَتبَعَهُ بِسِتٍّ مِن…

باب، كيف كان يصوم النبي -صلى الله عليه وسلم-؟

٢٤٣٤حدَّثنا عبدُ الله بنُ مسلمةَ، عن مالكٍ، عن أبي…٢٤٣٥حدَّثنا موسى بنُ إسماعيلَ، حدَّثنا حمادٌ، عن محمدِ…

باب في صوم الاثنين والخميس

٢٤٣٦إن أعمالَ الناسِ تُعرَضُ يومَ الاثنين ويومَ الخميس

باب في صوم العشر

٢٤٣٧حدَّثنا مُسَدَّدٌ، حدَّثنا أبو عَوانة، عن الحرِّ…٢٤٣٨ما مِنْ أيَّامٍ العمَلُ الصَّالِحُ فيها أحبُّ إلى…

باب في فطره

٢٤٣٩حدَّثنا مُسَدَّدٌ، حدَّثنا أبو عَوَانَةَ، عن…

باب في صوم يوم عرفة بعرفة

٢٤٤٠حدَّثنا سليمانُ بنُ حرب، حدَّثنا حَوشَبُ بنُ…٢٤٤١حدَّثنا القعنبيُّ، عن مالكٍ، عن أبي النَّضْرِ، عن…

باب في صوم يوم عاشوراء

٢٤٤٢حدَّثنا عبدُ الله بن مسلمةَ، عن مالكٍ، عن هشامِ…٢٤٤٣هذا يومٌ من أيام الله، فمن شاء صامه، ومن شاء تركه٢٤٤٤نحنُ أولى بموسى مِنْكُم

باب ما روي أن عاشوراء اليوم التاسع

٢٤٤٥فإذا كان العامُ المقبلُ صُمْنا يومَ التَّاسِعِ٢٤٤٦حدَّثنا مُسَدَّد، حدَّثنا يحيي -يعني ابن سعيدٍ، عن…

باب في فضل صومه

٢٤٤٧صُمتم يومكم هذا؟

باب في صوم يوم وفطر يوم

٢٤٤٨أحبُّ الصِّيامِ إلى الله صِيَامُ داودَ، وأحبُّ…

باب في صوم الثلاث من كل شهر

٢٤٤٩هُنَ كهيئةِ الدَّهرِ٢٤٥٠حدَّثنا أبو كامِل، حدَّثنا أبو داودَ، حدَّثنا…

باب من قال الاثنين والخميس

٢٤٥١حدَّثنا موسى بنُ إسماعيلَ، حدَّثنا حماد، عن عاصم…٢٤٥٢حدَّثنا زهيرُ بنُ حربٍ، حدَّثنا محمدُ بنُ فضيلٍ،…

باب من قال: لا يبالي من أي الشهر

٢٤٥٣حدَّثنا مُسَدَّدٌ، حدَّثنا عبدُ الوارثِ، عن يزيدَ…

باب النية في الصيام

٢٤٥٤من لم يُجْمّعِ الصِّيامَ قبلَ الفجرِ فلا صِيامَ له

باب في الرخصة في ذلك

٢٤٥٥هل عِندَكُم طعامٌ؟٢٤٥٦أكُنْتِ تقضِينَ شيئاً؟

باب من رأى عليه القضاء

٢٤٥٧لا عليكما، صُوما مكانَه يوماَ آخر

باب المرأة تصوم بغير إذن زوجها

٢٤٥٨لا تَصوُمُ المرأةُ وبعْلُهَا شاهِدٌ إلا بإذنه،…٢٤٥٩لو كانت سورةً واحِدةً لكَفَتِ الناس

باب في الصائم يدعى إلى وليمة

٢٤٦٠إذا دُعي أحدُكُم، فليجِبْ، فإن كان مُفْطِراً…

باب ما يقول الصائم إذا دعي إلى الطعام

٢٤٦١إذا دُعِيَ أحدُكُم إلى طعامٍ وهو صَائِمٌ،…

باب الاعتكاف

٢٤٦٢حدَّثنا قُتيبةُ بنُ سعيدٍ، حدَّثنا الليثُ، عن…٢٤٦٣حدَّثنا موسى بنُ إسماعيلَ، حدَّثنا حماد، أخبرنا…٢٤٦٤ما هذه آلبِرَّ تُرِدْنَ؟

باب أين يكون الاعتكاف؟

٢٤٦٥حدَّثنا سليمانُ بنُ داود المَهريُّ، أخبرنا ابنُ…٢٤٦٦حدَّثنا هنّاد، عن أبي بكير، عن أبي حَصين، عن أبي…

باب المعتكف يدخل البيت لحاجته

٢٤٦٧حدَّثنا عبدُ الله بنُ مسلمةَ، عن مالكٍ، عن ابنِ…٢٤٦٨حدَّثنا قتيبةُ بنُ سعيدٍ وعبدُ الله بنُ مسلمة،…٢٤٦٩حدَّثنا سليمانُ بنُ حربٍ ومُسَدد، قالا: حدَّثنا…٢٤٧٠على رِسْلِكُما، إنَّها صَفيةُ بنتُ حُيَيٌّ٢٤٧١حدَّثنا محمدُ بنُ يحيى بن فِارسٍ، حدَّثنا أبو…

باب المعتكف يعود المريض

٢٤٧٢حدَّثنا عبدُ اللهِ بنُ محمدٍ النُّفيليُّ ومحمدُ…٢٤٧٣حدَّثنا وهبُ بنُ بقيةَ، أخبرَنا خالِدٌ، عن عبدِ…٢٤٧٤اعْتكِفْ وصُمْ٢٤٧٥حدَّثنا عبدُ الله بنُ عمر بنِ محمد بنِ أبانَ بنِ…

باب المستحاضة تعتكف

٢٤٧٦حدَّثنا محمدُ بنُ عيسى وقتيبةُ بنُ سعيد، قالا:…
