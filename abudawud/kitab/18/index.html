٨٤ حديثًا

باب التغليظ في اليمين الفاجرة

٣٢٤٢من حَلَفَ على يمين مَصبورَةٍ كاذباً، فليتبَؤَأ…

باب فيمن حلف يمينا ليقتطع بها مالا لأحد

٣٢٤٣مَنْ حَلَفَ على يمينِ هو فيها فاجرٌ ليقتطعَ بها…٣٢٤٤هل لكَ بينةٌ؟٣٢٤٥ألك بينةٌ؟

باب في تعظيم اليمين على منبر رسول الله - صلى الله عليه وسلم

٣٢٤٦لا يَحلفُ أحدٌ عند منبري هذا على يمين آثمةٍ ولو…

باب الحلف بالأنداد

٣٢٤٧مَن حلَفَ فقال في حَلفِه: واللاَّتِ، فليقل: لا إله…٣٢٤٨لا تحلِفُوا - بآبائكم، ولا بأمهاتِكم، ولا…

باب في كراهية الحلف بالآباء

٣٢٤٩إن الله عزّ وجلّ ينهاكُم أن تَحلِفُوا بآبائكم، فمن…٣٢٥٠بآبائكم٣٢٥١مَن حلَفَ بغيرِ الله فقد أشركَ٣٢٥٢أفلحَ وأبيه إن صَدَقَ، دخل الجنَّةَ وأبيهِ إن صَدَقَ

باب كراهية الحلف بالأمانة

٣٢٥٣مَنْ حلفَ بالأمانةِ فليس منا

باب لغو اليمين

٣٢٥٤هو كلامُ الرجلِ في بيتِه، كلاَّ واللهِ، وبَلَى والله

باب المعاريض في اليمين

٣٢٥٥يمينُكَ على ما يُصدِّقُك عليها صاحبُك٣٢٥٦صَدَقْتَ، المسلمُ أخو المسلم

باب ما جاء في الحلف بالبراءة وبملة غير الإسلام

٣٢٥٧مَن حلف بملةِ غيرِ الإسلام كاذباً فهو كما قال، ومن…٣٢٥٨من حلفَ فقال: إني بَرِيءٌ من الإسلام: فإن كان…

باب من حلف أن لا يتأدم

٣٢٥٩هذه إدامُ هذه٣٢٦٠حدَّثنا هارونُ بن عَبد الله، حدَّثنا عُمر بن…

باب الاستثناء في اليمين

٣٢٦١مَنْ حَلَفَ على يمين، فقال: إن شاء الله فَقَدِ…٣٢٦٢مَنْ حلفَ فاستثنى فإن شاء رجعَ، وإن شاءَ ترك غيرَ…

باب ما جاء في يمين النبي -صلى الله عليه وسلم- ما كانت

٣٢٦٣لا، ومُقَلِّبِ القلوب٣٢٦٤والذي نفسُ أبي القاسم بيده٣٢٦٥لا، وأستغفرُ اللهَ٣٢٦٦لَعَمْرُ إلهِكَ

باب في القسم هل يكون يمينا

٣٢٦٧لا تُقسِمْ٣٢٦٨أصَبتَ بعضاً وأخطأتَ بعضاً٣٢٦٩حدَّثنا محمدُ بن يحيى بن فارس، حدَّثنا محمدُ بن…

باب فيمن حلف على طعام لا يأكله

٣٢٧٠بل أنتَ أبَرُّهُم وأصدقُهم٣٢٧١حدَّثنا ابنُ المثنَّى، حدَّثنا سالمُ بن نُوحٍ…

باب اليمين في قطيعة الرحم

٣٢٧٢لا يَمينَ عليك، ولا نذرَ في معصية الربِّ، ولا في…٣٢٧٣لا نَذرَ إلا فيما يُبتَغَى به وَجْهُ الله عز وجل،…٣٢٧٤لا نَذْرَ ولا يمينَ فيما لا يملكُ ابنُ آدم، ولا في…

باب فيمن يحلف كاذبا متعمدا

٣٢٧٥بلى قَدْ فَعَلْتَ، ولكن غُفر لك بإخلاصِ قولِ لا…

باب الرجل يكفر قبل أن يحنث

٣٢٧٦إني والله إن شاء الله لا أحْلِف على يمينِ فأرى…٣٢٧٧يا عَبْدَ الرحمن ابن سمُرة، إذا حلفتَ على يمينٍ…٣٢٧٨حدَّثنا يحيى بن خَلَفٍ، حدَّثنا عبدُ الأعلى،…

باب كم الصاع في الكفارة

٣٢٧٩حدَّثنا أحمدُ بن صالح، قال: قرأتُ على أنسِ بن…٣٢٨٠حدَّثنا محمدُ بن محمدِ بن خلاَّد أبو عمر، قال: كان…٣٢٨١حدَّثنا محمدُ بن محمد بن خلاّد، حدَّثنا مُسدَّدٌ…

باب الرقبة المؤمنة

٣٢٨٢ائتِني بها٣٢٨٣ادْعُ بها٣٢٨٤أينَ اللهُ؟

باب الاستثناء في اليمين بعد السكوت

٣٢٨٥والله لأغزُوَنَّ قُريشاً، والله لأغزُوَنَّ…٣٢٨٦والله لأغزوَنَّ قريشاً

باب النهي عن النذر

٣٢٨٧لا يَرُدُّ شيئاً، وإنما يُسْتَخرجُ به من البخيل٣٢٨٨لا يأتي ابنَ آدم النذر القدرَ بشيءٍ لم أكُن…

باب ما جاء في النذر في المعصية

٣٢٨٩من نذَرَ أن يطيعَ اللهَ فليُطِعْه، ومن نذَرَ أن…٣٢٩٠لا نذرَ في معصية، وكفَّارَتُهُ كَفارةُ يمين٣٢٩١حدَّثنا ابنُ السَّرح، حدَّثنا ابنُ وهبِ، عن يونسَ،…٣٢٩٢لا نَذْرَ في معصيةٍ، وكفارتُهُ كفارةُ يمين٣٢٩٣مُرُوها فلتَخْتمِرْ ولتَركَبْ، ولتَصُمْ ثلاثةَ أيام٣٢٩٤حدَّثنا مَخلَدُ بن خالد، حدَّثنا عبدُ الرزاق،…٣٢٩٥إن الله لا يصنَع بشقاء أُختِك شيئاً، فلتَحُجَّ…٣٢٩٦حدَّثنا محمدُ بن المُثنَّى، حدَّثنا أبو الوليد،…٣٢٩٧إن الله لَغَنِيٌّ عن نذْرِها، مُرْها فلتَركَبْ٣٢٩٨مُرْ أختَك فلتَركَبْ٣٢٩٩لتَمْشِ ولْتَركَبْ٣٣٠٠مُرُوهُ فَلْيتكلمْ وليستظلَّ وليقعدْ، وليتمَّ صومه٣٣٠١حدَّثنا مُسَدَّدٌ، حدَّثنا يحيى، عن حميدٍ الطويل،…٣٣٠٢حدَّثنا يحيى بنُ مَعين، حدَّثنا حجاجٌ، عن ابن…٣٣٠٣إن الله لَغَنيٌّ عن مَشْي أختِك، فلْتَركَبْ،…٣٣٠٤إن الله لا يصنَع بمشْيِ أختِكَ إلى البيت شيئاً

باب من نذر أن يصلي في بيت المقدس

٣٣٠٥صَلِّ ها هنا٣٣٠٦والذي بعث محمداً بالحق لو صليتَ ها هنا لاجزأَ عنك…

باب في قضاء النذر عن الميت

٣٣٠٧اقْضِهِ عنها٣٣٠٨حدَّثنا عَمرو بن عَونٍ، أخبرنا هُشيمٌ، عن أبي…٣٣٠٩قد وجبَ أجرُك ورجعتْ إليك في الميراث

باب ما جاء فيمن مات وعليه صيام صام عنه وليه

٣٣١٠لو كان على أمكِ دَينٌ، أكنتِ قاضيتَه؟٣٣١١مَنْ مات وعليه صيام صامَ عنه وليُّه

باب ما يؤمر بوفائه من النذر

٣٣١٢أوفِي بنذْرِكِ٣٣١٣هل كان فيها وثنٌ من أوثانِ الجاهليةُ يُعبَدُ؟٣٣١٤هل بها من الأوثان شيءٌ؟٣٣١٥هل بها وثنٌ أو عيدٌ من أعياد الجاهليَّة؟

باب في النذر فيما لا يملك

٣٣١٦نأخُذُكَ بجَريرَةِ حُلفائكَ ثَقيفِ

باب فيمن نذر أن يتصدق بماله

٣٣١٧أمسِكْ عليك بعضَ مالِكَ، فهو خيرٌ لك٣٣١٨خيرٌ لك٣٣١٩يُجزئ عنك الثلث٣٣٢٠حدَّثنا محمدُ بن المتوكِّل العسقلاني، حدَّثنا عبدُ…٣٣٢١لا

باب النذر لا يسمى

٣٣٢٢من نذر نذْراً لم يُسمِّه فكفارتُه كفارةُ يمين، ومن…٣٣٢٣حدَّثنا هارونُ بن عبَّادِ الأزديُّ، حدَّثنا أبو…٣٣٢٤حدَّثنا محمد بن عَوفٍ، أن سعيدَ بن الحكَم حدثهم،…٣٣٢٥أوْفِ بنَذْرِكَ
البخاري مسلم أبو داود الترمذي النسائي ابن ماجه الموطأ الدارمي الأدب المفرد رياض الصالحين
