٧٦ حديثًا

٢٨٨٠لا إله إلا الله. ويل للعرب من شر قد اقترب. فتح…٢٨٨١فتح اليوم من ردم يأجوج ومأجوج مثل هذه٢٨٨٢يعوذ عائذ بالبيت فيبعث إليه بعث. فإذا كانوا ببيداء…٢٨٨٣ليؤمن هذا البيت جيش يغزونه. حتى إذا كانوا ببيداء…٢٨٨٤العجب إن ناسا من أمتي يؤمون بالبيت برجل من قريش.…٢٨٨٥هل ترون ما أرى؟ إني لأرى مواقع الفتن خلال بيوتكم،…٢٨٨٦ستكون فتن، القاعد فيها خير من القائم، والقائم فيها…٢٨٨٧إنها ستكون فتن. ألا ثم تكون فتنة القاعد فيها خير…٢٨٨٨إذا تواجه المسلمان بسيفيهما، فالقاتل والمقتول في…٢٨٨٩أن الله زوى لي الأرض. فرأيت مشارقها ومغاربها. وإن…٢٨٩٠سألت ربي ثلاثا. فأعطاني ثنتين ومنعني واحدة. سألت…٢٨٩١منهن ثلاث لا يكدن يذرن شيئا. ومنهن فتن كرياح…٢٨٩٢وحدثني يعقوب بن إبراهيم الدورقي وحجاج بن الشاعر.…٢٨٩٣وحَدَّثَنَا مُحَمَّدُ بْنُ الْمُثَنَّى وَمُحَمَّدُ…٢٨٩٤لا تقوم الساعة حتى يحسر الفرات عن جبل من ذهب.…٢٨٩٥يوشك الفرات أن يحسر عن جبل من ذهب. فإذا سمع به…٢٨٩٦منعت العراق درهمها وقفيزها. ومنعت الشام مديها…٢٨٩٧لا تقوم الساعة حتى ينزل الروم بالأعماق، أم بدابق.…٢٨٩٨تقوم الساعة والروم أكثر الناس٢٨٩٩إني لأعرف أسمائهم، وأسماء آبائهم، وألوان خيولهم.…٢٩٠٠تغزون جزيرة العرب، فيفتحها الله. ثم فارس، فيفتحها…٢٩٠١ما تذاكرون؟٢٩٠٢لا تقوم الساعة حتى تخرج نار من أرض الحجاز، تضئ…٢٩٠٣تبلغ المساكن إهاب أو يهاب٢٩٠٤ليست السنة بأن لا تمطروا. ولكن السنة أن تمطروا…٢٩٠٥ألا إن الفتنة ههنا. ألا إن الفتنة ههنا، من حيث…٢٩٠٦لَا تَقُومُ الساعة حتى تضطرب أليات نساء دوس. حول…٢٩٠٧لا يذهب الليل والنهار حتى تعبد اللات والعزى٢٩٠٨وَالَّذِي نَفْسِي بِيَدِهِ! ليأتين على الناس زمان…٢٩٠٩يخرب الكعبة ذو السويقتين من الحبشة٢٩١٠لا تقوم الساعة حتى يخرج رجل من قحطان يسوق الناس…٢٩١١لا تذهب الأيام والليالي، حتى يملك رجل يقال له…٢٩١٢لا تقوم الساعة حتى تقاتلوا قوما كأن وجوههم المجان…٢٩١٣يكون في آخر أمتي خليفة يحثي المال حثيا. لا يعده عددا٢٩١٤من خلفائكم خليفة يحثو المال حثيا. لا يعده عددا٢٩١٥بؤس ابن سمية. تقتلك فئة باغية٢٩١٦تقتلك الفئة الباغية٢٩١٧يهلك أمتي هذا الحي من قريش٢٩١٨قد مات كسرى فلا كسرى بعده. وإذا هلك قيصر فلا قيصر…٢٩١٩إذا هلك كسرى فلا كسرى بعده٢٩٢٠سمعتم بمدينة جانب منها في البر وجانب منها في البحر؟٢٩٢١لتقاتلن اليهود. فلنقتلنهم حتى يقول الحجر: يا مسلم!…٢٩٢٢لا تقوم الساعة حتى يقاتل المسلمون اليهود. فيقتلهم…٢٩٢٣إن بين يدي الساعة كذابين٢٩٢٤تربت يداك. أتشهد أني رسول الله؟٢٩٢٥أتشهد أني رسول الله؟٢٩٢٦حَدَّثَنَا يَحْيَى بْنُ حبيب ومحمد بن عبد الأعلى…٢٩٢٧إنه لا يولد له٢٩٢٨ما تربة الجنة؟٢٩٢٩حَدَّثَنَا عُبَيْدُ اللَّهِ بْنُ مُعَاذٍ…٢٩٣٠أتشهد أني رسول الله؟٢٩٣١لو تركته بين٢٩٣٢إنما يخرج من غضبة يغضبها؟٢٩٣٣ما من نبي إلا وقد أنذر أمته الأعور الكذاب. ألا إنه…٢٩٣٤الدجال أعمر العين اليسرى. جفال الشعر. معه جنة…٢٩٣٥لأنا بما مع الدجال أعلم منه. إن معه نهرا من ماء…٢٩٣٦ألا أخبركم عن الدجال حديثا ما حدثه نبي قومه؟ إنه…٢٩٣٧ما شأنكم؟٢٩٣٨يأتي، وهو محرم عليه أن يدخل نقاب المدينة. فينتهي…٢٩٣٩وما ينصبك منه؟ إنه لا يضرك٢٩٤٠يخرج الدجال في أمتي فيمكث أربعين (لا أدري: أربعين…٢٩٤١إن أول الآيات خروجا، طلوع الشمس من مغربها، وخروج…

باب قصة الجساسة

٢٩٤٢من أحبني فليحب أسامة٢٩٤٣ليس من بلد إلا سيطؤه الدجال. إلا مكة والمدينة.…٢٩٤٤يتبع الدجال، من يهود أصبهان، سبعون ألفا. عليهم…٢٩٤٥ليفرن الناس من الدجال في الجبال٢٩٤٦ما بين خلق آدم إلى قيام الساعة خلق أكبر من الدجال٢٩٤٧بَادِرُوا بالأعمال ستا: طلوع الشمس من مغربها، أو…٢٩٤٨العبادة في الهرج، كهجرة إلي٢٩٤٩لا تقوم الساعة إلا على شرار الناس٢٩٥٠بعثت أنا والساعة هكذا٢٩٥١بعثت أنا والساعة كهاتين٢٩٥٢إن يعش هذا، لم يدركه الهرم، قامت عليكم ساعتكم٢٩٥٣إن يعش هذا الغلام، فعسى أن لا يدركه الهرم، حتى…٢٩٥٤تقوم الساعة والرجل يحلب اللقحة، فما يصل الإناء إلى…٢٩٥٥ما بين النفختين أربعون
