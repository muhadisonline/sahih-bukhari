كتاب الجنة، وصفة نعيمها وأهلها

٥٨ حديثًا

٢٨٢٢حفت الجنة بالمكاره. وحفت النار بالشهوات٢٨٢٣وحدثني زهير بن حرب. حَدَّثَنَا شَبَابَةُ.…٢٨٢٤قال الله عز وجل: أعددت لعبادي الصالحين ما لا عين…٢٨٢٥فيها ما لا عين رأت، ولا أذن سمعت، ولا خطر على قلب…٢٨٢٦إِنَّ فِي الجنة لشجرة يسير الراكب في ظلها مائة سنة٢٨٢٧إن في الجنة لشجرة يسير الراكب في ظلها مائة عام لا…٢٨٢٨إن في الجنة شجرة يسير الراكب الجواد المضمر السريع،…٢٨٢٩إِنَّ الله يقول لأهل الجنة: يا أهل الجنة! فيقولون:…٢٨٣٠إن أهل الجنة ليتراءون الغرفة في الجنة كما تراءون…٢٨٣١كما تراءون الكوكب الدري في الأفق الشرقي أو الغربي٢٨٣٢من أشد أمتي لي حبا، ناس يكونون بعدي، يود أحدهم لو…٢٨٣٣إِنَّ فِي الجنة لسوقا. يأتونها كل جمعة. فتهب ريح…٢٨٣٤إِنَّ أول زمرة تدخل الجنة على صورة القمر ليلة…٢٨٣٥إن أهل الجنة يأكلون فيها ويشربون. ولا يتفلون ولا…٢٨٣٦من يدخل الجنة ينعم لا يبأس. لا تبلى ثيابه ولا يفنى…٢٨٣٧ينادي مناد: إن لكم أن تصحوا فلا تسقموا أبدا. وإن…٢٨٣٨إن للمؤمن في الجنة لخيمة من لؤلؤة واحدة مجوفة.…

باب ما في الدنيا من أنهار الجنة

٢٨٣٩سيحان وجيحان، والفرات والنيل، كل من أنهار الجنة٢٨٤٠يَدْخُلُ الجنة أقوام أفئدتهم مثل أفئدة الطير٢٨٤١خلق الله عز وجل آدم على صورته. طوله ستون ذراعا.…٢٨٤٢يؤتى بجهنم يومئذ لها سبعون ألف زمام. مع كل زمام…٢٨٤٣ناركم هذه، التي يوقد ابن آدم، جزء من سبعين جزءا من…٢٨٤٤تدرون ما هذا؟٢٨٤٥إِنَّ منهم من تأخذه النار إلى كعبيه. ومنهم من…٢٨٤٦احتجت النار والجنة. فقالت هذه: يدخلني الجبارون…٢٨٤٧احتجت الجنة والنار٢٨٤٨لا تزال جهنم تقول: هل من مزيد، حتى يضع فيها رب…٢٨٤٩يجاء بالموت يوم القيامة كأنه كبش أملح (وزاد أبو…٢٨٥٠يُدْخِلُ اللَّهُ أَهْلَ الْجَنَّةِ الْجَنَّةَ.…٢٨٥١ضرس الكافر، أو ناب الكافر، مثل أحد. وغلظ جلده…٢٨٥٢ما بين منكبي الكافر في النار، مسيرة ثلاثة أيام،…٢٨٥٣ألا أخبركم بأهل الجنة؟٢٨٥٤رب أشعث مدفوع بالأبواب، لو أقسم على الله لأبره٢٨٥٥إذ انبعث أشقاها: انبعث بها رجل عزيز عارم منيع في…٢٨٥٦رأيت عمرو بن لحي بن قمعة بن خندف، أبا بني كعب…٢٨٥٧يوشك، إن طالت بك مدة، أن ترى قوما في أيديهم مثل…

باب فناء الدنيا، وبيان الحشر يوم القيامة

٢٨٥٨والله! ما الدنيا في الآخرة إلا مثل ما يجعل أحدكم…٢٨٥٩يحشر الناس يوم القيامة حفاة عراة غرلا٢٨٦٠إنكم ملاقو الله مشاة حفاة عراة غرلا٢٨٦١يحشر الناس ثلاث طرائق راغبين راهبين. واثنان على…٢٨٦٢يقوم أحدهم في رشحه إلى أنصاف أذنيه٢٨٦٣إن العرق، يوم القيامة، ليذهب في الأرض سبعين باعا.…٢٨٦٤تدني الشمس، يوم القيامة، من الخلق، حتى تكون منهم…٢٨٦٥ألا إن ربي أمرني أن أعلمكم ما جهلتم مما علمني،…٢٨٦٦إن أحدكم إذا مات عرض عليه مقعده بالغداة والعشي. إن…٢٨٦٧من يعرف أصحاب هذه الأقبر؟٢٨٦٨لولا أن لا تدافنوا لدعوت الله أن يسمعكم من عذاب…٢٨٦٩يهود تعذب في قبورها٢٨٧٠إن العبد إذا وضع في قبره، وتولى عنه أصحابه، إنه…٢٨٧١{يثبت الله الذين آمنوا بالقول الثابت} [١٤ /إبراهيم…٢٨٧٢إذا خرجت روح المؤمن تلقاها ملكان يصعدانها٢٨٧٣هذا مصرع فلان غدا، إن شاء الله٢٨٧٤يا أبا جهل بن هشام! يا أمية بن خلف! يا عتبة بن…٢٨٧٥حَدَّثَنِي يُوسُفُ بْنُ حَمَّادٍ الْمَعْنِيُّ.…٢٨٧٦من حوسب، يوم القيامة، عذب٢٨٧٧لا يموتن أحدكم إلا وهو يحسن بالله الظن٢٨٧٨يبعث كل عبد على ما مات عليه٢٨٧٩إذا أراد الله بقوم عذابا، أصاب العذاب من كان فيهم،…
