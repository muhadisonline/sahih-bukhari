٩٥ حديثًا

٢٥٤٨أمك٢٥٤٩أحي والداك؟٢٥٥٠حَدَّثَنَا شَيْبَانُ بْنُ فَرُّوخَ. حَدَّثَنَا…٢٥٥١رغم أنف، ثم رغم أنف، ثم رغم أنف٢٥٥٢إن أبر البر صلة الولد أهل ود أبيه٢٥٥٣البر حسن الخلق. والإثم ما حاك في صدرك، وكرهت أن…٢٥٥٤أن الله خلق الخلق. حتى إذا فرغ منهم قامت الرحم…٢٥٥٥الرحم معلقة بالعرش تقول: من وصلني وصله الله. ومن…٢٥٥٦لا يدخل الجنة قاطع٢٥٥٧مَنْ سَرَّهُ أن يبسط عليه رزقه، أو ينسأ في أثره،…٢٥٥٨لئن كنت كما قلت، فكأنما تسفهم المل. ولا يزال معك…٢٥٥٩ولا تقاطعوا٢٥٦٠لا يحل لمسلم أن يهجر أخاه فوق ثلاث ليال. يلتقيان…٢٥٦١لا يحل للمؤمن أن يهجر أخاه فوق ثلاثة أيام٢٥٦٢لا هجرة بعد ثلاث

باب تحريم الظن والتجسس والتنافس والتناجش، ونحوها

٢٥٦٣إياكم والظن. فإن الظن أكذب الحديث. ولا تحسسوا، ولا…٢٥٦٤لا تحاسدوا. ولا تناجشوا، ولا تباغضوا، ولا تدابروا،…٢٥٦٥تفتح أبواب الجنة يوم الاثنين، ويوم الخميس. فيغفر…٢٥٦٦أن الله يقول يوم القيامة: أين المتحابون بجلالي.…٢٥٦٧أن رجلا زار أخا له في قرية أخرى. فأرصد الله له،…٢٥٦٨عائد المريض في مخرفة الجنة حتى يرجع٢٥٦٩إِنَّ اللَّهَ عَزَّ وجل يقول، يوم القيامة: يا ابن…٢٥٧٠حدثنا عثمان بن أبي شيبة وإسحاق بن إِبْرَاهِيمَ…٢٥٧١أجل. إني أوعك كما يوعك رجلان منكم٢٥٧٢ما من مسلم يشاك شوكة فما فوقها، إلا كتبت له بها…٢٥٧٣ما يصيب المؤمن من وصب، ولا نصب، ولا سقم، ولا حزن،…٢٥٧٤قاربوا وسددوا. ففي كل ما يصاب به المسلم كفارة. حتى…٢٥٧٥مالك؟ يا أم السائب! أو يا أم المسيب! تزفزفين؟٢٥٧٦إني أصرع. وإني أتكشف. فادع الله لي. قال٢٥٧٧يا عبادي! إني حرمت الظلم على نفسي وجعلته بينكم…٢٥٧٨اتقوا الظلم. فإن الظلم ظلمات يوم القيامة. واتقوا…٢٥٧٩إن الظلم ظلمات يوم القيامة٢٥٨٠المسلم أخو المسلم، لا يظلمه ولا يسلمه. من كان في…٢٥٨١أتدرون ما المفلس؟٢٥٨٢لتؤدن الحقوق إلى أهلها يوم القيامة. حتى يقاد للشاة…٢٥٨٣إِنَّ اللَّهَ عَزَّ وجل يملي للظالم. فإذا أخذه لم…٢٥٨٤ما هذا دعوى أهل الجاهلية؟٢٥٨٥المؤمن للمؤمن كالبنيان. يشد بعضه بعضا٢٥٨٦مثل المؤمنين في توادهم وتراحمهم وتعاطفهم، مثل…٢٥٨٧المستبان ما قالا. فعلى البادئ، ما لم يعتد المظلوم٢٥٨٨ما نقصت صدقة من مال وما زاد الله عبدا بعفو إلا…٢٥٨٩أتدرون ما الغيبة؟٢٥٩٠لا يستر الله على عبد في الدنيا، إلا ستره الله يوم…٢٥٩١ائذنوا له. فلبئس ابن العشيرة، أو بئس رجل العشيرة٢٥٩٢من يحرم الرفق، يحرم الخير٢٥٩٣يا عائشة! إن الله رفيق يحب الرفق. ويعطي على الرفق…٢٥٩٤إن الرفق لا يكون في شيء إلا زانه. ولا ينزع من شيء…

باب النهي عن لعن الدواب وغيرها

٢٥٩٥خذوا ما عليها ودعوها. فإنها ملعونة٢٥٩٦لَا تصاحبنا ناقة عليها لعنة٢٥٩٧لا ينبغي لصديق أن يكون لعانا٢٥٩٨لا يكون اللعانون شفعاء ولا شهداء، يوم القيامة٢٥٩٩إني لم أبعث لعانا. وإنما بعثت رحمة٢٦٠٠وما ذاك٢٦٠١اللهم! إنما أنا بشر. فأيما رجل من المسلمين سببته،…٢٦٠٢زكاة وأجرا٢٦٠٣آنت هيه؟ لقد كبرت لا كبر سنك٢٦٠٤اذهب وادع لي معاوية٢٦٠٥ليس الكذاب الذي يصلح بين الناس، ويقول خيرا وينمي…٢٦٠٦ألا أنبئكم ما العضة؟ هي النميمة القالة بين الناس٢٦٠٧أن الصدق يهدي إلى البر. وإن البر يهدي إلى الجنة.…٢٦٠٨ما تعدون الرقوب فيكم؟٢٦٠٩ليس الشديد بالصرعة. إنما الشديد الذي يملك نفسه عند…٢٦١٠إني لأعرف كلمة لو قالها لذهب عنه الذي يجد: أعوذ…٢٦١١لما صور الله آدم في الجنة تركه ما شاء الله أن…٢٦١٢إذا قاتل أحدكم أخاه، فليتجنب الوجه٢٦١٣إن الله يعذب الذين يعذبون في الدنيا٢٦١٤أمسك بنصالها٢٦١٥إذا مر أحدكم في مجلس أو سوق، وبيده نبل، فليأخذ…٢٦١٦من أشار إلى أخيه بحديدة، فإن الملائكة تلعنه. حتى…٢٦١٧لا يشير أحدكم إلى أخيه بالسلاح. فإنه لا يدري أحدكم…

باب فضل إزالة الأذى عن الطريق

٢٦١٨اعزل الأذى عن طريق المسلمين٢٦١٩دخلت امرأة النار من جراء هرة لها، أو هر. ربطتها.…٢٦٢٠العز إزاره. والكبرياء رداؤه. فمن ينازعني، عذبته٢٦٢١أن رجلا قال: والله! لا يغفر الله لفلان. وإن الله…٢٦٢٢رب أشعث مدفوع بالأبواب، لو أقسم على الله لأبره٢٦٢٣إذا قال الرجل: هلك الناس، فهو أهلكهم٢٦٢٤ما زال جبريل يوصيني بالجار حتى ظننت أنه ليورثنه٢٦٢٥ما زال جبريل يوصيني بالجار، حتى ظننت أنه سيورثه٢٦٢٦لا تحقرن من المعروف شيئا، ولو أن تلقى أخاك بوجه طلق٢٦٢٧اشفعوا فلتؤجروا. وليقض الله على لسان نبيه ما أحب٢٦٢٨إنما مثل الجليس الصالح والجليس السوء، كحامل المسك…٢٦٢٩مَنْ ابتلي من البنات بشيء، فأحسن إليهن، كن له سترا…٢٦٣٠إن الله قد أوجب لها بها الجنة. أو أعتقها بها من…٢٦٣١من عال جاريتين حتى تبلغا، جاء يوم القيامة أنا وهو٢٦٣٢لا يموت لأحد من المسلمين ثلاثة من الولد فتمسه…٢٦٣٣اجتمعن يوم كذا وكذا٢٦٣٤ثلاثة لم يبلغوا الحنث٢٦٣٥صغارهم دعاميص الجنة يتلقى أحدهم أباه، - أو قال…٢٦٣٦دفنت ثلاثة؟٢٦٣٧إن الله، إذا أحب عبدا، دعا جبريل فقال: إني أحب…٢٦٣٨الأرواح جنود مجندة. فما تعارف منها ائتلف. وما…٢٦٣٩ما أعددت لها؟٢٦٤٠المرء مع من أحب٢٦٤١حدثنا أبو بكر بن أبي شيبة وأبو كُرَيْبٍ. قَالَا:…٢٦٤٢تلك عاجل بشرى المؤمن
البخاري مسلم أبو داود الترمذي النسائي ابن ماجه الموطأ الدارمي الأدب المفرد رياض الصالحين
