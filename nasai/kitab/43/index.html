٢٥٧ حديثًا

باب الحث على الكسب

٤٤٤٩إِنَّ أَطْيَبَ ما أكلَ الرَّجلُ من كَسْبِه، وإِنَّ…٤٤٥٠إِنَّ أولادَكم من أطْيَبِ كَسْبِكم، فكُلوا من…٤٤٥١إِنَّ أَطْيَبَ ما أكل الرَّجلُ من كَسْبِه،…٤٤٥٢إنَّ أطْيَبَ ما أكلَ الرَّجلُ من كَسْبِه، وإِنَّ…

باب اجتناب الشبهات في الكسب

٤٤٥٣إنَّ الحلال بَيِّنٌ، وإنَّ الحرامَ بَيِّنٌ، وإنَّ…٤٤٥٤يأتي على النَّاس زمانٌ ما يبالي الرَّجلُ من أينَ…٤٤٥٥يأتي على النَّاس زمانٌ يأكلون الرِّبا، فمَنْ لم…

باب التجارة

٤٤٥٦إنَّ من أشراط السَّاعة أن يَفْشُوَ المالُ…

باب ما يجب على التجار من التوفية في مبايعتهم

٤٤٥٧البَيِّعان بالخِيار ما لم يَفْتَرقا، فإنْ صَدَقا…

باب المنفق سلعته بالحلف الكاذب

٤٤٥٨ثلاثةٌ لا يُكَلِّمُهُمُ اللهُ يومَ القيامة ولا…٤٤٥٩ثلاثةٌ لا ينظُر الله إليهم يومَ القيامة ولا…٤٤٦٠إِيَّاكُم وكَثرةَ الحَلِف في البيع، فإنَّه…٤٤٦١الحَلِفُ مَنْفَقَةٌ للسَّلعة، مَمْحَقَةٌ للكَسْب

باب الحلف الواجب للخديعة في البيع

٤٤٦٢ثلاثةٌ لا يُكلِّمُهُم الله يومَ القيامة ولا…

باب الأمر بالصدقة لمن لم يعتقد اليمين بقلبه في حال بيعه

٤٤٦٣يا معشرَ التُّجَّار، إنَّه يشهَدُ بيعَكُم الحَلِفُ…

باب وجوب الخيار للمتبايعين قبل افتراقهما

٤٤٦٤البَيِّعان بالخِيار ما لم يفتَرقا، فإنْ بَيَّنا…

باب ذكر الاختلاف على نافع في لفظ حديثه

٤٤٦٥المُتبايِعان كلُّ واحدٍ منهما بالخيار على صاحبه ما…٤٤٦٦البَيِّعان بالخِيار ما لم يتفرَّقا، أو يكون خيارًا٤٤٦٧المُتبايِعان بالخِيار ما لم يتفرَّقا، إلَّا أن…٤٤٦٨إذا تَبايعَ البَيِّعَانِ، فكلُّ واحدٍ منهما…٤٤٦٩البَيَّعان بالخِيار ما لم يتفرَّقا، أو يقول…٤٤٧٠البَيِّعَانِ بالخيار حتَّى يتفرَّقا، أو يكونَ…٤٤٧١البَيِّعان بالخِيار حتَّى يفتَرقا، أو يكونَ بَيعَ…٤٤٧٢إذا تَبايَعَ الرَّجلانِ، فكُلُّ واحدٍ منهما…٤٤٧٣إنَّ المُتبايعَينِ بالخِيار في بَيعِهما ما لم…٤٤٧٤المُتبايِعانِ لا بَيْعَ بينَهما حتَّى يتَفرَّقا…

باب ذكر الاختلاف على عبد الله بن دينار في لفظ هذا الحديث

٤٤٧٥كلُّ بَيِّعين لا بَيعَ بينَهما حتَّى يتفرَّقا،…٤٤٧٦كُلُّ بَيِّعينِ فلا بَيْعَ بينَهما حتَّى…٤٤٧٧كلُّ بَيِّعَينِ لا بَيْعَ بينهما حتَّى يَتفرَّقا،…٤٤٧٨كلُّ بَيْعَينِ لا بَيْعَ بينَهما حتَّى يَتفرَّقا،…٤٤٧٩كُلُّ بَيِّعَينِ فَلا بَيْعَ بينَهما حتَّى…٤٤٨٠البَيِّعان بالخيار ما لم يَتفرَّقا، أو يكونَ…٤٤٨١أخبرنا عَمرو بنُ عليٍّ قال: حدَّثنا معاذ بنُ هشام…٤٤٨٢البَيِّعان بالخِيار ما لم يَتفرَّقا، ويأخذَ…

باب وجوب الخيار للمتبايعين قبل افتراقهما بأبدانهما

٤٤٨٣المُتبايِعان بالخِيار ما لم يَتفرَّقا، إلا أن يكون…

باب الخديعة في البيع

٤٤٨٤إذا بِعْتَ فَقُلْ: لا خِلابةَ٤٤٨٥إذا بِعْتَ فقل: لا خِلابة

باب المحفلة

٤٤٨٦إذا باعَ أحدُكم الشَّاةَ أو اللِّقْحَةَ فلا…

باب النهي عن المصراة

٤٤٨٧لا تَلَقَّوا الرُّكبانَ للبيع، ولا تُصَرُّوا…٤٤٨٨مَنِ اشترى مُصَرَّاةً، فإن رَضِيَها إذا حلَبَها…٤٤٨٩مَنِ ابتاعَ مُحفَّلَةً أو مُصَرَّاةً، فهو بالخيار…

باب الخراج بالضمان

٤٤٩٠أخبرنا إسحاق بنُ إبراهيم قال: حدَّثنا عيسى بنُ…

باب بيع المهاجر للأعرابي

٤٤٩١أخبرني عبد الله بنُ محمد بن تميم قال: حدَّثنا…

باب بيع الحاضر للبادي

٤٤٩٢أخبرنا محمد بنُ بشَّار قال: حدَّثني محمد بنُ…٤٤٩٣أخبرنا محمد بنُ المثنَّى قال: حدَّثني سالم بنُ…٤٤٩٤أخبرنا محمد بنُ عبد الأعلى قال: حدَّثنا خالد قال:…٤٤٩٥لا يبيعُ حاضِرٌ لِبادٍ، دَعوا النَّاسَ يرزقُ الله…٤٤٩٦لا تَلَقَّوا الرُّكبانَ للبيع، ولا يَبِعْ بعضُكم…٤٤٩٧أخبرنا عبد الرَّحمن بنُ عبد الله بن عبد الحكم بن…

باب التلقي

٤٤٩٨أخبرنا عُبيد الله بنُ سعيد قال: حدَّثنا يحيى، عن…٤٤٩٩أخبرنا إسحاق بنُ إبراهيم قال: قلتُ: لأبي أسامة:…٤٥٠٠حاضِرٌ لِبادٍ٤٥٠١لا تَلَقَّوا الجَلَبَ، فَمَنْ تَلقَّاه فاشترى منه،…

باب سوم الرجل على سوم أخيه

٤٥٠٢لا يَبيعَنَّ حاضِرٌ لِبادٍ، ولا تناجَشوا، ولا…

باب بيع الرجل على بيع أخيه

٤٥٠٣لا يَبيعُ أحدُكم على بَيْعِ أخيه٤٥٠٤لا يَبيعُ الرَّجلُ على بَيْعِ أخيه حتَّى يَبْتَاعَ…

باب النجش

٤٥٠٥أخبرنا قُتيبة، عن مالك، عن نافع عن ابن عمر، أنَّ…٤٥٠٦لا يَبيعُ الرَّجلُ على بَيْعِ أخيه، ولا يَبيعُ…٤٥٠٧لا يَبيعٌ حاضِرٌ لِبادٍ، ولا تَناجَشوا، ولا يزيدُ…

باب البيع فيمن يزيد

٤٥٠٨أخبرنا إسحاق بنُ إبراهيم قال: حدَّثنا المُعْتَمِر…

باب بيع الملامسة

٤٥٠٩أخبرنا محمد بنُ سَلَمة والحارثُ بنُ مسكين - قراءةً…

باب تفسير ذلك

٤٥١٠أخبرنا إبراهيم بنُ يعقوب بن إسحاق قال: حدَّثنا عبد…

باب بيع المنابذة

٤٥١١أخبرنا يونس بنُ عبد الأعلى والحارث بنُ مسكين -…٤٥١٢أخبرنا الحُسين بنُ حُرَيث المَرْوَزِيُّ قال:…

باب تفسير ذلك

٤٥١٣أخبرنا محمد بنُ المُصفَّى بن بُهْلول، عن محمد بنُ…٤٥١٤أخبرنا أبو داودَ قال: حدَّثنا يعقوب بنُ إبراهيم…٤٥١٥أخبرنا محمد بنُ رافع قال: حدَّثنا عبد الرَّزَّاق…٤٥١٦أخبرنا هارون بنُ زيد بن أبي الزَّرقاء قال: حدَّثنا…٤٥١٧أخبرنا محمد بن عبد الأعلى قال: حدَّثنا المُعْتَمِر…

باب بيع الحصاة

٤٥١٨أخبرنا عُبيد الله بنُ سعيد قال: حدَّثنا يحيى، عن…

باب بيع الثمر قبل أن يبدو صلاحه

٤٥١٩لا تَبيعوا الثَّمرَ حتَّى يَبْدُوَ صلاحُه٤٥٢٠أخبرنا قُتيبة بنُ سعيد قال: حدَّثنا سفيان، عن…٤٥٢١لا تَبايَعوا الثَّمَرَ حتَّى يَبْدُوَ صلاحُه، ولا…٤٥٢٢لا تَبيعوا الثَّمرَ حتَّى يَبدُوَ صلاحُه٤٥٢٣حدَّثنا محمد بنُ منصور قال: حدَّثنا سفيان، عن ابن…٤٥٢٤أخبرنا قُتيبة قال: حدَّثنا المُفَضَّل، عن ابن…٤٥٢٥أخبرنا محمد بنُ عبد الأعلى قال: حدَّثنا خالد قال:…

باب شراء الثمار قبل أن يبدو صلاحها على أن يقطعها ولا يتركها إلى أوان إدراكها

٤٥٢٦حتَّى تَحمَرَّ

باب وضع الجوائح

٤٥٢٧إن بِعْتَ من أخيك ثمَرًا، فأصابَتْه جائحةٌ، فلا…٤٥٢٨مَنْ باعَ ثمرًا، فأصابَتْه جائحةٌ، فلا يأخُذُ من…٤٥٢٩أخبرنا محمد بنُ عبد الله بن يزيد قال: حدَّثنا…٤٥٣٠تصدَّقوا عليه

باب بيع الثمر سنين

٤٥٣١أخبرنا قُتيبة بنُ سعيد قال: حدَّثنا سفيان، عن…

باب بيع الثمر بالتمر

٤٥٣٢أخبرنا قُتيبة بنُ سعيد قال: حدَّثنا سفيان، عن…٤٥٣٣أخبرني زياد بنُ أيّوب قال: حدَّثنا ابن عُلَيَّة…

باب بيع الكرم بالزبيب

٤٥٣٤أخبرنا قُتيبة بنُ سعيد، عن مالك، عن نافع عن ابن…٤٥٣٥أخبرنا قُتيبة بنُ سعيد قال: حدَّثنا أبو الأحْوَص،…٤٥٣٦أخبرنا قُتيبة بنُ سعيد قال: حدَّثنا سفيان عن…٤٥٣٧قال الحارث بنُ مسكين - قراءةً عليه وأنا أسمع - عن…

باب بيع العرايا بخرصها تمرا

٤٥٣٨أخبرنا عُبيد الله بنُ سعيد حدَّثنا يحيى، عن عُبيد…٤٥٣٩حدثنا عيسى بنُ حمَّاد قال: حدَّثنا اللَّيث، عن…

باب بيع العرايا بالرطب

٤٥٤٠أخبرنا أبو داود قال: حدَّثنا يعقوب بنُ إبراهيم…٤٥٤١أخبرنا إسحاق بنُ منصور ويعقوب بنُ إبراهيم -…٤٥٤٢أخبرنا عبد الله بنُ محمد بن عبد الرَّحمن قال:…٤٥٤٣أخبرنا الحُسين بن عيسى قال: حدثنا أبو أسامة قال:…٤٥٤٤أخبرنا قُتيبة بنُ سعيد قال: حدَّثنا اللَّيث، عن…

اشتراء التمر بالرطب

٤٥٤٥أَيَنقُصُ الرُّطَبُ إِذا يَبِسَ؟٤٥٤٦أَينقُصُ إذا يبس؟

باب بيع الصبرة من التمر لا يعلم مكيلها بالكيل المسمى من التمر

٤٥٤٧أخبرنا إبراهيم بنُ الحسن قال: حدَّثنا حجَّاج قال…

باب بيع الصبرة من الطعام بالصبرة من الطعام

٤٥٤٨لا تُبَاعُ الصُّبْرةُ من الطَّعام بالصُّبْرةِ من…

باب بيع الزرع بالطعام

٤٥٤٩أخبرنا قُتيبة بنُ سعيد قال: حدَّثنا اللَّيث، عن…٤٥٥٠حدَّثنا عبد الحميد بنُ محمد قال: حدَّثنا مَخْلَد…

باب بيع السنبل حتى يبيض

٤٥٥١أخبرنا عليُّ بنُ حُجْر قال: حدَّثنا إسماعيل، عن…٤٥٥٢بِعْهُ بالوَرِق، ثُمَّ اشْتَرِ به

باب بيع التمر بالتمر متفاضلا

٤٥٥٣أكُلُّ تمرِ خَيبرَ هكذا؟٤٥٥٤أَنّى لكم هذا؟٤٥٥٥لا صاعَيْ تمرٍ بصاعٍ، ولا صاعَي حِنْطَةٍ بِصاعٍ،…٤٥٥٦لاصاعَي تمر بصاعٍ، ولا صاعَي حِنْطَةٍ بِصاعٍ، ولا…٤٥٥٧ما هذا؟٤٥٥٨الذهبُ - يعني - بالوَرِقِ ربًا، إِلَّا هاءَ وهاء…

باب بيع التمر بالتمر

٤٥٥٩التَّمرُ بالتَّمر، والحِنْطَةُ بالحِنْطةِ،…

باب بيع البر بالبر

٤٥٦٠أخبرنا محمد بنُ عبد الله بن بَزيعٍ قال: حدَّثنا…٤٥٦١أخبرنا المُؤمَّل بنُ هشام قال: حدَّثنا إسماعيل -…

باب بيع الشعير بالشعير

٤٥٦٢أخبرنا إسماعيل بنُ مسعود قال: حدَّثنا بشر بنُ…٤٥٦٣أخبرني محمد بنُ آدم، عن عَبْدَة، عن ابن أبي…٤٥٦٤الذَّهَبُ بالذَّهَبِ تِبْرُه وعَينُه وزنًا بوزن،…٤٥٦٥فإِنَّ الذَّهَبَ بالذَّهبِ، والوَرِقَ بالوَرِقِ -…٤٥٦٦الذَّهَبُ الكِفَّةُ بالكِفَّة

باب بيع الدينار بالدينار

٤٥٦٧الدِّينارُ بالدِّينار، والدِّرهمُ بالدِّرهم، لا…

باب بيع الدرهم بالدرهم

٤٥٦٨أخبرنا قُتيبة بنُ سعيد، عن مالك، عن حُمَيد بن قيس…٤٥٦٩الذَّهب بالذَّهب، وَزْنًا بوَزْن، مِثْلًا بمِثْل،…

باب بيع الذهب بالذهب

٤٥٧٠لا تَبيعوا الذَّهَبَ بالذَّهب، إِلَّا مِثْلًا…٤٥٧١أخبرنا حُمَيد بنُ مَسْعَدة وإسماعيل بنُ مسعود…٤٥٧٢حدَّثنا قُتيبة بنُ سعيد عن مالك، عن زيد بن أسلم،…

باب بيع القلادة فيها الخرز والذهب بالذهب

٤٥٧٣لا تُبَاعُ حتَّى تُفصَّلَ٤٥٧٤افصِلْ بعضها من بعض، ثُمَّ بعها

باب بيع الفضة بالذهب نسيئة

٤٥٧٥ما كان يَدًا بيد فلا بأسَ به، وما كان نَسيئةً فهو…٤٥٧٦إنْ كان يَدًا بيَدٍ فلا بأس، وإن كان نَسيئةً فلا…٤٥٧٧أخبرنا أحمد بنُ عبد الله بن الحَكَم عن محمد قال:…

باب بيع الفضة بالذهب وبيع الذهب بالفضة

٤٥٧٨وفيما قَرأ علينا أحمد بنُ مَنيعٍ قال: حدَّثنا…٤٥٧٩تَبايَعوا الذَّهَبَ بالفِضَّة كيف شِئْتُم،…٤٥٨٠لا رِبا إلَّا في النَّسيئة٤٥٨١إنَّما الرِّبا في النَّسيئة٤٥٨٢لا بأسَ أن تأخُذَها بسعرِ يومِها ما لم تَفرَّقا…

باب أخذ الورق من الذهب، والذهب من الورق، وذكر اختلاف ألفاظ الناقلين لخبر ابن…

٤٥٨٣إذا بايَعْتَ صاحِبَكَ فلا تُفارِقُه وبينَكَ وبينَه…٤٥٨٤أخبرنا محمد بنُ بشَّار قال: حدَّثنا وكيع قال:…٤٥٨٥أخبرنا محمد بن بشَّار قال: حدَّثنا مُؤمَّل قال:…٤٥٨٦أخبرنا محمد بنُ بشَّار قال: حدَّثنا عبد الرَّحمن…٤٥٨٧أخبرنا محمد بن بشَّار قال: حدَّثنا عبد الرَّحمن…٤٥٨٨أخبرنا محمد بنُ بشَّار قال: حدَّثنا وكيع قال:…

باب أخذ الورق من الذهب

٤٥٨٩لا بأس أن تأخُذَ بسعرِ يومِها، ما لم تَفْتَرِقا…

باب الزيادة في الوزن
