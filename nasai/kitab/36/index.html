١٦٧ حديثًا

تحريم الدم

٣٩٦٦أُمِرْتُ أن أُقاتِلَ المشركينَ حتَّى يشهدوا أن لا…٣٩٦٧أُمِرْتُ أن أُقاتِلَ النَّاسَ حتّى يشهدوا أن لا…٣٩٦٨أخبرنا محمد بنُ المُثنَّى قال: حدَّثنا محمد بنُ…٣٩٦٩أُمِرْتُ أن أُقاتِلَ النَّاسَ حتَّى يشهدوا أن لا…٣٩٧٠أُمِرْتُ أن أُقاتِلَ النَّاسَ حتَّى يقولوا: لا…٣٩٧١أُمِرْتُ أن أُقاتِلَ النَّاسَ حتَّى يقولوا: لا…٣٩٧٢أُمِرْتُ أن أُقاتِلَ النَّاسَ حتّى يقولوا: لا إلَه…٣٩٧٣أُمِرْتُ أن أُقاتِلَ النَّاسَ حتَّى يقولوا: لا إله…٣٩٧٤أخبرنا أحمد بنُ محمد بن المغيرةِ قال: حَدَّثَنَا…٣٩٧٥أخبرنا أحمد بنُ سُليمان قال: حدَّثنا مُؤَمَّل بنُ…٣٩٧٦أخبرنا محمد بن عبد الله بن المُبارك قال: حدَّثنا…٣٩٧٧أُمِرْتُ أن أُقاتِلَ النَّاسَ حتَّى يقولوا: لا…٣٩٧٨أخبرنا القاسم بنُ زكريَّا بن دينار قال: حَدَّثَنَا…٣٩٧٩اقتُلوه٣٩٨٠إِنَّه أُوحِيَ إليَّ أن أُقاتِلَ النَّاسَ حتَّى…٣٩٨١أخبرنا أحمد بنُ سليمان قال: حدَّثنا الحسن بنُ محمد…٣٩٨٢اذْهَبْ فاقتُلْه٣٩٨٣أُمِرْتُ أن أُقاتِلَ النَّاسَ حتَّى يشهدوا أن لا…٣٩٨٤كُلُّ ذَنْبٍ عسى اللهُ أن يَغْفِرَه، إِلَّا…٣٩٨٥لا تُقتَلُ نفسٌ ظُلمًا إِلَّا كان على ابن آدمَ…

باب تعظيم الدم

٣٩٨٦والَّذي نفسي بيَدِهِ، لَقَتْلُ مؤمنٍ أعظَمُ عندَ…٣٩٨٧لَزَوالُ الدُّنيا أهوَنُ عندَ اللهِ من قَتْلِ رجلٍ…٣٩٨٨أخبرنا محمد بنُ بشَّار قال: حَدَّثَنَا محمد، عن…٣٩٨٩أخبرنا عَمرو بنُ هشام قال: حدَّثنا مَخْلَدُ بنُ…٣٩٩٠قَتْلُ المؤمنِ أعظَمُ عندَ اللهِ من زوالِ الدُّنيا٣٩٩١أوَّلُ ما يُحاسَبُ به العبدُ الصَّلاةُ، وأوَّلُ ما…٣٩٩٢أوَّلُ ما يُحكَمُ بين النَّاسِ في الدِّماء٣٩٩٣أخبرنا أحمد بنُ سليمان قال: حَدَّثَنَا أبو داود،…٣٩٩٤أخبرنا أحمد بنُ حفص قال: حدَّثني أبي قال: حدَّثني…٣٩٩٥أَوَّلُ ما يُقضى فيه بينَ النَّاسِ يومَ القيامةِ…٣٩٩٦أخبرنا محمد بنُ العلاء قال: حَدَّثَنَا أبو معاوية…٣٩٩٧يَجِيءُ الرَّجلُ آخِذًا بيَدِ الرَّجلِ، فيقول: يا…٣٩٩٨يَجِيءُ المقتولُ بقاتِلِه يومَ القيامةِ فيقول:…٣٩٩٩يَجِيءُ المقتولُ مُتعلِّقًا بالقاتلِ تَشْخَبُ…٤٠٠٠قال: وأخبرني أزْهَرُ بنُ جَميلٍ البصريُّ قال:…٤٠٠١أخبرنا عَمرو بنُ عليٍّ قال: حَدَّثَنَا يحيى قال:…٤٠٠٢أخبرنا محمد بنُ المُثنَّى قال: حدَّثنا محمد قال:…٤٠٠٣أخبرنا حاجب بنُ سليمان المَنْبِجيُّ قال:…٤٠٠٤أخبرنا الحسنُ بن محمد الزَّعفرانيُّ قال: حدَّثنا…٤٠٠٥يَجِيءُ المقتولُ بالقاتلِ يومَ القيامةِ ناصِيَتُه…٤٠٠٦أخبرنا محمد بنُ المُثنَّى قال: حدَّثنا الأنصاريُّ…٤٠٠٧تبارك٤٠٠٨أخبرنا عَمرو بنُ عليٍّ، عن مسلم بن إبراهيم قال:…

باب ذكر الكبائر

٤٠٠٩مَنْ جاءَ يعبدُ الله، ولا يُشرِكُ به شيئًا،…٤٠١٠الكبائر: الشِّركُ بالله، وعُقوقُ الوالِدَين،…٤٠١١الكبائر: الإشراكُ بالله، وعُقوقُ الوالِدَين، وقتلُ…٤٠١٢هُنَّ سَبْعٌ، أعظمُهُنَّ إشراكٌ بالله، وقَتْلُ…

باب ذكر أعظم الذنب، واختلاف يحيى وعبد الرحمن

٤٠١٣أن تجعلَ لله نِدًّا، وهو خَلَقَكَ٤٠١٤أن تجعَلَ للهِ نِدًّا، وهو خَلَقَكَ٤٠١٥الشِّركُ؛ أن تجعَلَ لله نِدًّا، وأن تُزانِيَ…

باب ذكر ما يحل به دم المسلم

٤٠١٦والَّذي لا إلهَ غيرُه، لا يَحِلُّ دَمُ امرئٍ مسلمٍ…٤٠١٧لا يَحِلُّ دمُ امرئٍ مسلمٍ، إلَّا رجلٌ زنى بعدَ…٤٠١٨أخبرنا هلال بنُ العلاء قال: حَدَّثَنَا حُسينٌ قال:…٤٠١٩لا يَحِلُّ دَمُ امرئٍ مسلمٍ إلَّا بإحدى ثلاث:…

باب قتل من فارق الجماعة، وذكر الاختلاف على زياد بن علاقة، عن عرفجة فيه

٤٠٢٠إنَّه سيكونُ بَعدي هَنَاتٌ وهَنَاتٌ، فَمَنْ…٤٠٢١أخبرنا أبو عليٍّ محمدُ بنُ عليّ المَرْوَزيُّ قال:…٤٠٢٢إنَّها ستكونُ بَعدي هَنَاتٌ وهَنَاتٌ، فمَنْ أرادَ…٤٠٢٣أَيُّما رجُلٍ خَرَجَ يُفرِّقُ بينَ أُمَّتِي،…

باب تأويل قول الله عز وجل: ﴿إنما جزاء الذين يحاربون الله ورسوله ويسعون في…

٤٠٢٤ألا تَخْرُجُون مع رَاعِينا في إبلِه فتُصِيبُوا من…٤٠٢٥أخبرني عَمْرُو بنُ عثمانَ بن سعيدِ بن كَثيرِ بن…٤٠٢٦أخبرنا إسحاقُ بن منصورٍ قال: حَدَّثَنَا محمدُ بنُ…٤٠٢٧أخبرنا أحمدُ بنُ سليمانَ قال: حَدَّثَنَا محمدُ بنُ…

ذكر اختلاف الناقلين لخبر حميد عن أنس بن مالك فيه:

٤٠٢٨أخبرنا أحمدُ بنُ عَمْرِو بن السَّرْحِ قال: أخبرني…٤٠٢٩لو خَرَجْتُم إِلى ذَوْدِنا فكُنتم فيها، فشَرِبتُم…٤٠٣٠لو خَرَجْتُمْ إلى ذَوْدِنا، فشَرِبْتُم من ألبانِها٤٠٣١لو خَرَجْتُم إلى ذَوْدٍ لنا فَشَرِبْتُم من ألبانِها٤٠٣٢أخبرنا محمد بنُ عبد الأعلى قال: حَدَّثَنَا يزيد -…٤٠٣٣أخبرنا محمدُ بنُ المُثَنَّى، عن عبدِ الأعلى، نحوَه.٤٠٣٤أخبرنا محمدُ بنُ نافع أبو بكر قال: حَدَّثَنَا…

باب ذكر اختلاف طلحة بن مصرف ومعاوية بن صالح على يحيى بن سعيد في هذا الحديث

٤٠٣٥أخبرني محمدُ بنُ وَهْب قال: حدَّثنا محمدُ بنُ…٤٠٣٦اللَّهُمَّ عَطِّشَ مَنْ عَطَّشَ آلَ محمدٍ اللَّيلة٤٠٣٧أخبرنا محمد بن عبد الله الخَلَنْجيُّ قال: حدَّثنا…٤٠٣٨أخبرنا محمد بنُ المُثَنَّى، عن إبراهيم بن أبي…٤٠٣٩أخبرنا عيسى بنُ حمَّاد قال: أخبرنا اللَّيث، عن…٤٠٤٠أخبرنا أحمد بنُ عَمرو بن السَّرْح قال: أخبرنا ابن…٤٠٤١أخبرنا أحمد بنُ عَمرو بن السَّرْح قال: أخبرني ابن…٤٠٤٢أخبرنا أحمد بنُ عَمرو بن السَّرْح قال: أخبرنا ابن…٤٠٤٣أخبرنا الفضل بنُ سهل الأعرج قال: حَدَّثَنَا بنُ…٤٠٤٤أخبرنا أحمد بنُ عَمرو بن السَّرْح والحارث بنُ…٤٠٤٥أخبرنا يوسف بنُ سعيد قال: حدَّثنا حجَّاج، عن ابن…٤٠٤٦أخبرنا زكريَّا بنُ يحيى قال: حَدَّثَنَا إسحاق بنُ…

باب الحكم في المرتد

باب قتال المسلم

٤١٠٤قِتالُ المسلم كُفْرٌ، وسِبابُه فُسوقٌ٤١٠٥أخبرنا محمد بنُ بشَّار قال: حدَّثنا عبد الرَّحمن…٤١٠٦أخبرنا يحيى بنُ حَكيم قال: حدَّثنا عبد الرَّحمن…٤١٠٧أخبرنا أحمد بنُ حرب قال: حدَّثنا سفيان بنُ عُيينة،…٤١٠٨سِبابُ المسلم فُسوقٌ، وقِتالُه كُفْرٌ٤١٠٩سِبابُ المسلم فُسوقٌ، وقِتالُه كُفْرٌ٤١١٠سِبابُ المسلم فُسوقٌ، وقِتالُه كُفْرٌ٤١١١سِبابُ المسلم فُسوقٌ، وقِتالُه كُفْرٌ٤١١٢أخبرنا قُتيبة بنُ سعيد قال: حدَّثنا جَرير، عن…٤١١٣أخبرنا محمد بنُ العلاء، عن أبي معاوية، عن الأعمش،…

٤١١٦إذا أشارَ المسلمُ على أخيه المسلمِ بالسِّلاحِ،…٤١١٧أخبرنا أحمد بنُ سليمان قال: حدَّثنا يَعْلى قال:…٤١١٨إذا تواجَهَ المُسلِمانِ بسَيْفَيهما، فقتلَ أحدُهما…٤١١٩إذا تواجَهَ المُسلِمانِ بسَيْفَيهما، فقتَلَ…٤١٢٠إذا تواجَهَ المُسلِمانِ بِسَيْفَيهما كلُّ واحدٍ…٤١٢١إذا التقى المُسلِمانِ بسَيْفَيهما، فقتَلَ أحدُهما…٤١٢٢إذا تواجَهَ المُسلِمانِ بسَيْفَيهما، فقتَلَ…٤١٢٣إذا التقى المُسلِمانِ بسَيْفَيهما، فقتَلَ أحدُهما…٤١٢٤إذا تواجَهَ المُسلِمانِ بسَيْفَيهما، فقتلَ أحدُهما…٤١٢٥لا تَرْجِعوا بعدي كُفَّارًا؛ يضرِبُ بعضُكم رِقابَ…٤١٢٦لا تَرْجِعوا بعدي كُفَّارًا؛ يَضْرِبُ بعضُكم…٤١٢٧لا تَرْجِعوا بعدي كُفَّارًا، يَضْرِبُ بعضُكم…٤١٢٨أخبرنا محمد بنُ العلاء قال: حدَّثنا أبو معاوية، عن…٤١٢٩لا تَرْجِعوا بعدي كُفَّارًا٤١٣٠لا تَرْجِعوا بعدي ضُلَّالًا؛ يَصْرِبُ بعضُكم…٤١٣١لا تَرْجِعوا بعدي كُفَّارًا؛ يَضْرِبُ بعضُكم…٤١٣٢اسْتَنْصِتِ النَّاسَ
البخاري مسلم أبو داود الترمذي النسائي ابن ماجه الموطأ الدارمي الأدب المفرد رياض الصالحين

المتن بالعربية كما ورد في طبعاته، دون ترجمة.
