١١٢ حديثًا

ذكر أفضل ما تعوذ به المتعوذون، وذكر اختلاف الناقلين لإسناد الخبر فيه

٥٤٢٨قل٥٤٢٩قُلْ٥٤٣٠يا عُقبة، قُلْ٥٤٣١قُلْ - قلتُ: وما أقول؟ قال -: ﴿قُلْ هُوَ اللَّهُ…٥٤٣٢يا ابنَ عابِسٍ، ألا أدُلُّك - أو قال: ألا أخبِرُكَ…٥٤٣٣اقرأ٥٤٣٤أخبرنا موسى بنُ حِزام التَّرمذيُّ قال: أخبرنا أبو…٥٤٣٥أخبرنا محمد بنُ بشَّار قال: حدَّثنا عبد الرَّحمن…٥٤٣٦يا عُقبة، ألا أُعلِّمُكَ خيرَ سُورَتَين قُرِئتا؟٥٤٣٧ألا تركَبُ يا عُقبة؟٥٤٣٨يا عُقبة، قُلْ٥٤٣٩لَنْ تقرأ شيئًا أبْلَغَ عندَ الله من: ﴿قُلْ…٥٤٤٠أُنزِلَ عليَّ آيَاتٌ لم يُرَ مِثْلُهنَّ: ﴿قُلْ…٥٤٤١اقرَأْ يا جابر

باب الاستعاذة من قلب لا يخشع

٥٤٤٢أخبرنا يزيدُ بنُ سِنان قال: حدَّثنا عبد الرَّحمن…

باب الاستعاذة من فتنة الصدر

باب الاستعاذة من شر السمع والبصر

٥٤٤٤قُل: أعوذُ بِكَ من شَرِّ سمعي، وشرِّ بصري، وشرِّ…

باب الاستعاذة من الجبن

٥٤٤٥اللهمَّ إِنِّي أعوذُ بكَ من البُخل، وأعوذُ بِكَ من…

باب الاستعاذة من البخل

٥٤٤٦أخبرنا محمد بنُ عبد العزيز قال: حدَّثنا الفضل بنُ…٥٤٤٧اللهمَّ إِنِّي أعوذُ بِكَ من البُخل، وأعوذُ بِكَ…٥٤٤٨اللهمَّ إِنِّي أعوذُ بِكَ من العَجْزِ والكسَل،…

باب الاستعاذة من الهم

٥٤٤٩اللهمَّ إنِّي أعوذُ بِكَ من الهَمَّ والحَزَن،…٥٤٥٠اللهمَّ إنِّي أعوذُ بِكَ من الهَمِّ والحَزَن،…٥٤٥١اللهمَّ إِنِّي أعوذُ بِكَ من الكسَلِ والهَرَم،…٥٤٥٢اللهمَّ إِنِّي أعوذُ بِكَ من العَجْزِ والكسَل،…

باب الاستعاذة من الحزن

باب الاستعاذة من المغرم والمأثم

٥٤٥٤إنَّه مَنْ غَرِمَ حدَّث فكذَب، ووعدَ فأخلَفَ

باب الاستعاذة من شر السمع والبصر

٥٤٥٥قُلْ: أعوذُ بِكَ من شَرِّ سمعي، وشرِّ بصري، وشرِّ…

باب الاستعاذة من شر البصر

باب الاستعاذة من الكسل

٥٤٥٧اللهمَّ إِنِّي أعوذُ بِكَ من الكسَلِ والهَرَم،…

باب الاستعاذة من العجز

٥٤٥٨اللهمَّ إِنِّي أعوذُ بِكَ من العَجْزِ والكَسَل،…٥٤٥٩اللهمَّ إِنِّي أعوذُ بِكَ من العَجْزِ والكسَل،…

باب الاستعاذة الذلة

٥٤٦٠اللهمَّ إِنِّي أعوذُ بِكَ من الفقر، وأعوذُ بِكَ من…٥٤٦١تعوَّذوا بالله من الفقرِ والقِلَّةِ والذِّلَّة،…٥٤٦٢أخبرنا أحمد بنُ نصر قال: حدَّثنا عبد الصَّمد بنُ…

باب الاستعاذة من القلة

٥٤٦٣تعوَّذوا بالله من الفقرِ، ومن القِلَّةِ…

باب الاستعاذة من الفقر

٥٤٦٤تَعوَّذوا بالله من الفقرِ والقِلَّةِ والذِّلَّة،…٥٤٦٥أخبرنا محمد بنُ المثنَّى قال: حدَّثنا ابن أبيِّ…

باب الاستعاذة من شر فتنة القبر

٥٤٦٦اللَّهُمَّ إنِّي أعوذُ بك من فتنةِ النَّار وعذابِ…

باب الاستعاذة من نفس لا تشبع

٥٤٦٧اللهمَّ إِنِّي أعوذُ بِكَ من الأربع: من عِلمٍ لا…

باب الاستعاذة من الجوع

٥٤٦٨اللهمَّ إِنِّي أعوذُ بِكَ من الجوع، فإنَّه بِئسَ…

باب الاستعاذة من الخيانة

٥٤٦٩اللهمَّ إنِّي أعوذُ بِكَ من الجوع، فإنَّه بِئسَ…

باب الاستعاذة من الشقاق والنفاق وسوء الأخلاق

٥٤٧٠اللهمَّ إِنِّي أعوذُ بِكَ من الشِّقاقِ، والنِّفاق،…

باب الاستعاذة من علم لا ينفع

٥٤٧١اللهمَّ إِنِّي أعوذُ بِكَ من علمٍ لا ينفَع، وقلبٍ…

باب الاستعاذة من المغرم

٥٤٧٢إِنَّ الرَّجلَ إذا غَرِمَ حدَّث فكذَب، ووعدَ فأخلَف

باب الاستعاذة من الدين

٥٤٧٣أعوذُ باللهِ من الكُفرِ والدَّين٥٤٧٤أعوذُ بالله من الكُفرِ والدَّين

باب الاستعاذة من غلبة الدين

٥٤٧٥اللهمَّ إنِّي أعوذُ بِكَ من غَلَبَةِ الدَّين،…

باب الاستعاذة من ضلع الدين

٥٤٧٦اللهمَّ إِنِّي أعوذُ بِكَ من الهَمِّ والحَزَن،…

باب الاستعاذة من شر فتنة الغنى

٥٤٧٧اللَّهُمَّ إِنِّي أَعُوذُ بِكَ من عذاب القبر،…

باب الاستعاذة من فتنة الدنيا

٥٤٧٨اللهمَّ إنِّي أعوذُ بِكَ من البُخل، وأعوذُ بِكَ من…٥٤٧٩اللهمَّ إِنِّي أعوذُ بِكَ من البُخل، وأعوذُ بِكَ…٥٤٨٠أخبرنا أحمد بنُ فَضالةَ، عن عُبيد الله قال: أخبرنا…٥٤٨١اللهمَّ إنِّي أعوذُ بِكَ من الجُبنِ، والبُخل،…٥٤٨٢أخبرني هلال بنُ العلاء قال: حدَّثنا حُسين قال:…٥٤٨٣أخبرنا أحمد بنُ سليمان قال: حدَّثنا أبو داود، عن…

باب الاستعاذة من شر الذكر

٥٤٨٤قُلْ: اللهمَّ عافِني من شرِّ سمعي وبصري، ولساني…

باب الاستعاذة من شر الكفر

٥٤٨٥اللهمَّ إِنِّي أعوذُ بِكَ من الكفرِ والفقر

باب الاستعاذة من الضلال

٥٤٨٦بسم الله، رَبِّ أعوذُ بِكَ من أن أزِلَّ أو أضِلَّ،…

باب الاستعاذة من غلبة العدو

٥٤٨٧اللهمَّ إنِّي أعوذُ بِكَ من غَلَبةِ الدَّين،…

باب الاستعاذة من شماتة الأعداء

٥٤٨٨اللهمَّ إنِّي أعوذُ بِكَ من غَلَبةِ الدَّين،…

باب الاستعاذة من الهرم

٥٤٨٩اللهمَّ إنِّي أعوذُ بِكَ من الكسَل والهَرَم،…٥٤٩٠اللهمَّ إنِّي أعوذُ بِكَ من الكسَلِ والهَرَم،…

باب الاستعاذة من سوء القضاء

٥٤٩١أخبرنا إسحاقُ بنُ إبراهيم قال: أخبرنا سفيانُ، عن…

باب الاستعاذة من درك الشقاء

٥٤٩٢أخبرنا قُتيبةُ قال: حدَّثنا سفيانُ، عن سُميٍّ، عن…

باب الاستعاذة من الجنون

٥٤٩٣اللهمَّ إنِّي أعوذُ بِكَ من الجُنونِ، والجُذام،…

باب الاستعاذة من عين الجان

٥٤٩٤أخبرنا هلال بنُ العلاء قال: حدَّثنا سعيد بنُ…

باب الاستعاذة من سوء الكبر

٥٤٩٥اللهمَّ إنِّي أعوذُ بِكَ من الكَسلِ والهَرَم،…

باب الاستعاذة من أرذل العمر

٥٤٩٦اللهمَّ إنِّي أعوذُ بِكَ من البُخل، وأعوذُ بِكَ من…

باب الاستعاذة من سوء العمر

٥٤٩٧اللهمَّ إنِّي أعوذُ بِكَ من البُخلِ والجُبن وأعوذُ…

باب الاستعاذة من الحور بعد الكور

٥٤٩٨اللهمَّ إنِّي أعوذُ بِكَ من وَعْثاءِ السَّفَر،…٥٤٩٩اللهمَّ إنِّي أعوذُ بِكَ من وَعْثاءِ السَّفر،…

باب الاستعاذة من دعوة المظلوم

٥٥٠٠أخبرنا يوسف بنُ حمَّاد قال: حدَّثنا بشر بنُ منصور،…

باب الاستعاذة من كآبة المنقلب

٥٥٠١اللهمَّ أنتَ الصَّاحِبُ في السَّفَر، والخليفةُ في…

باب الاستعاذة من جار السوء

٥٥٠٢تَعوَّذوا باللهِ من جارِ السُّوء في دار المُقام،…

باب الاستعاذة من غلبة الرجال

٥٥٠٣التمِسْ لي غُلامًا من غلمانِكم يَخدُمني

باب الاستعاذة من فتنة الدجال

٥٥٠٤إنّكم تُفتنونَ في قُبورِكم

باب الاستعاذة من عذاب جهنم، وشر المسيح الدجال

٥٥٠٥أعوذُ بالله من عذابِ جهنَّم، وأعوذُ بالله من عذابِ…٥٥٠٦أخبرنا يحيى بنُ دُرُسْتَ قال: حدَّثنا أبو إسماعيل…

باب الاستعاذة من شر شياطين الإنس

٥٥٠٧يا أبا ذرٍّ، تَعوَّذُ بالله من شرِّ شياطين الجِنِّ…

باب الاستعاذة من فتنة المحيا

٥٥٠٨عُوذوا بالله من عذابِ القبر، عُوذوا بالله من فتنة…٥٥٠٩عُوذوا بالله من عذاب القبر، ومن عذابِ جَهنَّم، ومن…٥٥١٠مَنْ أطاعَني فقد أطاعَ الله، ومَنْ عصاني فقد عصى…٥٥١١استَعيذوا بالله من خمس: من عذابِ جهنَّم، وعذابِ…

باب الاستعاذة من فتنة الممات

٥٥١٢قولوا: اللهمَّ إنَّا نعوذُ بِكَ من عذاب جهنَّم،…٥٥١٣عُوذوا بالله من عذابِ الله، عُوذوا بالله من فتنة…

باب الاستعاذة من عذاب القبر

٥٥١٤اللهمَّ إنِّي أعوذُ بكَ من عذابِ جهنَّم، وأعوذُ…

باب الاستعاذة من فتنة القبر

٥٥١٥اللهمَّ إنِّي أعوذُ بِكَ من فتنةِ القبر، وفتنةِ…

باب الاستعاذة من عذاب الله

٥٥١٦عُوذوا بالله من عذابِ الله، عُوذوا بالله من عذابِ…

باب الاستعاذة من عذاب جهنم

٥٥١٧أخبرنا إسحاقُ بنُ إبراهيم قال: أخبرنا أبو عامر…

باب الاستعاذة من عذاب النار

٥٥١٨تَعوَّذوا بالله من عذابِ النَّار، وعذابِ القبر،…

باب الاستعاذة من حر النار

٥٥١٩اللهمَّ رَبَّ جبريل وميكائيل، وربَّ إسرافيل، أعوذُ…٥٥٢٠اللهمَّ إنِّي أعوذُ بِكَ من فتنةِ القبر، ومن فتنة…٥٥٢١مَنْ سألَ الله الجنَّةَ ثلاثَ مرَّات، قالتِ…

باب الاستعاذة من شر ما صنع، وذكر الاختلاف على عبد الله بن بريدة فيه

٥٥٢٢إنَّ سيِّدَ الاستغفار أن يقول العبدُ: اللهمَّ أنتَ…

باب الاستعاذة من شر ما عمل، وذكر الاختلاف على هلال

٥٥٢٣اللهمَّ إِنِّي أعوذُ بِكَ من شرِّ ما عمِلْتُ، ومن…٥٥٢٤اللهمَّ إنِّي أعوذُ بِكَ من شرِّ ما عمِلْتُ، ومن…٥٥٢٥أعوذُ بِكَ من شرِّ ما عمِلْتُ، ومن شرِّ ما لم أعمَلْ٥٥٢٦اللهمَّ إنِّي أعوذُ بِكَ من شرِّ ما عمِلْتُ، ومن…

باب الاستعاذة من شر ما لم يعمل

٥٥٢٧اللهمَّ إِنِّي أعوذُ بِكَ من شرِّ ما عمِلْتُ، ومن…٥٥٢٨اللهمَّ إنِّي أعوذُ بِكَ من شرِّ ما عمِلْتُ، ومن…

باب الاستعاذة من الخسف

٥٥٢٩اللهمَّ إنِّي أعوذُ بعظمَتِكَ أَن أُغتالَ من تحتي٥٥٣٠اللهمَّ …

باب الاستعاذة من التردي والهدم

٥٥٣١اللهمَّ إنِّي أعوذُ بِكَ من التَّردِّي والهَدْم،…٥٥٣٢اللَّهُمَّ إنِّي أعوذُ بِكَ من الهَرَمِ…٥٥٣٣اللهمَّ إنِّي أعوذُ بِكَ مِن الهَدْم، وأعوذُ بِكَ…

باب الاستعاذة برضاء الله من سخط الله تعالى

٥٥٣٤أعُوذُ بعَفْوِكَ من عِقَابِكَ، وأعُوذُ برِضَاكَ من…

باب الاستعاذة من ضيق المقام يوم القيامة
