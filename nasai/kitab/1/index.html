٣٢٤ حديثًا

باب تأويل قوله ﷿: ﴿إذا قمتم إلى الصلاة فاغسلوا وجوهكم وأيديكم إلى المرافق﴾

١إذا اسْتَيْقَظَ أَحَدُكُمْ من نَوْمِه، فلا…

باب السواك إذا قام من الليل

٢أخبرنا إسحاقُ بنُ إبراهيم وقُتيبةُ بنُ سعيد، عن…

باب كيف يستاك

٣أخبرنا أحمدُ بنُ عَبْدة قال: حدَّثنا حمَّادُ بنُ…

باب هل يستاك الإمام بحضرة رعيته

٤إِنَّا لا - أو: لَنْ - نستعينُ على العملِ مَنْ…

باب الترغيب في السواك

٥السَّواكُ مَظْهَرَةٌ للفَمِ، مَرْضاةٌ للرَّبِّ

باب الإكثار في السواك

٦قد أَكْثَرْتُ عليكم في السِّواك

باب الرخصة في السواك بالعشي للصائم

٧لولا أَنْ أَشُقَّ على أمَّتي لأمَرْتُهُمْ…

باب السواك في كل حين

٨أخبرنا عليُّ بنُ خَشْرَم قال: حدَّثنا عيسى - وهو…

باب ذكر الفطرة: الاختتان

٩الفِطْرَةُ خَمس: الاختتانُ، والاسْتِحْدادُ، وقَصُّ…

باب تقليم الأظفار

١٠خمسٌ من الفِطْرَة: قَصُّ الشَّارب، ونَتْفُ الإبْط،…

باب نتف الإبط

١١الفِطْرَةُ خمسٌ: الخِتان، وحَلْقُ العانة، ونَتْفُ…

باب حلق العانة

١٢الفِطْرةُ قَصُّ الأظفار، وأخْذُ الشَّارب، وحَلْقُ…

باب قص الشارب

١٣مَنْ لم يأخُذْ شاربَه فليس منَّا

باب التوقيت في ذلك

١٤أخبرنا قُتيبةُ قال: حدَّثنا جعفرٌ - هو ابنُ…

باب إحفاء الشارب وإعفاء اللحى

١٥أخبرنا عُبيدُ الله بنُ سعيد قال: حدَّثنا يحيى - هو…

باب الإبعاد عند إرادة الحاجة

١٦أخبرنا عَمْرُو بنُ عليٍّ قال: حدَّثنا يحيى بنُ…١٧ائْتِني بوَضُوء

باب الرخصة في ترك ذلك

١٨أخبرنا إسحاقُ بنُ إبراهيم، أخبرنا عيسى بنُ يونس،…

باب القول عند دخول الخلاء

١٩اللَّهُمَّ إنِّي أعوذُ بك من الخُبُثِ والخَبائث

باب النهي عن استقبال القبلة عند الحاجة

٢٠إذا ذَهَبَ أحدُكم إلى الغائط أو البول فلا يستقبلِ…

باب النهي عن استدبار القبلة عند الحاجة

٢١لا تَسْتَقْبِلُوا القِبْلَةَ ولا تَسْتَدْبِرُوها…

باب الأمر باستقبال المشرق أو المغرب عند الحاجة

٢٢إذا أتى أحدُكُمُ الغائطَ، فلا يستقبل القِبْلَةَ،…

الرخصة في ذلك في البيوت

٢٣أخبرنا قُتيبةُ بنُ سعيد، عن مالك، عن يحيى بن سعيد،…

باب النهي عن مس الذكر باليمين عند الحاجة

٢٤إذا بالَ أحدُكم فلا يأخُذْ ذَكَرَهُ بيمينه٢٥إذا دَخَلَ أحدُكُم الخَلاءَ فلا يَمَسَّ ذَكَرَهُ…

باب الرخصة في البول في الصحراء قائما

٢٦أخبرنا مؤمَّلُ بنُ هشام قال: حدَّثنا إسماعيلُ قال:…٢٧أخبرنا محمدُ بنُ بشَّار قال: حدَّثنا محمدٌ قال:…٢٨أخبرنا سليمانُ بنُ عُبيد الله، حدَّثنا بَهْزٌ،…

باب البول في البيت جالسا

٢٩أخبرنا عليُّ بنُ حُجْر، أخبرنا شَرِيك، عن…

باب البول إلى السترة يستتر بها

٣٠أَوَمَا عَلِمْتَ ما أصابَ صاحبَ بني إسرائيل؟ كانوا…

باب التنزه عن البول

٣١إنَّهما يُعَذَّبان، وما يُعَذَّبانِ في كبير، أمَّا…

باب البول في الإناء

٣٢أخبرنا أيوبُ بنُ محمد الوزَّان قال: حدَّثنا…

باب البول في الطست

٣٣أخبرنا عَمْرُو بن عليٍّ، حدَّثنا أَزْهَرُ قال:…

باب كراهية البول في الجحر

٣٤لا يَبُولَنَّ أَحَدُكُمْ في جُحْر

باب النهي عن البول في الماء الراكد

٣٥أخبرنا قُتيبةُ بن سعيدٍ قال: حدَّثنا اللَّيث، عن…

باب كراهية البول في المستحم

٣٦لا يَبُولَنَّ أحدُكُم في مُسْتَحَمِّه؛ فإنَّ…

باب السلام على من يبول

٣٧أخبرنا محمودُ بنُ غَيْلان، حدَّثنا زيدُ بنُ…

باب رد السلام بعد الوضوء

٣٨أخبرنا محمدُ بنُ بَشَّار قال: حدَّثنا معاذُ بنُ…

باب النهي عن الاستطابة بالعظم

٣٩أخبرنا أحمدُ بنُ عَمْرو بن السَّرْح قال: أخبرنا…

باب النهي عن الاستطابة بالروث

٤٠إنَّما أنا لكم مِثْلُ الوالد أُعَلِّمُكُم؛ إذا…

باب النهي عن الاكتفاء في الاستطابة بأقل من ثلاثة أحجار

٤١أخبرنا إسحاقُ بنُ إبراهيمَ قال: أخبرنا أبو معاويةَ…

باب الرخصة في الاستطابة بحجرين

٤٢هذه رِكْسٌ

باب الرخصة في الاستطابة بحجر واحد

٤٣إذا اسْتَجْمَرْتَ فَأَوْتِرْ

باب الاجتزاء في الاستطابة بالحجارة دون غيرها

٤٤إذا ذَهَبَ أَحَدُكُم إلى الغائط، فَلْيَذْهَبْ معه…

باب الاستنجاء بالماء

٤٥أخبرنا إسحاقُ بنُ إبراهيمَ قال: أخبرنا النَّضْرُ…٤٦أخبرنا قُتيبةُ بنُ سعيدٍ قال: حدَّثنا أبو عَوَانة،…

باب النهي عن الاستنجاء باليمين

٤٧إِذا شَرِبَ أَحَدُكُم فلا يَتَنَفَّسْ في إنائه،…٤٨أخبرنا عبدُ الله بنُ محمد بن عبد الرَّحمن قال:…٤٩لا يستنجي أحَدُكُمْ بدونِ ثلاثةِ أحجار

باب دلك اليد بالأرض بعد الاستنجاء

٥٠أخبرنا محمدُ بنُ عبد الله بن المُبارك قال: حدَّثنا…٥١يا جرير هاتِ طَهُورًا

باب التوقيت في الماء

٥٢إذا كانَ الماءُ قُلَّتَيْنِ لم يَحْمِل الخَبَثَ

باب ترك التوقيت في الماء

٥٣دَعُوهُ، لا تُزْرِمُوه٥٤أخبرنا قُتيبةُ قال: حدَّثنا عَبِيدة، عن يحيى بن…٥٥اتْرُكُوه٥٦دَعُوهُ، وأَهْرِيقُوا على بَولِه دَلْوًا من ماء؛…

باب الماء الدائم

٥٧لا يَبُولَنَّ أَحَدُكُمْ في الماء الدَّائم، ثم…٥٨لا يَبُولَنَّ أَحَدُكُمْ في الماءِ الدَّائم، ثم…

باب ماء البحر

٥٩هو الطَّهُورُ ماؤُهُ، الحِلُّ ميتتُهُ

باب الوضوء بالثلج

٦٠أقول: اللَّهُمَّ باعِد بيني وبين خَطاياي كما…

باب الوضوء بماء الثلج

٦١اللَّهُمَّ اغْسِلْ خَطَايَايَ بماء الثَّلج…

باب الوضوء بماء البرد

٦٢اللَّهُمَّ اغْفِرْ لَه وَارْحَمْهُ، وَعَافِهِ…

باب سؤر الكلب

٦٣إذا شربَ الكلبُ في إناء أحَدِكُمْ فَلْيَغْسِلْهُ…٦٤إذا وَلَغَ الكلبُ في إناءِ أحَدِكُم فَلْيَغْسِلْهُ…٦٥أخبرني إبراهيمُ بنُ الحَسَن قال: حدَّثنا حجَّاجٌ…

باب الأمر بإراقة ما في الإناء إذا ولغ فيه الكلب

٦٦إِذا وَلَغَ الكلبُ في إناءِ أحَدِكُمْ،…

باب تعفير الإناء الذي ولغ فيه الكلب بالتراب

٦٧إذا وَلَغَ الكلبُ في الإناء فاغْسِلُوهُ سبعَ…

باب سؤر الهرة

٦٨إنَّها ليست بنَجَس، إنَّما هي من الطَّوَّافِينَ…

باب سؤر الحمار

٦٩إِنَّ اللهَ ورسولَهُ يَنْهَيانِكُمْ عن لُحُوم…

باب سؤر الحائض

٧٠أخبرنا عَمْرُو بنُ عليٍّ قال: حدَّثنا عبد…

باب وضوء الرجال والنساء جميعا

٧١أخبرني هارونُ بنُ عبدِ الله قال: حدَّثنا مَعْنٌ…

باب فضل الجنب

٧٢أخبرنا قُتيبةُ بنُ سعيد قال: حدَّثنا اللَّيث، عن…

باب القدر الذي يكتفي به الرجل من الماء للوضوء

٧٣أخبرنا عَمْرُو بنُ عليٍّ قال: حدَّثنا يحيى قال:…٧٤أخبرنا محمدُ بنُ بشَّار قال: حدَّثنا محمد. ثم ذكر…

باب النية في الوضوء

٧٥إنَّما الأعمالُ بالنِّيَّة، وإنَّما لامْرِيٍ ما…

باب الوضوء من الإناء

٧٦أخبرنا قتيبة عن مالك، عن إسحاقَ بن عبد الله بن أبي…٧٧حَيَّ على الطَّهُور، والبَرَكةُ من الله ﷿

باب التسمية عند الوضوء

٧٨هل معَ أحَدٍ منكُم ماءٌ؟

باب صب الخادم الماء على الرجل للوضوء

٧٩أخبرنا سليمانُ بنُ داودَ والحارثُ بنُ مسكينٍ…

باب الوضوء مرة مرة

٨٠أخبرنا محمدُ بنُ المُثَنَّى قال: حدَّثنا يحيى، عن…

باب الوضوء ثلاثا ثلاثا

٨١أخبرنا سُوَيْدُ بنُ نَصْر قال: أخبرنا عبدُ الله…

غسل الكفين

٨٢أمَعَك ماءٌ؟

باب كم تغسلان

٨٣أخبرنا حُمَيْدُ بنُ مَسْعَدَة، عن سفيان - وهو ابنُ…

باب المضمضة والاستنشاق

٨٤مَنْ تَوَضَّأَ نحوَ وُضُوئي هذا، ثمَّ صَلَّى…

باب بأي اليدين يتمضمض

٨٥مَنْ تَوضَّأَ مثلَ وُضُوئي هذا، ثم قام فصلَّى…

باب إيجاب الاستنشاق

٨٦إذا تَوضَّأَ أَحَدُكُم، فَلْيَجْعَلْ في أنفِهِ…

باب المبالغة في الاستنشاق

٨٧أسْبغِ الوُضُوء، وبالِغ في الاستنشاق إلَّا أن تكون…

باب الأمر بالاستنثار

٨٨مَنْ تَوضَّأَ فَلْيَسْتَنْثِرْ، وَمَن اسْتَجْمَرَ…٨٩إذا تَوَضَّأتَ فاسْتَنْثِرُ، وإذا اسْتَجْمَرْتَ…

باب الأمر بالاستنثار عند الاستيقاظ من النوم

٩٠إذا اسْتَيْقَظَ أَحَدُكُم من مَنَامِه فتَوضَّأَ،…

باب بأي اليدين يستنثر

٩١أخبرنا موسى بنُ عبد الرَّحمن قال: حدَّثنا حُسينُ…

باب غسل الوجه

٩٢أخبرنا قُتيبةُ قال: حدَّثنا أبو عَوَانة، عن خالد…

باب عدد غسل الوجه

٩٣أخبرنا سويدُ بنُ نَصْر قال: أخبرنا عبدُ الله -وهو…

باب غسل اليدين

٩٤أخبرنا عَمْرُو بنُ عليٍّ وحُمَيْدُ بنُ مَسْعَدَةَ،…

باب صفة الوضوء

٩٥أخبرنا إبراهيمُ بنُ الحَسَن المِقْسَمِيُّ قال:…

باب عدد غسل اليدين

٩٦أخبرنا قُتيبةُ بنُ سعيد قال: حدَّثنا أبو…

باب حد الغسل

٩٧أخبرنا محمدُ بنُ سَلَمَةَ والحارثُ بن مسكين قراءةً…

باب صفة مسح الرأس

٩٨أخبرنا عُتبةُ بنُ عبد الله عن مالك - هو ابنُ أنس -…

عدد مسح الرأس

٩٩أخبرنا محمدُ بنُ منصور قال: حدَّثنا سفيان عن…

باب مسح المرأة رأسها

١٠٠أخبرنا الحُسينُ بنُ حُريث قال: حدَّثنا الفَضْلُ…

باب مسح الأذنين

١٠١أخبرنا الهيثمُ بنُ أَيُّوبَ الطَّالْقَانِيُّ قال:…

باب مسح الأذنين مع الرأس وما يستدل به على أنهما من الرأس

١٠٢أخبرنا مجاهدُ بنُ موسى قال: حدَّثنا عبد الله بنُ…١٠٣أخبرنا قُتيبةُ بن سعيد وعُتبةُ بنُ عبد الله، عن…

باب المسح على العمامة

١٠٤أخبرنا الحُسينُ بنُ منصور قال: حدَّثنا أبو معاوية،…١٠٥وأخبرنا الحُسينُ بنُ عبد الرَّحمن الجَرْجَرائيُّ،…١٠٦أخبرنا هنَّادُ بنُ السَّريّ، عن وكيع، عن شعبة، عن…

باب المسح على العمامة مع الناصية

١٠٧أخبرنا عَمْرُو بنُ عليٍّ قال: حدَّثنا يحيى بنُ…١٠٨أَمَعَكَ ماءٌ؟

باب كيف المسح على العمامة

١٠٩أخبرنا يعقوبُ بنُ إبراهيم قال: حدَّثنا هُشَيْمٌ…

باب إيجاب غسل الرجلين

١١٠وَيْلٌ للعَقِبِ من النَّار١١١وَيْلٌ للأَعْقاب من النَّارِ، أَسْبِغُوا الوُضوء

باب بأي الرجلين يبدأ بالغسل

١١٢أخبرنا محمدُ بنُ عبد الأعلى قال: حدَّثنا خالدٌ…

باب غسل الرجلين باليدين

١١٣أخبرنا محمدُ بنُ بشَّار قال: حدَّثنا محمدٌ قال:…

باب الأمر بتخليل الأصابع

١١٤إذا تَوَضَّأْتَ فَأَسْبعَ الوُضُوء، وخَلِّلْ بين…

باب عدد غسل الرجلين

١١٥أخبرنا محمدُ بنُ آدم، عن ابن أبي زائدة قال:…

باب حد الغسل

١١٦مَنْ توضَّأ نحوَ وُضوئي هذا، ثمَّ قام فركع ركعتين…

باب الوضوء في النعل

١١٧أخبرنا محمدُ بنُ العَلاء قال: حدَّثنا ابن إدريس،…

باب المسح على الخفين

١١٨أخبرنا قُتيبةُ بنُ سعيد، حدَّثنا حَفْص، عن الأعمش،…١١٩أخبرنا العبَّاسُ بنُ عبد العظيم قال: حدَّثنا عبد…١٢٠أخبرنا عبدُ الرَّحمن بنُ إبراهيمَ دُحَيْمٌ…١٢١أخبرنا سليمانُ بن داودَ والحارثُ بنُ مسكين قراءةً…١٢٢أخبرنا قُتيبةُ قال: حدَّثنا إسماعيلُ - وهو ابنُ…١٢٣أخبرنا عليُّ بنُ خَشْرَم قال: حدَّثنا عيسى، عن…١٢٤أخبرنا قُتيبةُ بنُ سعيد قال: حدَّثنا اللَّيثُ بن…

باب المسح على الخفين في السفر

١٢٥تَخَلَّفْ يا مُغيرة، وامْضُوا أيُّها النَّاس

باب التوقيت في المسح على الخفين للمسافر

١٢٦أخبرنا قُتيبةُ بنُ سعيد قال: حدَّثنا سفيان، عن…١٢٧أخبرنا أحمد بن سليمانَ الرُّهاويُّ قال: حدَّثنا…

باب التوقيت في المسح على الخفين للمقيم

١٢٨أخبرنا إسحاقُ بن إبراهيم، أخبرنا عبدُ الرَّزَّاق،…١٢٩أخبرنا هنَّادُ بنُ السَّرِيّ، عن أبي معاوية، عن…

باب صفة الوضوء من غير حدث

١٣٠أخبرنا عَمْرُو بنُ يزيد قال: حدَّثنا بَهْرُ بنُ…

٢٦٧إنِّي رأيتُكَ فَحِدْتَ عَنِّي٢٦٨إنَّ المُسلمَ لا يَنْجُس٢٦٩أينَ كنتَ يا أبا هريرة؟

باب استخدام الحائض

٢٧٠يا عائشة، ناوِلِينِي الثَّوب٢٧١ناوِلِينِي الخُمْرَةَ من المسجد٢٧٢أخبرنا إسحاقُ بنُ إبراهيمَ قال: حدَّثنا أبو…

باب بسط الحائض الخمرة في المسجد

٢٧٣أخبرنا محمدُ بنُ منصور، عن سفيان، عن مَنْبُوذ، عن…

باب في الذي يقرأ القرآن ورأسه في حجر امرأته وهي حائض

٢٧٤أخبرنا إسحاقُ بنُ إبراهيمَ وعليُّ بنُ حُجْر…

باب غسل الحائض رأس زوجها

٢٧٥أخبرنا عَمْرُو بنُ عليٍّ، حدَّثنا يحيى، حدَّثنا…٢٧٦أخبرنا محمدُ بنُ سَلَمةَ قال: حدَّثنا ابنُ وَهْب،…٢٧٧أخبرنا قُتيبةُ بنُ سعيد، عن مالك، عن هشام بن…٢٧٨أخبرنا قُتيبةُ بنُ سعيد، عن مالك. (ح): وأخبرنا…

باب مؤاكلة الحائض والشرب من سؤرها

٢٧٩أخبرنا قُتيبةُ قال: حدَّثنا يزيدُ -وهو ابنُ…٢٨٠أخبرنا أيوبُ بنُ محمد الوزَّان قال: حدَّثنا عبدُ…

باب الانتفاع بفضل الحائض

٢٨١أخبرنا محمدُ بنُ منصور قال: حدَّثنا سفيان، عن…٢٨٢أخبرنا محمودُ بنُ غَيْلان قال: حدَّثنا وكيع قال:…

باب مضاجعة الحائض

٢٨٣أَنَفِسْتِ؟٢٨٤أخبرنا محمدُ بن المثنَّى قال: حدَّثنا يحيى بنُ…

باب مباشرة الحائض

٢٨٥أخبرنا قُتيبةُ قال: حدَّثنا أبو الأحْوَص، عن أبي…٢٨٦أخبرنا إسحاقُ بنُ إبراهيم، أخبرنا جَرير، عن منصور،…٢٨٧أخبرنا الحارثُ بنُ مسكين قراءةً عليه وأنا أسمع، عن…

باب تأويل قول الله ﷿: ﴿ويسألونك عن المحيض﴾

٢٨٨أخبرنا إسحاقُ بنُ إبراهيم قال: حدَّثنا سُليمانُ…

باب ما يجب على من أتى حليلته في حال حيضها بعد علمه بنهي الله ﷿ عن وطئها

٢٨٩أخبرنا عَمْرُو بنُ عليٍّ قال: حدَّثنا يحيى، عن…

باب ما تفعل المحرمة إذا حاضت

٢٩٠ما لكِ، أَنْفِسْتِ؟

باب ما تفعل النفساء عند الإحرام

٢٩١اغْتَسِلِي واسْتَثْفِري، ثمَّ أَهِلِّي

باب دم الحيض يصيب الثوب

٢٩٢حُكِّيهِ بضِلَع، واغْسِلِيهِ بماءٍ وسِدْر٢٩٣حُتِّيهِ، ثمَّ اقْرُصِيهِ بالماء، ثمَّ انْضَحِيهِ…

باب المني يصيب الثوب

٢٩٤أخبرنا عيسى بنُ حمَّاد قال: حدَّثنا اللَّيث، عن…

باب غسل المني من الثوب

٢٩٥أخبرنا سُوَيْدُ بنُ نَصْر قال: أخبرنا عبد الله، عن…

باب فرك المني من الثوب

٢٩٦أخبرنا قُتيبةُ بنُ سعيدٍ قال: حدَّثنا حمَّاد، عن…٢٩٧أخبرنا عَمْرُو بنُ يزيدَ قال: حدَّثنا بَهْزٌ قال:…٢٩٨أخبرنا الحُسينُ بنُ حُريث، أخبرنا سفيان، عن منصور،…٢٩٩أخبرنا شعيبُ بنُ يوسف، عن يحيى بن سعيد، عن الأعمش،…٣٠٠أخبرنا قُتيبةُ قال: حدَّثنا حمَّادُ بنُ زيد، عن…٣٠١أخبرنا محمدُ بنُ كامل المَرْوَزيُّ قال: حدَّثنا…

باب بول الصبي الذي لم يأكل الطعام

٣٠٢أخبرنا قُتيبةُ بنُ سعيد، عن مالك، عن ابن شِهاب، عن…٣٠٣أخبرنا قُتيبة، عن مالك، عن هشام بن عُروة، عن أبيه…

باب بول الجارية

٣٠٤يُغْسَلُ من بول الجارية، ويُرَشُّ من بول الغلام

باب بول ما يؤكل لحمه

٣٠٥أخبرنا محمدُ بنُ عبد الأعلى قال: حدَّثنا يزيدُ بنُ…٣٠٦أخبرنا محمدُ بنُ وَهْب قال: حدَّثنا محمدُ بنُ…

باب فرث ما يؤكل لحمه يصيب الثوب

٣٠٧اللَّهُمَّ عليك بقُريش

باب البزاق يصيب الثوب

٣٠٨أخبرنا عليُّ بنُ حُجْر قال: حدَّثنا إسماعيل، عن…٣٠٩إذا صَلَّى أحدُكُم فلا يَبْزُقُ بينَ يدَيْه، ولا…

باب بدء التيمم

٣١٠أخبرنا قُتيبةُ بنُ سعيد عن مالك، عن عبد الرَّحمن…

باب التيمم في الحضر

٣١١أخبرنا الرَّبيعُ بنُ سُليمان قال: حدَّثنا شعيبُ…

التيمم في الحضر

٣١٢أخبرنا محمدُ بنُ بشَّار قال: حدَّثنا محمدٌ قال:…٣١٣إنَّما كان يَجْزِيكَ من ذلك التَّيمُّم

باب التيمم في السفر

٣١٤أخبرني محمدُ بنُ يحيى بن عبد الله قال: حدَّثنا…

باب الاختلاف في كيفية التيمم

٣١٥أخبرنا العبَّاسُ بنُ عبد العظيم العَنْبَرِيُّ قال:…

نوع آخر من التيمم والنفخ في اليدين

باب التيمم بالصعيد

٣٢١يا فلان، ما مَنَعَكَ أن تُصلِّيَ مع القوم؟

باب الصلوات بتيمم واحد

٣٢٢الصَّعِيدُ الطَّيِّبُ وَضُوءُ المُسلم، وإنْ لم…
